static class Program {
 static int assertions;
 static void Check(bool ok,string name) { if(!ok) throw new Exception("FAIL "+name); assertions++; }
 static void Main() {
  var owner=new OptionsOwner(); Hero.MainHero=new Hero { Gold=5000 };
  var options=owner.BuildShoutTradeOptions();
  Check(options.Count==1 && options[0].IsGold && options[0].AvailableAmount==5000,"partyless captive retains personal gold");
  Hero.MainHero.Gold=0; Check(owner.BuildShoutTradeOptions().Count==0,"no gold does not fabricate inventory");
  Hero.MainHero=null; Check(owner.BuildShoutTradeOptions().Count==0,"missing main hero is empty");
  Hero.MainHero=new Hero { Gold=10 }; owner._shoutTradeMode="show";
  Check(owner.BuildShoutTradeOptions().Single().AvailableAmount==8,"show allowance preserved without party");
  owner._shoutTradeMode="give"; Hero.MainHero.PartyBelongedTo=new();
  var item=new ItemObject {StringId="grain",Name="grain"};
  Hero.MainHero.PartyBelongedTo.ItemRoster.Add(new() {EquipmentElement=new() {Item=item},Amount=2});
  Hero.MainHero.PartyBelongedTo.ItemRoster.Add(new() {EquipmentElement=new() {Item=item},Amount=3});
  options=owner.BuildShoutTradeOptions();
  Check(options.Count==2 && options[1].AvailableAmount==5 && options[1].InventoryUnitValue==5,"owned items still aggregate and value");
  owner._shoutTradeMode="show";Check(owner.BuildShoutTradeOptions()[1].AvailableAmount==4,"show item allowance preserved");
  owner._shoutTradeMode="troops";Check(owner.BuildShoutTradeOptions().Single().PartyEntry!=null,"troop mode keeps separate port");
  owner._shoutTradeMode="fiefs";Check(owner.BuildShoutTradeOptions().Single().SettlementEntry!=null,"fief mode keeps separate port");
  Console.WriteLine("PASS "+assertions+" production trade options assertions; liveGame=false");
 }
