    internal static class Program
    {
        static void Check(bool pass,string name) { if(!pass)throw new Exception("FAIL "+name);Console.WriteLine("PASS "+name); }
        static CompressedMemoryBlock Block(string id,int day,string title="title")=>new(){Id=id,HeroId="ada",GameDayIndex=day,StartHour=3,EndHour=4,RichTitle=title,Summary="summary "+id,AfefLines=new(){"[AFEF NPC行为补充] fact "+id}};
        static MemoryRecallRequest Request(List<CompressedMemoryBlock> blocks)=>new(){Generation=1,GameDay=10,Blocks=blocks,FinalCount=2,CandidateLimit=3,PreprocessMode=1,Scene="square",RecallQuery="query",FormatFailureDetail=(reason,reply)=>reason+reply};
        static void Main()
        {
            var blocks=new List<CompressedMemoryBlock>{Block("old",1),Block("new",3),Block("middle",2)};
            var request=Request(blocks); var engine=OnnxEmbeddingEngine.Instance;
            Check(MemoryRecallContextOwner.TryBuildMemoryRecallCandidates(blocks.Take(2).ToList(),"query","",null,3,out var small,out _,1,request)&&small.Select(x=>x.Block.Id).SequenceEqual(new[]{"old","new"})&&engine.Calls==0,"below cap date order no embedding");
            Check(MemoryRecallContextOwner.TryBuildMemoryRecallCandidates(blocks,"query","",null,3,out var equal,out _,1,request)&&equal.Select(x=>x.DisplayId).SequenceEqual(new[]{1,2,3})&&engine.Calls==0,"equal cap IDs no embedding");
            Check(MemoryRecallContextOwner.TryBuildMemoryRecallCandidates(blocks,"query","",null,2,out var over,out _,1,request)&&over.Select(x=>x.Block.Id).SequenceEqual(new[]{"middle","new"})&&engine.Calls==4,"over cap embedding tie newest selection chronological render");
            var missing=new List<CompressedMemoryBlock>{Block("missing",1,""),Block("good",2),Block("latest",3)};
            Check(MemoryRecallContextOwner.TryBuildMemoryRecallCandidates(missing,"query","",null,1,out var nonempty,out _,1,request)&&nonempty[0].Block.Id=="latest","missing rich title skipped not summary embedding fallback");
            engine.IsAvailable=false;
            Check(!MemoryRecallContextOwner.TryBuildMemoryRecallCandidates(blocks,"query","",null,1,out _,out _,1,request)&&request.BlockingMessage.Contains("不会静默"),"unavailable embedding blocks above cap");
            engine.IsAvailable=true;engine.Compute=text=>text=="query"?new float[]{1,0}:null;
            Check(!MemoryRecallContextOwner.TryBuildMemoryRecallCandidates(blocks,"query","",null,1,out _,out _,1,request)&&request.BlockingMessage.Contains("修复"),"all title embedding failures block");
            engine.Compute=text=>new float[]{1,0}; request.BlockingTitle=null;
            var candidates=blocks.Select((b,i)=>new MemoryRecallCandidate{Block=b,DisplayId=i+1,Score=i}).ToList();
            AIConfigHandler.Response="{\"rule_codes\":[],\"memory_ids\":[999,\"#2\",2]}";
            Check(MemoryRecallContextOwner.TrySelectMemoryIdsWithPreprocess(candidates,2,"query","",out var ids,out _,1,request)&&ids.SequenceEqual(new[]{2,3}),"invalid preselected ID ignored dedup fill score order");
            string context=MemoryRecallContextOwner.BuildCompressedMemoryContextById("ada","query","",request);
            Check(context.Contains("summary new")&&context.Contains("[AFEF NPC行为补充] fact new")&&context.Contains("时")&&context.Contains("天前"),"full context latest summary AFEF age hour preserved");
            Check(context.Contains("summary middle")&&!context.Contains("summary old"),"preprocess limit plus mandatory latest block");
            var queries=HistoryArchiveRecallOwner.BuildHistoryRecallQueryInputs(null,"trade horses; debt payment","duel stake");
            Check(queries.Count>1&&queries.Count<=4&&queries.All(x=>x.Weight==1&&x.Terms.Count>0),"multi intent query both speakers bounded");
            var older=Enumerable.Range(0,300).Select(i=>new HistoryLineEntry{Index=i,Line="trade debt "+i}).ToList();
            var hits=HistoryArchiveRecallOwner.FindRelevantArchiveHits(older,queries,4,out var used,out var mode,out var rerank,out var recall);
            Check(hits.Select(x=>x.Entry.Index).SequenceEqual(new[]{299,297})&&mode=="semantic_multi"&&rerank==4&&recall==10,"archive per-intent rerank four then adjacent suppression yields two");
            Check(hits.Select(x=>x.Entry.Index).Zip(hits.Select(x=>x.Entry.Index).Skip(1),(a,b)=>Math.Abs(a-b)>1).All(x=>x),"archive adjacent suppression");
            SaveRuntimeGuard.Current=2;
            int before=engine.Calls;
            Check(!MemoryRecallContextOwner.TryBuildMemoryRecallCandidates(blocks,"query","",null,1,out _,out var stale,1,request)&&stale=="memory_recall_stale_generation"&&engine.Calls==before,"cancel stale generation before embedding");
            Check(MemoryRecallContextOwner.BuildCompressedMemoryContextById("ada","query","",request)=="","late result generation discarded");
            var seed = new InteractionMemoryRecoverySeed { SubjectId="ada", Components=new[] {
                new InteractionMemoryRecoveryComponentSeed { Part="user",DailyText="question",IsLlmDialogue=true },
                new InteractionMemoryRecoveryComponentSeed { Part="USER",DailyText="duplicate",IsLlmDialogue=true },
                new InteractionMemoryRecoveryComponentSeed { Part="fact",DailyText="delivered",IsAfef=true },
                new InteractionMemoryRecoveryComponentSeed { Part="assistant",DailyText="answer",IsLlmDialogue=true } } };
            var receipts = new HashSet<string>(); int notices=0;
            MemoryAuxiliaryReceiptOutcome Notify(InteractionMemoryRecoverySeed s,string id,string hash,string part)
            { notices++;return receipts.Add(part)?MemoryAuxiliaryReceiptOutcome.Accepted:MemoryAuxiliaryReceiptOutcome.Duplicate; }
            var first = InteractionMemoryAuxiliaryCompletionCoordinator.CompleteInitial(seed,"id","hash",(s,id,h,p)=>true,Notify);
            Check(first.Accepted==2&&notices==2,"auxiliary only distinct published dialogue components not facts");
            var repeated = InteractionMemoryAuxiliaryCompletionCoordinator.CompleteInitial(seed,"id","hash",(s,id,h,p)=>true,Notify);
            Check(repeated.Accepted==0&&repeated.Duplicate==2,"independent Social duplicate receipt not H marker success");
            var missingReceipt = InteractionMemoryAuxiliaryCompletionCoordinator.CompleteInitial(seed,"id","hash",(s,id,h,p)=>false,Notify);
            Check(!missingReceipt.HasAttempt&&notices==4,"unpublished Daily component never notifies Social");
            var unavailable = InteractionMemoryAuxiliaryCompletionCoordinator.CompleteInitial(seed,"id","hash",(s,id,h,p)=>true,(s,id,h,p)=>MemoryAuxiliaryReceiptOutcome.Unavailable);
            Check(unavailable.Accepted==0&&unavailable.Unavailable==2,"unavailable Social is not promoted into success");
            Check(MemoryRecoveryStateOwner.ShouldCompleteInitialInteractionMemoryNotoriety(InteractionMemoryRecoveryBeginStatus.Began,true,"id","id")
                && !MemoryRecoveryStateOwner.ShouldCompleteInitialInteractionMemoryNotoriety(InteractionMemoryRecoveryBeginStatus.ExistingPending,true,"id","id")
                && !MemoryRecoveryStateOwner.ShouldCompleteInitialInteractionMemoryNotoriety(InteractionMemoryRecoveryBeginStatus.DuplicateCompleted,true,"id","id"),"initial guard excludes recovered or completed duplicate notification");
        }
    }
