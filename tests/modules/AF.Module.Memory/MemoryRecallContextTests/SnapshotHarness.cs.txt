    internal static class SnapshotChecks {
        static void Check(bool ok,string name) { if(!ok)throw new Exception("FAIL "+name); Console.WriteLine("PASS "+name); }
        internal static void Run() {
            TWParallel.MainThread=Environment.CurrentManagedThreadId;
            SaveRuntimeGuard.Current=1;
            bool ownerCurrent=true; int rawCaptures=0; bool rawIncludesCurrent=false;
            var hero=new Hero(); var heroes=new Dictionary<int,Hero>{{7,hero}};
            var blocks=new List<CompressedMemoryBlock>();
            var raw=new List<ConversationMessage>{new(){Role="assistant",Content="I remember our promise."}};
            string overview="overview before";
            var ports=new SharedPromptCaptureBannerlordAdapter.HistoryWorkCapturePorts {
                IsCurrentOwner=()=>ownerCurrent,
                LoadBlocks=id=> {Check(TWParallel.IsMainThread(),"blocks captured on main thread");return blocks;},
                LoadDrafts=id=>new(), Overview=id=>overview,
                Uncompressed=(id,name,agent,include)=> {
                    Check(TWParallel.IsMainThread(),"raw projected on same main thread"); rawCaptures++; rawIncludesCurrent=include;
                    return raw.Select(x=>new ConversationMessage{Role=x.Role,Content=x.Content}).ToList();
                },
                BuildContext=(id,name,max,input,secondary,include,snapshot)=>snapshot.Overview+" "+string.Join(" ",snapshot.Blocks.Select(b=>b.Summary))
            };
            // The request captures BEFORE compression commits. Simulate ApplyDaily's publish+draft removal.
            HistoryPromptSnapshot before=null;
            var work=SharedPromptCaptureBannerlordAdapter.CaptureHistoryContextWorkForHero(()=>ports,hero,"hi","",false,1,s=>before=s,7);
            blocks.Add(new(){Id="day-1",GameDayIndex=1,Summary="summary of our promise",AfefLines=new(){"fact"}});
            raw[0].Content="mutated"; raw.Clear(); overview="overview after";
            Task.Run(work).GetAwaiter().GetResult();
            Check(before.UncompressedMessages.Single().Content=="I remember our promise." && !before.Context.Contains("summary of"),"compression after capture retains original even without new summary");
            Check(before.Context.Contains("overview before"),"overview frozen with blocks and raw");
            HistoryPromptSnapshot after=null;
            work=SharedPromptCaptureBannerlordAdapter.CaptureHistoryContextWorkForHero(()=>ports,hero,"hi","",false,1,s=>after=s,7);
            Task.Run(work).GetAwaiter().GetResult();
            Check(after.UncompressedMessages.Count==0 && after.Context.Contains("summary of our promise"),"next request uses committed summary");
            blocks[0].Summary="changed"; blocks[0].AfefLines.Clear();
            Check(after.Blocks[0].Summary=="summary of our promise" && after.Blocks[0].AfefLines.Count==1,"snapshot block and AFEF detached");
            SaveRuntimeGuard.Current=2; Check(work()=="","captured work discarded after save change");
            SaveRuntimeGuard.Current=1; ownerCurrent=false; Check(work()=="","captured work discarded after owner change"); ownerCurrent=true;

            MyBehavior.Ports=ports;
            foreach(bool nonHero in new[]{false,true}) {
                HistoryPromptSnapshot native=null; blocks.Clear();raw.Add(new(){Role="assistant",Content="native original"});
                work=SceneHistoryPromptCaptureAdapter.CaptureNativeConversationPersistedHistoryWork(new(),new(),nonHero?null:hero,null,"hi","previous",false,1,s=>native=s);
                raw.Clear();blocks.Add(new(){Id="native",Summary="new summary"});Task.Run(work).GetAwaiter().GetResult();
                Check(rawIncludesCurrent && native.UncompressedMessages.Single().Content=="native original" && !native.Context.Contains("new summary"),"native hero/nonhero capture retains active raw during compression "+nonHero);
            }
            var scene=new SceneHistoryPromptCaptureAdapter(new(new(),()=>ownerCurrent,()=>1,()=>ports));
            var cache=new Dictionary<int,ShoutBehavior.PrecomputedShoutRagContext>();
            raw.Add(new(){Role="assistant",Content="scene original"}); blocks.Clear();
            var task=scene.StartPrecomputedPersistedHistoryContextTask(7,"hi",heroes,cache,"fixture");
            raw.Clear(); blocks.Add(new(){Id="scene",Summary="scene compressed"});
            var result=scene.AwaitPrecomputedPersistedHistoryContextAsync(7,"hi",heroes,cache,task,"fixture").GetAwaiter().GetResult();
            Check(result.UncompressedMessages.Single().Content=="scene original" && !result.Context.Contains("scene compressed"),"scene task accepts paired original snapshot after compression");
            int count=rawCaptures;
            task=scene.StartPrecomputedPersistedHistoryContextTask(7,"hi",heroes,cache,"fixture");
            result=scene.AwaitPrecomputedPersistedHistoryContextAsync(7,"hi",heroes,cache,task,"fixture").GetAwaiter().GetResult();
            Check(rawCaptures==count && result.UncompressedMessages.Single().Content=="scene original","scene cached text and raw remain paired");
            task=scene.StartPrecomputedPersistedHistoryContextTask(7,"next",heroes,cache,"fixture");
            SaveRuntimeGuard.Current=2;
            Check(scene.AwaitPrecomputedPersistedHistoryContextAsync(7,"next",heroes,cache,task,"fixture").GetAwaiter().GetResult()==null,"scene stale task discards raw and summary together");
            SaveRuntimeGuard.Current=1;
            task=scene.StartPrecomputedPersistedHistoryContextTask(7,"retire",heroes,cache,"fixture"); ownerCurrent=false;
            Check(scene.AwaitPrecomputedPersistedHistoryContextAsync(7,"retire",heroes,cache,task,"fixture").GetAwaiter().GetResult()==null,"scene owner replacement discards snapshot"); ownerCurrent=true;
            blocks.Clear(); raw.Clear(); overview="";
            result=scene.BuildPersistedHeroHistoryContext(7,"empty",heroes);
            Check(string.IsNullOrWhiteSpace(result.Context) && result.UncompressedMessages.Count==0,"scene genuinely empty paired snapshot");
        }
    }
