        {

        private static MyBehavior NewOwner()

        private static void Worker(Action action) => Task.Run(action).GetAwaiter().GetResult();
        private static void Drain(MyBehavior owner)
        {
            for (int i = 0; i < 40 && owner.Pending > 0; i++) owner.Tick();
            Check(owner.Pending == 0, "queued writes terminate within fixture bound");
        }

        private static void EqualCall(WriteCall actual, string kind, object[] args, string label)
        {
            Check(actual != null && actual.Kind == kind, label + " terminal owner");
            if (actual == null) return;
            Check(actual.Arguments.SequenceEqual(args), label + " exact argument forwarding");
        }
        private static IEnumerable<MemberInfo> Members(Type type) => type
            .GetMembers(BindingFlags.Instance | BindingFlags.Public)
            .Where(m => m is FieldInfo || m is PropertyInfo p && p.CanRead && p.CanWrite)
            .OrderBy(m => m.Name, StringComparer.Ordinal);
        private static Type MemberType(MemberInfo m) => m is FieldInfo f ? f.FieldType : ((PropertyInfo)m).PropertyType;
        private static object Get(MemberInfo m, object value) => m is FieldInfo f ? f.GetValue(value) : ((PropertyInfo)m).GetValue(value);
        private static void Set(MemberInfo m, object target, object value)
        {
            if (m is FieldInfo f) f.SetValue(target, value); else ((PropertyInfo)m).SetValue(target, value);
        }
        private static object ValueFor(Type type, int seed)
        {
            if (type == typeof(string)) return "field-" + seed;
            if (type == typeof(int)) return seed;
            if (type == typeof(bool)) return seed < 1000;
            if (type.IsEnum) return Enum.GetValues(type).GetValue(Enum.GetValues(type).Length - 1);
            if (type == typeof(List<int>)) return new List<int> { seed, seed + 1 };
            if (type == typeof(EquipmentElement)) return new EquipmentElement { Token = seed };
            if (type.Namespace != null && type.Namespace.StartsWith("TaleWorlds.", StringComparison.Ordinal))
                return Activator.CreateInstance(type);
            throw new InvalidOperationException("New DTO member requires explicit fixture support: " + type);
        }
        private static T Populate<T>(int seed) where T : new()
        {
            var value = new T();
            foreach (var member in Members(typeof(T))) Set(member, value, ValueFor(MemberType(member), ++seed));
            return value;
        }
        private static Dictionary<string, object> Snapshot(object value) => Members(value.GetType())
            .ToDictionary(m => m.Name, m => Get(m, value) is List<int> ints ? (object)ints.ToArray() : Get(m, value));
        private static void Mutate(object value)
        {
            int seed = 2000;
            foreach (var m in Members(value.GetType()))
            {
                if (Get(m, value) is List<int> ints) { ints.Clear(); ints.Add(-100); }
                Set(m, value, ValueFor(MemberType(m), ++seed));
            }
        }
        private static void CompareDto(object actual, Dictionary<string, object> expected, string label)
        {
            Check(actual != null, label + " copied DTO exists");
            if (actual == null) return;
            Check(Members(actual.GetType()).Count() == expected.Count, label + " field inventory unchanged");
            foreach (var member in Members(actual.GetType()))
            {
                object value = Get(member, actual), wanted = expected[member.Name];
                bool equal = wanted is int[] indices
                    ? value is List<int> actualIndices && indices.SequenceEqual(actualIndices)
                    : Equals(value, wanted);
                Check(equal, label + "." + member.Name + " preserved");
            }
        }
        private static void WeeklyCopies(bool all)
        {
            var owner = NewOwner();
            var reward = Populate<RewardSystemBehavior.RewardItemInfo>(10);
            var troop = Populate<MyBehavior.PartyTransferPromptEntry>(30);
            var prisoner = Populate<MyBehavior.PartyTransferPromptEntry>(60);
            var settlement = Populate<MyBehavior.SettlementTransferPromptEntry>(90);
            var allTroop = Populate<MyBehavior.PartyTransferPromptEntry>(130);
            var allPrisoner = Populate<MyBehavior.PartyTransferPromptEntry>(160);
            var rewards = new List<RewardSystemBehavior.RewardItemInfo> { reward, null };
            var troops = new List<MyBehavior.PartyTransferPromptEntry> { troop, null };
            var prisoners = new List<MyBehavior.PartyTransferPromptEntry> { prisoner, null };
            var settlements = new List<MyBehavior.SettlementTransferPromptEntry> { settlement, null };
            var allTroops = new List<MyBehavior.PartyTransferPromptEntry> { allTroop, null };
            var allPrisoners = new List<MyBehavior.PartyTransferPromptEntry> { allPrisoner, null };
            object[] dtos = all ? new object[] {reward, troop, prisoner, settlement, allTroop, allPrisoner}
                : new object[] {reward, troop, prisoner, settlement};
            var expected = dtos.Select(Snapshot).ToArray();
            var playerSettlements = new List<MyBehavior.SettlementTransferPromptEntry> { Populate<MyBehavior.SettlementTransferPromptEntry>(300) };
            Worker(() =>
            {
                if (all) MyBehavior.MarkWeeklyMemoryMaterialTriggerWithAllSnapshotsForExternal(null, "id", "npc", "tag",
                    rewardOptions: rewards, partyTransferTroopOptions: troops, partyTransferPrisonerOptions: prisoners,
                    settlementTransferNpcOptions: settlements, partyTransferAllTroopOptions: allTroops,
                    partyTransferAllPrisonerOptions: allPrisoners);
                else MyBehavior.MarkWeeklyMemoryMaterialTriggerForExternal(null, "id", "npc", "tag",
                    rewardOptions: rewards, partyTransferTroopOptions: troops, partyTransferPrisonerOptions: prisoners,
                    settlementTransferNpcOptions: settlements, settlementTransferPlayerOptions: playerSettlements);
            });
            Check(Campaign.OffThreadReads == 0 && owner.OffThreadWrites == 0 && owner.Calls.Count == 0,
                "weekly captures without Campaign or terminal access");
            foreach (object dto in dtos) Mutate(dto);
            rewards.Clear(); troops.Clear(); prisoners.Clear(); settlements.Clear(); allTroops.Clear(); allPrisoners.Clear(); playerSettlements.Clear();
            Drain(owner);
            Check(owner.Calls.Count == 1, "weekly publishes once");
            if (owner.Calls.Count == 0) return;
            int[] positions = all ? new[] {7, 8, 9, 10, 12, 13} : new[] {7, 8, 9, 10};
            for (int i = 0; i < positions.Length; i++)
            {
                var list = owner.Calls[0].Arguments[positions[i]] as IList;
                Check(list != null && list.Count == 2 && list[1] == null, "weekly outer list/null slot snapshot " + i);
                if (list == null || list.Count == 0) continue;
                Check(!ReferenceEquals(list[0], dtos[i]), "weekly DTO independently owned " + i);
                CompareDto(list[0], expected[i], (all ? "all." : "plain.") + dtos[i].GetType().Name + "[" + i + "]");
            }
            Check(owner.OffThreadWrites == 0, "weekly terminal stayed main");
            owner.Tick(); Check(owner.Calls.Count == 1, "weekly cannot replay");
        }
        public static int Main()
        {
            try
            {
                var cases = Calls();
                var owner = NewOwner();
                foreach (var item in cases)
                {
                    int before = owner.Calls.Count;
                    item.Call();
                    Check(owner.Calls.Count == before + 1 && owner.Pending == 0, item.Name + " main remains synchronous");
                    EqualCall(owner.Calls.LastOrDefault(), item.Kind, item.Args, item.Name + " main");
                }
                owner = NewOwner();
                foreach (var item in cases) Worker(item.Call);
                Check(owner.Calls.Count == 0 && owner.Pending == 8, "all 8 worker facades only publish");
                Check(Campaign.OffThreadReads == 0 && owner.OffThreadWrites == 0, "no worker Campaign access or terminal write");
                Drain(owner);
                Check(owner.Calls.Count == 8, "all 8 facades reach terminal once");
                for (int i = 0; i < Math.Min(owner.Calls.Count, cases.Count); i++)
                    EqualCall(owner.Calls[i], cases[i].Kind, cases[i].Args, cases[i].Name + " queued FIFO");
                owner.Tick(); Check(owner.Calls.Count == 8, "extra tick cannot duplicate writers");
                foreach (string retirement in new[] {"generation", "owner", "campaign-owner", "reset", "campaign-missing"})
                {
                    owner = NewOwner();
                    foreach (var item in cases) Worker(item.Call);
                    var replacement = new MyBehavior();
                    if (retirement == "generation") SaveRuntimeGuard.AdvanceGeneration("load");
                    if (retirement == "owner") { MyBehavior.Instance = replacement; Campaign.Owner = replacement; }
                    if (retirement == "campaign-owner") Campaign.Owner = replacement;
                    if (retirement == "reset") owner.Reset();
                    if (retirement == "campaign-missing") Campaign.Available = false;
                    Drain(owner);
                    Check(owner.Calls.Count == 0 && replacement.Calls.Count == 0, retirement + " retires all queued writers");
                }
                owner = NewOwner(); MyBehavior.Instance = null;
                foreach (var item in cases) Worker(item.Call);
                Check(owner.Pending == 0 && owner.Calls.Count == 0 && Campaign.OffThreadReads == 0, "missing Instance does not inspect Campaign on worker");
                WeeklyCopies(false); WeeklyCopies(true);
                owner = NewOwner();
                Worker(() => MyBehavior.AppendExternalDialogueHistory(null, null, null, null));
                Worker(() => MyBehavior.AppendExternalNonHeroDialogueHistory(null, null, null, null, null));
                Worker(() => MyBehavior.MigrateNonHeroPartyScopedMemoryForExternal(null, null));
                Drain(owner);
                Check(owner.Calls.Count == 3 && owner.Calls.All(x => x.Arguments.Take(x.Kind == "hero" ? 4 : x.Kind == "nonhero" ? 5 : 2).All(a => a == null)), "null arguments preserved without facade invention");
                foreach (Type type in new[] {typeof(RewardSystemBehavior.RewardItemInfo), typeof(MyBehavior.PartyTransferPromptEntry), typeof(MyBehavior.SettlementTransferPromptEntry)})
                    Console.WriteLine("DTO_INVENTORY " + type.Name + "=" + string.Join(",", Members(type).Select(x => x.Name)));
            }
            catch (Exception ex) { failures++; Console.WriteLine("FAIL fixture exception " + ex); }
            Console.WriteLine("WRITER_RESULT checks=" + checks + " failures=" + failures);
            return failures == 0 ? 0 : 1;
        }
