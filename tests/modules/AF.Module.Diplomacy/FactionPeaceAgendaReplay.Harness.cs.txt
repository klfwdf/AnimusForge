        public int AddCalls;
        public void AddDecision(KingdomDecision decision, bool ignoreInfluenceCost)
        {
            AddCalls++;
            // Invoke the actual extracted production takeover prefix.
            if (!AnimusForge.WorldDiplomacyBehavior.Admit(this, decision, ignoreInfluenceCost)) return;
            if (!ignoreInfluenceCost) decision.ProposerClan.Influence -= decision.GetInfluenceCost(decision.ProposerClan);
            CampaignEventDispatcher.Instance.OnKingdomDecisionAdded(decision, false);
            if (!Campaign.DropFirst) UnresolvedDecisions.Add(decision);
        }
        public void RemoveDecision(KingdomDecision decision) => UnresolvedDecisions.Remove(decision);
    }

// A fake SaveSystem remaps each native decision to a newly materialized instance, then
// production SyncData restores those exact references. No original object remains in queues.
internal sealed class CampaignSave : IDataStore
{
    internal static bool DropReview;
    internal Dictionary<string,object> Values = new();
    internal bool Saving = true;
    public bool IsSaving => Saving;
    public bool IsLoading => !Saving;
    public void SyncData<T>(string key, ref T value)
    {
        if (Saving)
        {
            if (value is Dictionary<string,string> map) Values[key] = new Dictionary<string,string>(map);
            else if (value is List<KingdomDecision> refs) Values[key] = new List<KingdomDecision>(refs);
            else Values[key] = value;
        }
        else value = Values.TryGetValue(key,out var stored) ? (T)stored : default;
    }
    internal Dictionary<KingdomDecision,KingdomDecision> Rematerialize()
    {
        var map = new Dictionary<KingdomDecision,KingdomDecision>();
        var refs = (List<KingdomDecision>)Values["_vdFactionPeaceDecisionRefs"];
        foreach (KingdomDecision old in refs)
        {
            if (old == null || map.ContainsKey(old)) continue;
            var p = (MakePeaceKingdomDecision)old;
            map.Add(old,new MakePeaceKingdomDecision(p.ProposerClan,p.FactionToMakePeaceWith,p.DailyTributeToBePaid,p.DailyTributeDurationInDays,true,p.Review));
        }
        Values["_vdFactionPeaceDecisionRefs"] = refs.Select(d => d == null ? null : map[d]).ToList();
        foreach (Kingdom k in Kingdom.All)
            for (int i=0;i<k.UnresolvedDecisions.Count;i++)
                if (map.TryGetValue(k.UnresolvedDecisions[i],out var replacement)) k.UnresolvedDecisions[i]=replacement;
        Saving=false;
        return map;
    }
}

internal static class Replay
{
    private static int assertions;
    private static Kingdom home, target;
    private static Clan proposer;
    private static void Check(bool ok,string message) { assertions++; if (!ok) throw new Exception("FAIL " + message); }
    private static void Setup()
    {
        Kingdom.All.Clear(); FactionManager.War=true; MakePeaceAction.Calls=0; MakePeaceAction.AuthorizedCalls=0;
        MakePeaceAction.Block=false; Campaign.DropFirst=false; CampaignSave.DropReview=false; DecisionSettings.Disallowed=false;
        AnimusForge.Refactor.Modules.TeamModuleServices.CivilWar = new();
        home=new Kingdom{StringId="home"}; target=new Kingdom{StringId="target"};
        proposer=new Clan{StringId="proposer",Kingdom=home,Leader=new Hero()}; home.RulingClan=proposer;
        target.RulingClan=new Clan{StringId="target_ruler",Kingdom=target,Leader=new Hero()};
        Kingdom.All.Add(home); Kingdom.All.Add(target); Clan.PlayerClan=proposer;
        AnimusForge.VoteDealBehavior.Instance=new();
    }
    private static string Query(string id="op") => AnimusForge.VoteDealBehavior.GetOrSubmitFactionPeaceAgenda(id,"antiwar",home,target,proposer,out _);
    private static KingdomDecision First => home.UnresolvedDecisions.Single();
    private static KingdomDecision Review => target.UnresolvedDecisions.Single();
    private static void Reload()
    {
        var save=new CampaignSave(); AnimusForge.VoteDealBehavior.Instance.SyncData(save);
        var old = new List<KingdomDecision>(home.UnresolvedDecisions.Concat(target.UnresolvedDecisions));
        save.Rematerialize(); AnimusForge.VoteDealBehavior.Instance=new(); AnimusForge.VoteDealBehavior.Instance.SyncData(save);
        foreach (var d in old) Check(!AnimusForge.VoteDealBehavior.IsFactionPeaceDecision(d),"old object does not acquire restored admission");
    }
    public static void Main()
    {
        Setup(); Check(Query()=="pending","submit first"); var first=First;
        Check(proposer.Influence==90 && home.AddCalls==1,"first costs exactly once");
        Check(Query()=="pending" && home.AddCalls==1 && proposer.Influence==90,"duplicate operation submits/charges once");
        Check(!AnimusForge.VoteDealBehavior.RoyalSupportsPeace(First,proposer),"player ruler retains explicit voting control");
        Clan.PlayerClan=null;
        Check(AnimusForge.VoteDealBehavior.RoyalSupportsPeace(First,proposer),"AI ruler promise is bound to the exact tracked first peace decision");
        Check(!AnimusForge.VoteDealBehavior.RoyalSupportsPeace(new MakePeaceKingdomDecision(proposer,target),proposer),"promise cannot force an unrelated peace agenda");
        Clan.PlayerClan=proposer;
        var unrelated=new MakePeaceKingdomDecision(proposer,target);
        Check(!AnimusForge.WorldDiplomacyBehavior.Admit(home,unrelated,false),"ordinary native add blocked");
        Check(AnimusForge.VoteDealBehavior.HiddenByAgendaUi(unrelated),"ordinary native agenda hidden");
        Check(!AnimusForge.VoteDealBehavior.HiddenByAgendaUi(first),"bound first agenda visible");
        home.UnresolvedDecisions.Add(unrelated); AnimusForge.WorldDiplomacyBehavior.Sanitize();
        Check(home.UnresolvedDecisions.Count==1 && ReferenceEquals(First,first),"sanitizer removes ordinary native but retains bound first");
        MakePeaceAction.ApplyByKingdomDecision(home,target,0,0);
        Check(FactionManager.War && MakePeaceAction.AuthorizedCalls==0,"ordinary native final action blocked");
        MakePeaceAction.Calls=0;
        Reload(); Check(Query()=="pending" && AnimusForge.VoteDealBehavior.IsFactionPeaceDecision(First),"first refs restored by production SyncData");
        first=First; AnimusForge.VoteDealBehavior.Vote(first,true); var review=Review;
        AnimusForge.VoteDealBehavior.Vote(first,true);
        Check(target.UnresolvedDecisions.Count==1 && proposer.Influence==90,"replayed first approval cannot duplicate review or charge");
        Check(Query()=="pending" && FactionManager.War && MakePeaceAction.Calls==0,"first approval only queues review");
        Check(target.RulingClan.Influence==100 && proposer.Influence==90,"review has no proposal charge");
        Check(AnimusForge.WorldDiplomacyBehavior.Admit(target,review,true) && !AnimusForge.VoteDealBehavior.HiddenByAgendaUi(review),"review add gate and UI permitted");
        AnimusForge.WorldDiplomacyBehavior.Sanitize(); Check(target.UnresolvedDecisions.Contains(review),"review survives native sanitization");
        Reload(); Check(Query()=="pending" && AnimusForge.VoteDealBehavior.IsFactionPeaceDecision(Review),"review exact refs restored");
        review=Review; AnimusForge.VoteDealBehavior.Vote(review,true);
        Check(Query()=="completed" && !FactionManager.War && MakePeaceAction.Calls==1 && MakePeaceAction.AuthorizedCalls==1,"final bilateral approval uses authorized actual peace");
        Check(AnimusForge.Refactor.Modules.TeamModuleServices.CivilWar.Notifications==1
            && AnimusForge.Refactor.Modules.TeamModuleServices.CivilWar.StatesAtNotice.Single()=="completed","terminal receipt precedes political notification");
        AnimusForge.VoteDealBehavior.Vote(review,true); Check(Query()=="completed" && MakePeaceAction.Calls==1,"duplicate final callback cannot reapply");
        Reload(); Check(Query()=="completed" && home.AddCalls==1,"terminal persisted, never resubmitted");
        foreach(bool second in new[]{false,true})
        {
            Setup(); Query(); if(second) AnimusForge.VoteDealBehavior.Vote(First,true);
            AnimusForge.VoteDealBehavior.Vote(second?Review:First,false);
            Check(Query()=="rejected" && FactionManager.War && MakePeaceAction.Calls==0,"first/review rejection never makes peace");
            Check(AnimusForge.Refactor.Modules.TeamModuleServices.CivilWar.Notifications==1,"rejection notifies owner");
            Reload(); Check(Query()=="rejected" && home.AddCalls==1,"rejected tombstone survives reload");
        }
        foreach(bool second in new[]{false,true})
        {
            Setup(); Query(); if(second) AnimusForge.VoteDealBehavior.Vote(First,true);
            AnimusForge.VoteDealBehavior.Cancel(second?Review:First);
            Check(Query()=="failed" && FactionManager.War && MakePeaceAction.Calls==0,"first/review cancellation fails without peace");
            Reload(); Check(Query()=="failed" && home.AddCalls==1,"cancelled operation never auto-resubmits");
        }
        Setup(); Query(); AnimusForge.VoteDealBehavior.Vote(First,true); MakePeaceAction.Block=true;
        AnimusForge.VoteDealBehavior.Vote(Review,true); Check(Query()=="failed" && FactionManager.War,"blocked actual effect cannot report completed");
        Setup(); Query(); FactionManager.War=false; Check(Query()=="failed","external peace cannot complete operation");
        Setup(); Query();
        var damaged=new CampaignSave(); AnimusForge.VoteDealBehavior.Instance.SyncData(damaged);
        damaged.Values["_vdFactionPeaceDecisionRefs"]=new List<KingdomDecision>(); damaged.Saving=false;
        AnimusForge.VoteDealBehavior.Instance=new(); AnimusForge.VoteDealBehavior.Instance.SyncData(damaged);
        Check(Query()=="failed" && !AnimusForge.VoteDealBehavior.IsFactionPeaceDecision(First),"missing saved refs fail closed without pair-based adoption");
        Setup(); proposer.Influence=0; Check(Query()=="failed" && home.AddCalls==0,"insufficient influence fails before submit");
        proposer.Influence=100; Check(Query()=="failed" && home.AddCalls==0,"failed operation cannot be reused");
        Setup(); DecisionSettings.Disallowed=true; Check(Query()=="failed" && home.AddCalls==0,"IsAllowed enforced");
        Setup(); Campaign.DropFirst=true; Check(Query()=="failed" && home.AddCalls==1,"not retained is explicit failure");
        Setup(); Query(); Check(AnimusForge.VoteDealBehavior.GetOrSubmitFactionPeaceAgenda("op","other",home,target,proposer,out _)=="failed","operation identity cannot be repurposed");
        Check(Query()=="pending" && home.AddCalls==1,"identity mismatch does not mutate original operation");
        Console.WriteLine("PASS faction peace production control-flow replay: " + assertions + " assertions; Harmony and TaleWorlds SaveSystem NOT-RUN");
    }
}
