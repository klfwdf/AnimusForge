    internal static void Require(bool ok, string message)

    {
        string[] actual = owner.Mutations.Select(m => m.Kind+":"+m.Key+":"+m.Quantity+":"+m.Forced).ToArray();
        Require(actual.SequenceEqual(expected), "expected="+JsonSerializer.Serialize(expected)+" actual="+JsonSerializer.Serialize(actual));
    }

    internal LegacyEconomyRewardDebtMainThreadPort Port(bool current=true)
    {
        int mainThread=Environment.CurrentManagedThreadId;

            _=>current,ReplayEconomyRewardDebtPlanOnMainThread);
    }
    internal List<Mutation> Mutations=new();
    internal bool ReturnZero;
    internal string UnknownKey,ThrowKey;
    internal (bool Applied,string Fact) Apply(string channel,EconomyRewardDebtAction action)
    {
        var observation=new EconomyMutationObservation(); string fact; bool ok;
        switch(channel)
        {
            case "hero": ok=TryReplayGiveAsset(action,Hero.Npc,Hero.MainHero,observation,out fact); break;
            case "party": ok=TryReplayPartyAsset(action,new PartyBase(),new BasicCharacterObject(),Hero.MainHero,"NPC",observation,out fact); break;
            default: ok=TryReplayMerchantAsset(action,new CharacterObject(),new Settlement(),Hero.MainHero,"NPC",observation,out fact); break;
        }
        return (ok,fact);
    }
    internal EconomyRewardDebtReplayResult Replay(params EconomyRewardDebtAction[] actions) =>
        ReplayEconomyRewardDebtPlanOnMainThread(new EconomyRewardDebtReplayPlan(actions,null),new GameInteractionSnapshot());
    private List<RewardItemInfo> GetHeroInventoryItems(Hero hero) => Items;
    private List<RewardItemInfo> BuildHeroRewardPostprocessItems(Hero hero) => Items.Concat(MarketItems.Select(

    // Exact fixture lookup does not reproduce the global fuzzy-name algorithm.
    private bool TryResolveRewardItemByNameOrId(string token,IEnumerable<RewardItemInfo> items,out RewardItemResolution resolution,string source)
    {
        bool found=TryResolveExactAuthorizedRewardItem(items,token,out var info,out var key);
        resolution=found ? new RewardItemResolution{Info=info,ActionKey=key}:null; return found;
    }
    private static string BuildRewardItemTransferLookup(RewardItemResolution r) => r?.ActionKey;
    private static bool TryResolveKnownItemAssetTokenForExternal(string token,out string id)
    { id=token; return new[]{"grain","iron","sword"}.Contains(token); }
    private int Mutate(string kind,string key,int amount,bool forced,EconomyMutationObservation observation)
    {
        Mutations.Add(new Mutation(kind,key,amount,forced));
        if(key==UnknownKey) observation?.MarkUnknown("fixture.roster_exception");
        if(key==ThrowKey) throw new InvalidOperationException("fixture mutation failed after start");
        return ReturnZero ? 0 : amount;
    }
    private int TransferItemByIdForEconomyReplay(Hero giver,Hero receiver,string key,int amount,out string itemName,bool forceComplete,EconomyMutationObservation mutationObservation)
