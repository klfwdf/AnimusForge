
var carried = new PartyBase(); var allied = new PartyBase(); var recipient = new PartyBase();
Hero.MainHero = new Hero { PartyBelongedTo = new MobileParty { Party=carried } };
MobileParty.MainParty = new MobileParty { Party=carried };
Clan.PlayerClan = new Clan();
var carriedTroop = new CharacterObject(); var alliedTroop = new CharacterObject();
carried.PrisonRoster.Values[carriedTroop]=new(){Number=3};
allied.PrisonRoster.Values[alliedTroop]=new(){Number=4};
ActualAdapterFixture.Counterparty=allied;
var candidates=ActualAdapterFixture.BuildPartyTransferPromptEntriesInternal(new Hero{Clan=new Clan()}, null, -1, out _);
Check(candidates.Count(e=>e.Section==Section.PlayerPrisoners)==1 && candidates.Single(e=>e.Section==Section.PlayerPrisoners).OwnerParty==carried, "player candidates contain carried prisoners only");
Check(ActualAdapterFixture.PlayerDungeonReads==0 && ActualAdapterFixture.NpcDungeonReads==1, "player dungeon is not scanned; NPC dungeon behavior retained");
var outgoing=new Entry{Section=Section.PlayerPrisoners,OwnerParty=carried,Character=carriedTroop,Count=3};
Check(ActualAdapterFixture.TransferPartyPrisonerEntry(outgoing,2,recipient)==2 && carried.PrisonRoster.Values[carriedTroop].Number==1,"carried regular prisoner transfer succeeds");
var foreign=new Entry{Section=Section.PlayerPrisoners,OwnerParty=allied,Character=alliedTroop,Count=4};
Check(ActualAdapterFixture.TransferPartyPrisonerEntry(foreign,2,recipient)==0 && allied.PrisonRoster.Values[alliedTroop].Number==4,"allied party cannot be relabeled as player prisoners");
outgoing.SourceSettlement=new();
Check(ActualAdapterFixture.TransferPartyPrisonerEntry(outgoing,1,recipient)==0 && carried.PrisonRoster.Values[carriedTroop].Number==1,"stale dungeon selection denied even with current player party reference");
outgoing.SourceSettlement=null;
Hero.MainHero.PartyBelongedTo=new MobileParty{Party=new()};
Check(ActualAdapterFixture.TransferPartyPrisonerEntry(outgoing,1,recipient)==0,"old party selection denied after player changes party");
Hero.MainHero.PartyBelongedTo=null;
candidates=ActualAdapterFixture.BuildPartyTransferPromptEntriesInternal(new Hero{Clan=new Clan()},null,-1,out _);
Check(!candidates.Any(e=>e.Section==Section.PlayerPrisoners || e.Section==Section.PlayerTroops),"no owned party cannot fall back to global MainParty");
Check(ActualAdapterFixture.MoveCharacterFromPlayerPrisonerSourceToParty("c",1,new MobileParty{Party=recipient},false,"")==0,"courier has no main-party fallback");
Hero.MainHero.PartyBelongedTo=new MobileParty{Party=carried};
var lord=new CharacterObject{IsHero=true,HeroObject=new Hero{IsPrisoner=true,PartyBelongedToAsPrisoner=carried}};
carried.PrisonRoster.Values.Clear();carried.PrisonRoster.Values[lord]=new(){Number=1};
var heroEntry=new Entry{Section=Section.PlayerPrisoners,OwnerParty=carried,Character=lord};
var heroTarget=new PartyBase();
Check(ActualAdapterFixture.TransferPartyPrisonerEntry(heroEntry,1,heroTarget)==1 && lord.HeroObject.PartyBelongedToAsPrisoner==heroTarget,"carried hero transferred using native action boundary");
heroEntry.OwnerParty=heroTarget;heroEntry.SourceSettlement=new(){OwnerClan=Clan.PlayerClan,IsFortification=true,Party=heroTarget};
Check(ActualAdapterFixture.TransferPartyPrisonerEntry(heroEntry,1,new PartyBase(),Clan.PlayerClan)==0 && lord.HeroObject.PartyBelongedToAsPrisoner==heroTarget,"owned dungeon hero cannot be given using old entry");
ActualAdapterFixture.CourierSource=null;
Check(ActualAdapterFixture.MoveCharacterFromPlayerPrisonerSourceToParty("lord",1,new MobileParty{Party=recipient},true,"old_castle")==0 && ActualAdapterFixture.CourierSource==null,"old courier dungeon draft does not reach transfer");
Check(ActualAdapterFixture.MoveCharacterFromPlayerPrisonerSourceToParty("c",1,new MobileParty{Party=recipient},false,"")==1 && ActualAdapterFixture.CourierSource==carried,"courier carried source routed correctly");
Console.WriteLine("PASS carried prisoner scope: candidates, no dungeon scan, allied/stale/dungeon rejection, regular/hero transfers, courier admission");
