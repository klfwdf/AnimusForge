static class ConsumerTests {
 static TtsEngine.PlaybackRequest Request(TtsEngine e,int agent=-1){TtsEngine.PlaybackRequest r=null;e.SpeakAsync("x",-1,1,agent,null,x=>r=x);return r;}
 static TtsEngine Engine(){var e=new TtsEngine();typeof(TtsEngine).GetField("_initialized",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(e,true);return e;}
 static void Assert(bool b){if(!b)throw new Exception("consumer assertion");}
 public static void Run(Action<string,Action> check){
  check("consumer: old negative-agent failure cannot consume replacement wait",()=>{TtsConsumerSubject.Reset();using var e=Engine();var a=Request(e);var b=Request(e);TtsConsumerSubject.Register(a);TtsConsumerSubject.Register(b);var wait=TtsConsumerSubject.Wait;Assert(!TtsConsumerSubject.Complete(a)&&!wait.IsCompleted);Assert(TtsConsumerSubject.Complete(b)&&wait.IsCompleted);});
  check("consumer: same Scene agent late finish does not consume new wait",()=>{TtsConsumerSubject.Reset();using var e=Engine();var a=Request(e,7);var b=Request(e,7);TtsConsumerSubject.Register(a);TtsConsumerSubject.Register(b);Assert(!TtsConsumerSubject.Complete(a)&&TtsConsumerSubject.Matches(b));});
  check("consumer: cancelled request cannot register new wait",()=>{TtsConsumerSubject.Reset();using var e=Engine();var a=Request(e);e.StopPlayback();Assert(TtsConsumerSubject.Register(a)==0&&TtsConsumerSubject.Wait==null);});
  check("consumer: cancellation closes only its already-registered wait",()=>{TtsConsumerSubject.Reset();using var e=Engine();var a=Request(e);TtsConsumerSubject.Register(a);var wait=TtsConsumerSubject.Wait;e.OnRequestPlaybackCancelled+=r=>TtsConsumerSubject.Complete(r);e.StopPlayback();Assert(wait.IsCompleted);});
  check("consumer: old cancellation does not close newer wait",()=>{TtsConsumerSubject.Reset();using var e=Engine();var a=Request(e);e.StopPlayback();var b=Request(e);TtsConsumerSubject.Register(b);Assert(!TtsConsumerSubject.Complete(a)&&TtsConsumerSubject.Matches(b));});
  check("consumer: timeout token cannot close newer wait",()=>{TtsConsumerSubject.Reset();using var e=Engine();var a=Request(e);var b=Request(e);long token=TtsConsumerSubject.Register(a);TtsConsumerSubject.Register(b);TtsConsumerSubject.CompleteToken(token);Assert(TtsConsumerSubject.Matches(b));});
  check("consumer: cancelled generation invalidates typewriter fallback token",()=>{TtsConsumerSubject.Reset();using var e=Engine();var a=Request(e);long t=TtsConsumerSubject.Register(a);Assert(TtsConsumerSubject.Token(t,-1));e.StopPlayback();Assert(!TtsConsumerSubject.Token(t,-1));});
  check("consumer: late retired event cannot mutate queued same-agent Scene owner",()=>{using var e=Engine();var s=new TtsConsumerSubject();var a=Request(e,7);s.Track(a);var b=Request(e,7);s.Track(b);Assert(s.IsCurrent(a)&&s.IsCurrent(b));s.Retire(a);Assert(!s.IsCurrent(a)&&s.IsCurrent(b)&&s.Owners==1);});
  check("consumer: session change invalidates queued UI event",()=>{using var e=Engine();var s=new TtsConsumerSubject();var a=Request(e,7);s.Track(a);s.NextSession();Assert(!s.IsCurrent(a));});
  check("consumer: epoch change invalidates queued UI event",()=>{using var e=Engine();var s=new TtsConsumerSubject();var a=Request(e,7);s.Track(a);s.NextEpoch();Assert(!s.IsCurrent(a));});
  check("consumer: mission replacement invalidates queued UI event",()=>{using var e=Engine();var s=new TtsConsumerSubject();var a=Request(e,7);s.Track(a);Mission.Current=new();Assert(!s.IsCurrent(a));});
  check("consumer: save generation invalidates queued UI event",()=>{using var e=Engine();var s=new TtsConsumerSubject();var a=Request(e,7);s.Track(a);SaveRuntimeGuard.Generation++;Assert(!s.IsCurrent(a));});
  check("consumer: cancelled cleanup requires exact owner and lifetime",()=>{using var e=Engine();var s=new TtsConsumerSubject();var a=Request(e,7);s.Track(a);e.StopPlayback();Assert(!s.IsCurrent(a)&&s.IsCurrent(a,true));var b=Request(e,7);s.Track(b);s.Prepare(b);Assert(!s.Active(a,true)&&s.Active(b));s.Retire(a);Assert(s.IsCurrent(b)&&s.Owners==1);});
 }
