static class Check { internal static int Count; internal static void That(bool ok,string name){Count++;if(!ok)throw new Exception("ASSERT "+name);} }
sealed class TrackedContent : StringContent
{
    internal bool Disposed; internal TrackedContent(string text,string media="text/event-stream"):base(text,Encoding.UTF8,media){}
    protected override void Dispose(bool disposing){Disposed=true;base.Dispose(disposing);}
}
sealed class ThrowOnReadContent : HttpContent
{
    internal bool Disposed; protected override bool TryComputeLength(out long length){length=-1;return false;}
    protected override Task SerializeToStreamAsync(Stream stream,TransportContext context)=>throw new Exception("body must not be read");
    protected override Task<Stream> CreateContentReadStreamAsync()=>throw new Exception("body must not be read");
    protected override void Dispose(bool disposing){Disposed=true;base.Dispose(disposing);}
}
sealed class FaultingContent : HttpContent
{
    readonly byte[] prefix;internal bool Disposed;internal FaultingContent(string text){prefix=Encoding.UTF8.GetBytes(text);Headers.ContentType=new MediaTypeHeaderValue("text/event-stream");}
    protected override bool TryComputeLength(out long length){length=-1;return false;}
    protected override Task SerializeToStreamAsync(Stream stream,TransportContext context)=>throw new NotSupportedException();
    protected override Task<Stream> CreateContentReadStreamAsync()=>Task.FromResult<Stream>(new FaultStream(prefix));
    protected override void Dispose(bool disposing){Disposed=true;base.Dispose(disposing);}
}
sealed class FaultStream : MemoryStream
{
    bool failed;internal FaultStream(byte[] bytes):base(bytes){}
    public override int Read(byte[] buffer,int offset,int count){int n=base.Read(buffer,offset,count);if(n>0)return n;if(!failed){failed=true;throw new IOException("fixture read failure");}return 0;}
    public override async Task<int> ReadAsync(byte[] buffer,int offset,int count,CancellationToken token){await Task.Yield();return Read(buffer,offset,count);}
}
