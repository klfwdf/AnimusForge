internal static class Program
{
    private static int checks;
    private static void Check(bool value, string name) { checks++; if (!value) throw new Exception(name); }
    private static GuardrailEvalSnapshot Evaluation(params string[] ids)
    {
        var result = new GuardrailEvalSnapshot();
        foreach (var id in ids) result.Rules[id] = new GuardrailRuleEval { Hit = true, AmpScore = 0.8f, MatchedSeed = id };
        result.MentionedEntities.Entities.Add("mention");
        return result;
    }
    private static void Main()
    {
        AIConfigHandler.Evaluation = Evaluation("custom");
        AIConfigHandler.OnEvaluate = () => AIConfigHandler.Reload("new text", "new other");
        var hits = AIConfigHandler.GetHits("input", null, out var mentions);
        Check(hits.Count == 1 && hits[0].Instruction == "old text", "evaluation and rule body stay on one captured revision");
        Check(mentions.Entities.SequenceEqual(new[] { "mention" }), "mentions delivered from evaluation snapshot");
        mentions.Entities.Clear();
        Check(AIConfigHandler.Evaluation.MentionedEntities.Entities.Count == 1, "mentions do not leak mutable evaluation state");
        Check(AIConfigHandler.ReadRevision() == AIConfigHandler.LiveRevision(), "capture released after normal return");
        AIConfigHandler.OnEvaluate = null;
        AIConfigHandler.Evaluation = Evaluation("custom", "other");
        hits = AIConfigHandler.GetHits("input", new[] { "custom" }, out _);
        Check(hits.Count == 1 && hits[0].RuleId == "other", "exclusion and return cap apply after registry lookup");
        AIConfigHandler.ThrowFormat = true;
        try { AIConfigHandler.GetHits("input", null, out _); throw new Exception("format exception swallowed"); }
        catch (PreprocessFormatException) { }
        Check(AIConfigHandler.ReadRevision() == AIConfigHandler.LiveRevision(), "capture released after exception");
        AIConfigHandler.ThrowFormat = false;
        hits = AIConfigHandler.GetHits("", null, out _);
        Check(hits.Count == 0 && AIConfigHandler.ReadRevision() == AIConfigHandler.LiveRevision(), "empty input releases capture");
        Console.WriteLine("PromptJ03 production rule-hit checks=" + checks);
    }
}
