    private static bool TryGetPhraseEmbedding(string input, out float[] vector, long revision)
    {
        WarmupWorkerThread = Environment.CurrentManagedThreadId;
        if (revision == Volatile.Read(ref BlockWarmupRevision))
        {
            BlockedWarmupEntered.Set();
            ReleaseBlockedWarmup.Wait(TimeSpan.FromSeconds(5));
        }
        if (revision != _promptConfiguration.Capture().Revision) { vector = null; return false; }
        Interlocked.Increment(ref Warmed);
        vector = new[] { 1f, 0f };
        return true;
    }

    private static PromptRuleEvaluationPorts Ports()
    {
        return new PromptRuleEvaluationPorts
        {
            EligibleRules = () => Eligible,
            InputEmbedding = text => { InputCalls++; return text.Contains("duel") ? new[] { 1f, 0f } : text.Contains("reward") ? new[] { 0f, 1f } : null; },
            PhraseEmbedding = text => { PhraseCalls++; return text.Contains("duel") ? new[] { 1f, 0f } : text.Contains("reward") ? new[] { 0f, 1f } : null; },
            Rerank = (query, texts) => { RerankCalls++; return texts.Select(t => t.Contains("duel") ? 0.9f : 0.2f).ToList(); },
            AuxiliarySnapshot = (key, cap) => { AuxiliaryCalls++; return new GuardrailEvalSnapshot { Key = key, MatchMode = "auxiliary", ReturnCap = cap, Rules = new Dictionary<string, GuardrailRuleEval> { ["duel"] = new GuardrailRuleEval { RuleTag = "duel", Hit = true } } }; }
        };
    }

    public static void Main()
    {
        Eligible = new List<GuardrailRulePromptConfig>
        {
            new GuardrailRulePromptConfig { Id = "duel", Instruction = "duel", TriggerKeywords = new List<string> { "duel" } },
            new GuardrailRulePromptConfig { Id = "reward", Instruction = "reward", TriggerKeywords = new List<string> { "reward" } }
        };
        var ports = Ports();
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var first, null, true, ports) && first.MatchMode.StartsWith("rerank"), "semantic entry uses deterministic embedding and rerank");
        Check(InputCalls > 0 && PhraseCalls > 0 && RerankCalls > 0, "semantic provider callbacks were used");
        int inputAtHit = InputCalls;
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var hit, null, true, ports) && ReferenceEquals(first, hit) && InputCalls == inputAtHit, "same-key cache hit bypasses provider");
        Settings.GuardrailDirectTopN = 3;
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var mcm, null, true, ports) && !ReferenceEquals(first, mcm), "MCM hot edit invalidates cache key");
        _guardrailRuntimeTargetHeroId.Value = "hero-two";
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var target, null, true, ports) && !ReferenceEquals(mcm, target), "target identity isolates cache");
        Eligible = Eligible.Take(1).ToList();
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var eligible, null, true, ports) && !ReferenceEquals(target, eligible) && eligible.Rules.Count == 1, "eligibility isolates cache");
        TryGetGuardrailEvalSnapshot("duel", "", out var excluded, new[] { "duel" }, true, ports);
        Check(!ReferenceEquals(eligible, excluded) && (excluded == null || !excluded.Rules.ContainsKey("duel")), "exclusion isolates cache and suppresses rule");
        Settings.UseAuxiliaryRuleApi = true;
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var aux, null, true, ports) && aux.MatchMode == "auxiliary" && AuxiliaryCalls == 1, "auxiliary entry uses deterministic callback");
        _guardrailEvalCache.Clear();
        ports.AuxiliarySnapshot = (key, cap) => { AuxiliaryCalls++; return null; };
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var auxFallback, null, true, ports) && auxFallback.MatchMode.StartsWith("rerank") && AuxiliaryCalls == 2,
            "failed auxiliary result falls back to semantic pipeline");
        Settings.UseAuxiliaryRuleApi = false;
        _guardrailEvalCache.Clear();
        ports.Rerank = (query, texts) => new[] { 0.9f };
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var rerankFallback, null, true, ports) && rerankFallback.Rules.ContainsKey("duel"),
            "invalid rerank result retains semantic rule evidence");
        ports.Rerank = null;
        _guardrailEvalCache.Clear();
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var semanticOnly, null, true, ports) && semanticOnly.MatchMode.StartsWith("semantic"),
            "unavailable reranker uses semantic-only entry");
        _guardrailEvalCache.Clear();
        bool reloaded = false;
        ports.PhraseEmbedding = text =>
        {
            if (!reloaded) { reloaded = true; _promptConfiguration.Reload(() => "two", _ => "default"); }
            return new[] { 1f, 0f };
        };
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var old, null, true, ports) && reloaded, "in-flight old revision returns local result");
        Check(!_guardrailEvalCache.TryGet(old.Key, 1, out _), "late old revision cannot publish evaluation cache");
        Check(TryGetGuardrailEvalSnapshot("duel", "", out var current, null, true, ports) && !ReferenceEquals(old, current), "new revision computes its own result");
        _guardrailEvalCache.Clear();
        ports.InputEmbedding = _ => throw new InvalidOperationException("provider failed");
        Check(!TryGetGuardrailEvalSnapshot("duel", "", out _, null, true, ports), "provider exception uses existing failure return");
        AllowWarmupEligibility = true;
        int callerThread = Environment.CurrentManagedThreadId;
        var missionSeeds = CaptureGuardrailSemanticWarmupSeeds();
        Check(CaptureThread == callerThread && missionSeeds.Revision == _promptConfiguration.Capture().Revision && missionSeeds.Seeds.Count > 0,
            "mission seed capture stays on owning caller and pins revision");
        TryStartBackgroundSemanticWarmup("mission_start", missionSeeds);
        Check(SpinWait.SpinUntil(() => Volatile.Read(ref _guardrailWarmupState) == 2, TimeSpan.FromSeconds(5)) && Warmed > 0 && WarmupWorkerThread != callerThread,
            "mission warmup embeds on worker only after caller captures seed");
        int warmedBeforeStale = Warmed;
        Interlocked.Exchange(ref _guardrailWarmupState, 0);
        TryStartBackgroundSemanticWarmup("stale_mission", new PromptSemanticWarmupSeedBatch(1, new[] { "duel" }));
        Check(Volatile.Read(ref _guardrailWarmupState) == 0 && Warmed == warmedBeforeStale, "old revision seed cannot enter warmup gate");
        Interlocked.Exchange(ref _guardrailWarmupState, 0);
        TryStartBackgroundSemanticWarmup("session_launch");
        Check(SpinWait.SpinUntil(() => Volatile.Read(ref _guardrailWarmupState) == 2, TimeSpan.FromSeconds(5)) && CaptureThread == callerThread,
            "session entry captures seed on caller before scheduling worker");
        Interlocked.Exchange(ref _guardrailWarmupState, 0);
        OnnxCrossEncoderReranker.WarmupAllowed = true;
        RagWarmupCoordinator.TryStartBackgroundWarmup("mission_start", missionSeeds);
        Check(SpinWait.SpinUntil(() => Volatile.Read(ref _guardrailWarmupState) == 2
            && Logger.LastWarmupMessage?.Contains("source=rag_warmup_complete") == true, TimeSpan.FromSeconds(5)),
            "RAG completion forwards the caller-captured mission seed to semantic warmup");
        long oldRevision = _promptConfiguration.Capture().Revision;
        Volatile.Write(ref BlockWarmupRevision, (int)oldRevision);
        Interlocked.Exchange(ref _guardrailWarmupState, 0);
        TryStartBackgroundSemanticWarmup("old_worker", new PromptSemanticWarmupSeedBatch(oldRevision, new[] { "duel" }));
        Check(BlockedWarmupEntered.Wait(TimeSpan.FromSeconds(5)), "old worker entered embedding before reload");
        lock (_promptConfigurationReloadLock)
        {
            _promptConfiguration.Reload(() => "three", _ => "default");
            Interlocked.Exchange(ref _guardrailWarmupState, 0);
            Interlocked.Exchange(ref _guardrailWarmupVersion, -1L);
        }
        long newRevision = _promptConfiguration.Capture().Revision;
        TryStartBackgroundSemanticWarmup("new_worker", new PromptSemanticWarmupSeedBatch(newRevision, new[] { "reward" }));
        ReleaseBlockedWarmup.Set();
        Check(SpinWait.SpinUntil(() => Volatile.Read(ref _guardrailWarmupState) == 2
            && Logger.WarmupMessages.Any(message => message.Contains("complete source=old_worker"))
            && Logger.WarmupMessages.Any(message => message.Contains("complete source=new_worker")), TimeSpan.FromSeconds(5))
            && Volatile.Read(ref _guardrailWarmupVersion) == newRevision,
            "late old worker cannot overwrite new revision warmup gate");
        Console.WriteLine("PromptJ03 production evaluation checks=" + Checks);
    }
