    internal static class Program
    {
        static int checks;
        static WeeklyPromptSnapshot Weekly() => new("", "", "", "", "", "");
        static void Check(bool ok,string name){checks++;if(!ok)throw new Exception("FAIL "+name);}
        static PromptContextCaptureBannerlordPorts Ports()=>new(){
            IsPartyTransferEligible=()=>false,HasDuelRuntimeTarget=()=>true,WasRecentlyDefeated=_=>true,WasRecentlyReleased=_=>true,
            BuildPlayerDisplayName=(h,c,i)=>"player",BuildPrisonerStatus=_=>"prisoner",BuildHeroArmyFact=()=>"hero-army",BuildPlayerArmyFact=()=>"player-army",BuildResidentRecentActions=()=>"resident",
            BuildTriggeredRules=flags=>{Trace.Add("rules:"+flags.UseRewardContext+":"+flags.IncludeDuelStakeContext);return "rules";},
            IsWorldBulletinEnabled=()=>true,CaptureWorldBulletinSnapshot=()=>{Trace.Add("weekly-snapshot");return Weekly();},
            ShouldExcludeWeeklyShortReport=(r,s)=>{Trace.Add("weekly-exclude");return true;},BuildWeeklyShortReports=(e,s)=>{Trace.Add("weekly-short");return "short";},BuildWeeklyFullReports=(r,s)=>{Trace.Add("weekly-full");return "full";},ObserverKnowsPlayer=()=>true,LogStage=(s,d,i)=>{}
        };
        static PromptBuildRequest Request()=>new(){Input="input",HasAnyHero=true,PlayerClanTier=3,MinimumClanTier=1,TargetDisplayName="NPC"};
        static PromptRoutingResult Routing()=>new(){AuxiliaryRuleHitIds=new(){"kingdom_agenda"},Duel=new(){Hit=true},Marriage=new(){Hit=true},Surroundings=new(){Hit=true}};
        static void Capture(PromptBuildRequest request,PromptRoutingResult routing,Hero hero,CharacterObject character,WeeklyPromptSnapshot weekly,out PromptContextFlags flags,out PromptExtrasSections extras,out PromptEntityCapture entity)
            =>PromptContextCaptureBannerlordAdapter.CapturePromptSections(Ports(),request,routing,new(){AuxiliaryMentions=new(){Entities=new(){"mention"}}},new(),hero,character,weekly,out flags,out extras,out entity,out _);
        static void Main()
        {
            Trace.Events.Clear();DuelBehavior.HasResult=true;Capture(Request(),Routing(),new(),new(),null,out var flags,out var extras,out var entity);
            Check(Trace.Events.IndexOf("trust")<Trace.Events.IndexOf("duel")&&Trace.Events.Contains("trust"),"trust captured before destructive Duel consume");
            Check(Trace.Events.IndexOf("duel")<Trace.Events.IndexOf("rules:True:True"),"triggered rules see consumed Duel reward promotion");
            Check(flags.UseRewardContext&&flags.IncludeDuelStakeContext&&flags.PlayerWonLastDuel,"returned flags preserve Duel result");
            Check(extras.TrustPrompt=="trust"&&!string.IsNullOrWhiteSpace(extras.DuelResultLine),"pre-duel trust retained alongside result");
            Check(Trace.Events.Count(x=>x=="weekly-snapshot")==1&&Trace.Events.IndexOf("weekly-exclude")<Trace.Events.IndexOf("weekly-short")&&Trace.Events.IndexOf("weekly-short")<Trace.Events.IndexOf("weekly-full"),"single weekly snapshot shared in legacy layer order");
            Check(Trace.Events.IndexOf("entity")<Trace.Events.IndexOf("agenda")&&Trace.Events.Last()=="marriage","entity then agenda then final marriage effect");
            Check(extras.HeroArmyRuntimeFact=="hero-army"&&extras.PlayerArmyRuntimeFact=="player-army"&&extras.ResidentRecentActions=="resident"&&extras.NearbySettlementsDetail=="nearby"&&extras.ActivePolicyContext=="policy","all live leaf facts captured");
            Check(entity.HasContent&&entity.AgendaHasContent&&entity.MainPromptBlock=="entity-main"&&entity.PostprocessPromptBlock=="entity-post","complete entity and agenda outputs");
            Trace.Events.Clear();Capture(Request(),Routing(),new(),new(),Weekly(),out flags,out extras,out entity);
            Check(!Trace.Events.Contains("weekly-snapshot")&&Trace.Events.Contains("rules:False:False")&&extras.DuelResultLine==null,"retry does not reconsume result or recapture supplied weekly snapshot");
            var loanRouting=Routing();loanRouting.Loan=new(){Hit=true};Trace.Events.Clear();Capture(Request(),loanRouting,new(),new(),Weekly(),out flags,out extras,out entity);
            Check(Trace.Events.Contains("loan-date")&&Trace.Events.Contains("loan")&&!Trace.Events.Contains("trust"),"loan capture excludes trust");
            var merchant=Routing();merchant.Reward=new(){Hit=true};Trace.Events.Clear();Capture(Request(),merchant,null,new(),Weekly(),out flags,out extras,out entity);
            Check(Trace.Events.Contains("merchant")&&!Trace.Events.Contains("duel")&&!RomanceSystemBehavior.Enabled,"character merchant captures debt without Hero effects");
            var suppressed=Request();suppressed.SuppressDynamicRuleAndLore=true;Trace.Events.Clear();Capture(suppressed,Routing(),new(),new(),Weekly(),out flags,out extras,out entity);
            Check(!Trace.Events.Contains("lore")&&!Trace.Events.Any(x=>x.StartsWith("rules:"))&&!Trace.Events.Contains("entity")&&entity==null,"suppression preserves dynamic capture exclusions");
            var prefetched=Request();prefetched.UsePrefetchedLoreContext=true;prefetched.PrefetchedLoreContext="prefetched";Trace.Events.Clear();Capture(prefetched,Routing(),new(),new(),Weekly(),out flags,out extras,out entity);
            Check(extras.LoreContext=="prefetched"&&!Trace.Events.Contains("lore"),"prefetched lore avoids live lookup");
            RewardSystemBehavior.ThrowDebt=true;Trace.Events.Clear();Capture(Request(),Routing(),new(),new(),Weekly(),out flags,out extras,out entity);
            Check(!flags.PersistentAdpDebtPostprocess&&Trace.Events.Contains("trust"),"debt lookup failure retains old fallback and capture continuation");
            RewardSystemBehavior.ThrowDebt=false;DuelBehavior.HasResult=true;Trace.Events.Clear();var failingPorts=Ports();
            failingPorts.BuildTriggeredRules=_=>throw new Exception("rule-capture-failed");bool failed=false;
            try{PromptContextCaptureBannerlordAdapter.CapturePromptSections(failingPorts,Request(),Routing(),new(){AuxiliaryMentions=new()},new(),new(),new(),null,out _,out _,out _,out _);}catch(Exception e){failed=e.Message=="rule-capture-failed";}
            Check(failed&&!DuelBehavior.HasResult&&Trace.Events.Contains("duel")&&!Trace.Events.Contains("weekly-short")&&!Trace.Events.Contains("marriage"),"rule failure preserves consumed Duel and suppresses later effects without rollback");
            Console.WriteLine("PASS PromptContextCaptureAdapter checks="+checks+" actual-adapter=true dependencies=STUBBED live=NOT_RUN");
        }
    }
