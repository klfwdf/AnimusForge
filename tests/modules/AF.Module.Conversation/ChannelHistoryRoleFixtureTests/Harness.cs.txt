    public static class Program
    {
        private static string Field(object value,string name)
            => (string)value.GetType().GetProperty(name).GetValue(value);
        private static void Check(bool yes,string why) { if(!yes) throw new Exception("FAIL "+why); }
        private static ConversationMessage Message(string role,string speaker,string text)
            => new() { Role=role, SpeakerName=speaker, Content=text, GameDate="Autumn 11", GameDayIndex=100,
                GameHour=9, Scene="town square", SpeakerAgentIndex=speaker=="Ada"?7:8,
                SpeakerHeroId=speaker=="Ada"?"ada":"other", TargetAgentIndex=7 };
        private static void Case(string name,ConversationMessage input,string courierRole,string sceneRole,string token)
        {
            object courier=CourierDeliveryBehavior.Project(input),scene=ShoutBehavior.Project(input);
            Check(courier!=null && scene!=null,name+" both projected");
            Check(Field(courier,"role")==courierRole && Field(scene,"role")==sceneRole,name+" role");
            Check(Field(courier,"content").Contains(token) && Field(scene,"content").Contains(token),name+" fact preserved");
            Check(Field(courier,"content").Contains("Autumn 11") && Field(scene,"content").Contains("Autumn 11"),name+" date");
            Check(Field(courier,"content").Contains("town square") && Field(scene,"content").Contains("town square"),name+" scene");
            Console.WriteLine("PASS "+name);
        }
        public static void Main()
        {
            Case("assistant-self",Message("assistant","Ada","Ada: hello"),"assistant","assistant","hello");
            Case("user-player",Message("user","Player","Player: question"),"user","user","question");
            Case("assistant-other-observer",Message("assistant","Borin","Borin: heard"),"user","user","heard");
            Case("system-afef",Message("system","AFEF","[AFEF NPC行为补充] delivered"),"user","user","[AFEF NPC行为补充]");
            Case("system-other",Message("system","System","road blocked"),"user","user","road blocked");
            var repeated=Message("system","AFEF","[AFEF NPC行为补充] delivered");
            Check(Field(CourierDeliveryBehavior.Project(repeated),"content")==Field(CourierDeliveryBehavior.Project(repeated),"content"),"Courier duplicate stable projection");
            Check(Field(ShoutBehavior.Project(repeated),"content")==Field(ShoutBehavior.Project(repeated),"content"),"Scene duplicate stable projection");
            Console.WriteLine("PASS duplicate stable projection (builder dedupe not asserted)");
            var alias=Message("assistant","Alias","Alias: hello"); alias.SpeakerHeroId="ada";
            Case("known-alias-divergence",alias,"user","assistant","hello");
            Console.WriteLine("PASS cross-channel roles; alias divergence documented; full builders/live not run");
        }
    }
