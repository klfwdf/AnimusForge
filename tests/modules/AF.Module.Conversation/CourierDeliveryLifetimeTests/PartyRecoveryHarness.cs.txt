static class AddHeroToPartyAction
{
    public static Action<Hero, MobileParty, int> Fault;
    public static void Apply(Hero hero, MobileParty target, bool notification)
    {
        Fault?.Invoke(hero, target, 0);
        var source = hero.PartyBelongedTo;
        if (source != null)
        {
            source.MemberRoster.Rows.RemoveAll(row => row.Character == hero.CharacterObject);
            if (source.LeaderHero == hero) source.LeaderHero = null;
        }
        hero.PartyBelongedTo = null;
        Fault?.Invoke(hero, target, 1);
        target.MemberRoster.Rows.Add(new() { Character = hero.CharacterObject });
        hero.PartyBelongedTo = target;
        Fault?.Invoke(hero, target, 2);
    }
}
static class TakePrisonerAction
{
    public static int Captures;
    public static void Apply(PartyBase capturer, Hero hero)
    {
        Captures++;
        hero.PartyBelongedTo.MemberRoster.Rows.RemoveAll(row => row.Character == hero.CharacterObject);
        hero.PartyBelongedTo = null;
        hero.IsPrisoner = true;
        hero.PartyBelongedToAsPrisoner = capturer;
    }
}

    static int Checks;
    static void Check(bool condition, string name)
    {
        Checks++;
        if (!condition) throw new Exception("FAIL " + name);
    }
    static Hero NewHero(MobileParty party)
    {
        var hero = new Hero { StringId = Guid.NewGuid().ToString() };
        hero.CharacterObject = new() { StringId = hero.StringId, HeroObject = hero };
        Characters[hero.StringId] = hero.CharacterObject;
        AddHeroToPartyAction.Apply(hero, party, false);
        return hero;
    }
    public static void Main()
    {
        var owner = new CourierDeliveryBehavior();
        PartyBase.MainParty = MobileParty.MainParty.Party;
        foreach (int phase in new[] { -1, 0, 1, 2 })
        {
            var source = new MobileParty();
            var hero = NewHero(source);
            source.LeaderHero = hero;
            var session = new CourierSession();
            AddHeroToPartyAction.Fault = (_, _, point) => { if (point == phase) throw new Exception("transfer fault"); };
            bool returned = owner.ReturnCourierContentsToPlayer(session, source);
            Check(returned == (phase == -1 || phase == 2), "native transfer outcome " + phase);
            if (!returned)
            {
                Check(session.PendingReturnHeroIds.Contains(hero.StringId), "retain hero intent " + phase);
                var options = new JsonSerializerOptions { IncludeFields = true };
                session = JsonSerializer.Deserialize<CourierSession>(JsonSerializer.Serialize(session, options), options);
            }
            AddHeroToPartyAction.Fault = null;
            Check(owner.ReturnCourierContentsToPlayer(session, source), "retry after save/load " + phase);
            Check(hero.PartyBelongedTo == MobileParty.MainParty && CountRosterCharacter(MobileParty.MainParty.MemberRoster, hero.CharacterObject) == 1, "one main party hero " + phase);
            Check(source.LeaderHero == null && source.MemberRoster.TotalManCount == 0, "old party released " + phase);
        }
        var destroyed = new MobileParty();
        var other = new MobileParty();
        var movedHero = NewHero(other);
        var remainingHero = NewHero(destroyed);
        var losses = new CourierSession { CrewEntries = new() {
            new() { Id = movedHero.StringId, IsHero = true, Kind = "crew" },
            new() { Id = remainingHero.StringId, IsHero = true, Kind = "crew" } } };
        owner.TryMoveHeroLossesToDestroyer(losses, destroyed, new MobileParty().Party);
        Check(TakePrisonerAction.Captures == 1 && movedHero.PartyBelongedTo == other && !movedHero.IsPrisoner, "capture only present crew");
        var reassignedIntent = new List<string> { movedHero.StringId };
        Check(MoveWholeMemberRoster(destroyed.Party, MobileParty.MainParty.Party, reassignedIntent)
            && reassignedIntent.Count == 0 && movedHero.PartyBelongedTo == other, "later reassignment retires pending return");
        foreach (string laterState in new[] { "dead", "prisoner", "traveling", "settlement" })
        {
            var departed = NewHero(destroyed);
            destroyed.MemberRoster.Rows.Clear();
            departed.PartyBelongedTo = null;
            departed.IsDead = laterState == "dead";
            departed.IsPrisoner = laterState == "prisoner";
            departed.IsTraveling = laterState == "traveling";
            departed.StayingInSettlement = laterState == "settlement" ? new object() : null;
            Check(MoveWholeMemberRoster(destroyed.Party, MobileParty.MainParty.Party, new() { departed.StringId })
                && departed.PartyBelongedTo == null, "later native state supersedes return " + laterState);
        }
        destroyed.MemberRoster.Rows.Add(new() { Character = movedHero.CharacterObject });
        Check(!MoveWholeMemberRoster(destroyed.Party, MobileParty.MainParty.Party, new()), "stale source cannot reclaim another party hero");
        Check(movedHero.PartyBelongedTo == other, "other party remains authoritative");
        foreach (bool failRecovery in new[] { false, true })
        {
            owner = new() { ThrowPayload = true };
            var hero = NewHero(MobileParty.MainParty);
            var flow = new PendingCourierFlow();
            flow.CrewRoster.Rows.Add(new() { Character = hero.CharacterObject });
            AddHeroToPartyAction.Fault = (_, target, phase) => { if (failRecovery && target == MobileParty.MainParty && phase == 1) throw new Exception("recovery fault"); };
            bool threw = false;
            try { owner.CreateCourierSession(flow, "letter"); } catch { threw = true; }
            Check(threw, "dispatch failure reported");
            Check(owner._sessions.Count == (failRecovery ? 1 : 0), "dispatch recovery registration");
            if (failRecovery)
            {
                var pending = owner._sessions.Values.Single();
                Check(pending.Stage == "Returning" && CustomPartyComponent.Last.IsActive && pending.PendingReturnHeroIds.Contains(hero.StringId), "partial dispatch retained");
                AddHeroToPartyAction.Fault = null;
                Check(owner.ReturnCourierContentsToPlayer(pending, CustomPartyComponent.Last), "partial dispatch retry");
                owner.CompleteAndDestroyCourier(pending, CustomPartyComponent.Last);
            }
            AddHeroToPartyAction.Fault = null;
            Check(hero.PartyBelongedTo == MobileParty.MainParty && !CustomPartyComponent.Last.IsActive, "dispatch recovery completed");
        }
        foreach (int dispatchPhase in new[] { 0, 1, 2 })
        {
            owner = new();
            var hero = NewHero(MobileParty.MainParty);
            var flow = new PendingCourierFlow();
            flow.CrewRoster.Rows.Add(new() { Character = hero.CharacterObject });
            AddHeroToPartyAction.Fault = (_, target, phase) =>
            {
                if (target != MobileParty.MainParty && phase == dispatchPhase)
                    throw new Exception("outbound transfer interrupted");
            };
            bool threw = false;
            try { owner.CreateCourierSession(flow, "letter"); } catch { threw = true; }
            AddHeroToPartyAction.Fault = null;
            Check(threw && owner._sessions.Count == 0 && !CustomPartyComponent.Last.IsActive,
                "outbound transfer fault cleaned up " + dispatchPhase);
            Check(hero.PartyBelongedTo == MobileParty.MainParty
                && CountRosterCharacter(MobileParty.MainParty.MemberRoster, hero.CharacterObject) == 1,
                "outbound transfer fault returns hero once " + dispatchPhase);
        }
        owner = new();
        var normalHero = NewHero(MobileParty.MainParty);
        var normalFlow = new PendingCourierFlow();
        normalFlow.CrewRoster.Rows.Add(new() { Character = normalHero.CharacterObject });
        var normalSession = owner.CreateCourierSession(normalFlow, "letter");
        Check(owner._sessions.Count == 1 && normalHero.PartyBelongedTo == CustomPartyComponent.Last, "normal dispatch");
        Check(normalSession.PendingReturnHeroIds.Count == 0, "successful dispatch clears recovery intents");
        normalSession.PendingReturnHeroIds = null;
        Check(owner.ReturnCourierContentsToPlayer(normalSession, CustomPartyComponent.Last), "old null recovery field");
        owner.CompleteAndDestroyCourier(normalSession, CustomPartyComponent.Last);
        Check(normalHero.PartyBelongedTo == MobileParty.MainParty, "normal return");
        Console.WriteLine($"PASS courier party recovery checks={Checks} live=NOT_RUN");
    }
