    }

class Program {
    static int tests;
    static void Check(bool ok, string name) { if (!ok) throw new Exception("FAIL " + name); tests++; Console.WriteLine("PASS " + name); }
    static async Task Play() => await new Relay().Run();
    static string Field(object value, string name) => (string)value.GetType().GetProperty(name).GetValue(value);
    static int Stops => F.Messages.Count(m => !m.StartsWith("thinking:"));
    static async Task Main() {
        F.Reset(); await Play();
        Check(F.Speakers.SequenceEqual(new[] {17,27,17,27,17,27,17,27}), "two people can alternate eight utterances");
        Check(F.CandidateSets[1].Contains(17), "A remains eligible after A to B");
        Check(Stops == 1 && F.Messages.Last().Contains("发言次数上限"), "one limit stop notification");
        F.Reset(town:true); await Play();
        Check(F.Speakers.Count == 8 && F.Claims.SequenceEqual(new[] {17,27}), "town repeat speakers consume no new-person claim");
        F.Reset(3, true, "third"); await Play();
        Check(F.Speakers.SequenceEqual(new[] {17,27,17}) && F.Claims.SequenceEqual(new[] {17,27,37}), "town rejects new C after A B A");
        Check(Stops == 1 && F.Messages.Last().Contains("回应人数预算"), "town budget stop is shown once");
        F.Reset(town:true); F.EmptyHistory = true; await Play();
        Check(F.Claims.Count == 2 && F.Speakers.Count == 8, "budget admission does not depend on visible history count");
        foreach (var mode in new[] {"self", "missing", "invalid", "failed", "target-unavailable"}) {
            F.Reset(mode:mode); await Play();
            Check(F.Speakers.Count == 1 && Stops == 1, mode + " terminates once");
        }
        F.Reset(1); await Play(); Check(Stops == 1 && F.Speakers.Count == 1, "no candidates terminates once");
        foreach (var boundary in new[] {"scene_relay_turn_validate", "scene_relay_next_validate"}) {
            F.Reset(); F.UnavailableAt = boundary; await Play();
            Check(Stops == 1 && F.Speakers.Count <= 1, boundary + " unavailable notification");
        }
        foreach (var boundary in new[] {"scene_relay_turn_validate", "scene_relay_next_candidates", "scene_relay_next_validate", "scene_relay_gate"}) {
            F.Reset(); F.StaleAt = boundary; await Play();
            Check(Stops == 0 && F.Speakers.Count <= 1, boundary + " stale context has no late effect");
        }
        F.Reset(town:true); F.Budget.EndScene(); await Play();
        Check(F.Speakers.Count == 0 && Stops == 1 && !F.Messages[0].Contains("人数") && F.Logs.Any(s => s.Contains("InactiveScene")), "inactive budget keeps actual reason");
        F.Reset(); F.End = true; await Play(); Check(Stops == 1 && F.Speakers.Count == 1, "explicit end does not run relay");
        F.Reset(); F.Suppressed = true; await Play(); Check(Stops == 0 && F.Speakers.Count == 1, "battle speech suppression preserved");
        F.Reset(); var candidates = F.People.Skip(1).ToList();
        Check(Relay.NormalizeAutoGroupRelayPostprocessTagsForScene("[RELAY:17]", candidates, 17) == "[RELAY:17]", "normalizer permits self stop outside candidate list");
        Check(Relay.NormalizeAutoGroupRelayPostprocessTagsForScene("[RELAY:999][relay:27][RELAY:17]", candidates, 17) == "[RELAY:27]", "normalizer chooses first valid tag only");
        Check(Relay.NormalizeAutoGroupRelayPostprocessTagsForScene("[RELAY:999]", candidates, 17) == "", "normalizer rejects invented id");
        string prompt = Relay.BuildSceneRelayTargetListForPostprocess(candidates, 17, 27);
        Check(prompt.Contains("已经发言的人仍可再次接话") && prompt.Contains("[RELAY:17]"), "runtime prompt permits repeats and explicit self stop");
        Check(prompt.Contains("当前发言者编号: 17") && prompt.Contains("主对话对象编号: 27"), "runtime relay prompt identifies speaker/primary even when speaker omitted from candidates");
        var fallback = Relay.BuildAutoGroupRelayPostprocessRulesForScene(true, false);
        Check(fallback[0].Description.Contains("接力选择协议") && fallback[0].Description.Contains("接力终止"), "fallback prompt uses the same protocol");
        AIConfigHandler.Rules = new() { new() { Tag="[RELAY:接力编号]", Description=@@CONFIG_GROUP@@, SingleFramedNpcDescription=@@CONFIG_SINGLE@@ } };
        string configuredGroup = Relay.BuildAutoGroupRelayPostprocessRulesForScene(true, false)[0].Description;
        string configuredSingle = Relay.BuildAutoGroupRelayPostprocessRulesForScene(true, true)[0].Description;
        Check(configuredGroup.Contains("对话调度") && configuredGroup.Contains("动作许可门槛"), "canonical scene relay is scheduling, not consent-based action execution");
        Check(configuredGroup.Contains("当前NPC的回答只代表自己") && configuredGroup.Contains("整个群体都已回应"), "canonical group-address rule does not treat first NPC as everyone's answer");
        Check(configuredSingle.Contains("不能因只有一人被框选就忽略群体问话"), "single framed speaker still recognizes plural audience addressed by latest utterance");
        Check(configuredGroup.Contains("明确告别") && configuredGroup.Contains("没有合适接话者") && configuredGroup.Contains("当前发言者自己的编号"), "configured prompt retains legal self-stop and unsuitable-audience boundary");
        AIConfigHandler.Rules = new() { new() { Tag="[RELAY:接力编号]", Description="custom", SingleFramedNpcDescription="single-custom" } };
        Check(Relay.BuildAutoGroupRelayPostprocessRulesForScene(true, false)[0].Description == "custom" && Relay.BuildAutoGroupRelayPostprocessRulesForScene(true, true)[0].Description == "single-custom", "existing custom rules retained");
        F.Reset(3, mode:"three"); await Play();
        File.WriteAllText("three-turn-request-assembly.json", System.Text.Json.JsonSerializer.Serialize(new { speakers = F.Speakers, directPlayerResponse = F.DirectFlags, mainMessages = F.MainRequests, relayPostprocessRuntime = F.PostRuntimes, note = "Real production main callsite/history/role/assembly owners; detached NPC/network leaves. This is not a paid-provider or game acceptance." }, new System.Text.Json.JsonSerializerOptions { WriteIndented = true }));
        Check(F.Speakers.SequenceEqual(new[] {17,27,37}), "three speakers run under the real continuation controller before voluntary self stop");
        Check(F.DirectFlags.SequenceEqual(new[] {true,false,false}), "relayed turns retain first-turn-only direct action qualification");
        Check(F.MainRequests.Count == 3 && F.PostRuntimes.Count == 3, "every executed turn builds actual main messages and independent relay runtime");
        string firstMain = Field(F.MainRequests[0].Last(), "content"), secondMain = Field(F.MainRequests[1].Last(), "content"), thirdMain = Field(F.MainRequests[2].Last(), "content");
        Check(!firstMain.Contains("玩家暂时没有插话"), "first response keeps direct reply semantics");
        Check(secondMain.Contains("【本轮接力话题】") && secondMain.Contains(F.OriginInput) && secondMain.Contains("玩家暂时没有插话"), "second actual assembled request ends with original topic and continuation task");
        Check(Field(F.MainRequests[1].Last(), "role") == "user" && secondMain.Contains("上一位发言者: npc0 | 接力编号: 17") && secondMain.Contains(F.Reply(17)), "second final user message identifies actual predecessor and contribution");
        Check(F.MainRequests[1].Any(m => Field(m,"role") == "user" && Field(m,"content").Contains(F.Reply(17)) && !Field(m,"content").Contains("【本轮接力话题】")), "real history owner also preserves predecessor as overheard NPC speech");
        Check(thirdMain.Contains("当前发言轮次: 3") && thirdMain.Contains("上一位发言者: npc1 | 接力编号: 27") && thirdMain.Contains(F.Reply(27)), "third actual request updates predecessor instead of copying first speaker");
        Check(thirdMain.Contains("本轮已发言编号: 17, 27"), "third main request uses this-round spoken state before current speech");
        Check(F.PostRuntimes[1].Contains(F.OriginInput) && F.PostRuntimes[1].Contains(F.Reply(17)) && F.PostRuntimes[1].Contains("本轮已发言编号: 17, 27"), "second post runtime retains topic and predecessor independently of latest player input");
        Check(F.PostRuntimes[2].Contains("本轮已发言编号: 17, 27, 37") && F.PostRuntimes[2].Contains("[RELAY:37]"), "third post runtime includes accepted spoken state and preserves self-end protocol");
        Check(F.PreflightRecords == 0 && F.SettingsReads == 0, "verbose false completely skips request recorder and settings lookup");
        F.Reset(mode:"self"); F.Verbose = true; F.Settings.EnableTokenStatsLog = false; await Play();
        Check(F.PreflightRecords == 0, "token log false completely skips recorder");
        F.Reset(mode:"self"); F.Verbose = true; F.Settings = null; await Play();
        Check(F.PreflightRecords == 0, "missing settings skip recorder");
        F.Reset(mode:"self"); F.Verbose = true; await Play();
        Check(F.PreflightRecords == 1 && F.PreflightMax == 5000 && F.PreflightMode == "scene_relay_turn_preflight", "opt-in records prepared request once with existing mode and max metadata");
        Check(!F.PreflightBody.Contains(F.Settings.ApiKey), "request recorder arguments contain no provider credential");
        Console.WriteLine("PASS " + tests + " production relay boundary assertions; no live game or network");
    }
