    internal static class SystemProbe
    {
        internal static bool ThrowAfterIssue,WorldExit;
        static bool Enabled,OwnerCurrent;
        static int MainThread,Epoch,Session;
        static readonly List<string> Steps=new();
        static List<Action> Queue;
        static AnimusForge.Refactor.Runtime.PendingOperationRegistry Pending;
        static SceneSystemNpcShoutRuntime Runtime;
        internal static void Atom(string step){if(!Enabled)return;Require(Thread.CurrentThread.ManagedThreadId==MainThread,"game atom off main");Steps.Add(step);}
        static void Require(bool value,string text){if(!value)throw new Exception("ASSERT "+text);}
        static Agent Setup()
        {
            Enabled=true;ThrowAfterIssue=WorldExit=false;OwnerCurrent=true;MainThread=Thread.CurrentThread.ManagedThreadId;Epoch=Session=1;Steps.Clear();Queue=new();Pending=new();
            Probe.IssueEffects=Probe.Speech=Probe.History=Probe.PrivateHistory=Probe.Facts=0;Probe.TransferPartial=false;SaveRuntimeGuard.Generation=1;
            Mission.Current=new();var agent=new Agent{Index=7,Character=new CharacterObject{HeroObject=new Hero()}};Mission.Current.Agents.Add(agent);
            var ports=new SceneSpeechEffectPorts
            {
                ShowNpcSpeechOutput=(n,a,text,tts,attach,suppress)=>{Atom("speech");Probe.Speech++;if(Probe.FailStep=="system_speech")throw new Exception("speech_failed");return new();},
                RecordResponseForAllNearbySafe=(list,i,name,text,receipt)=>{Atom("history");Probe.History++;return true;},
                PersistNpcSpeechToNamedHeroes=(i,name,text,list,receipt)=>{Atom("private");Probe.PrivateHistory++;return true;},
                RecordSystemFactForNearbySafe=(list,fact)=>{Require(fact.StartsWith("AFEF"),"partial actual fact");Atom("fact");Probe.Facts++;},
                TryConsumeSceneNpcSurrenderTag=(NpcDataPacket n,ref string t,out Hero h,out CharacterObject c,out int i)=>{h=null;c=null;i=-1;return false;},

                AddAgentToStareList=(a,interrupt)=>Atom("stare"),
                ScheduleWorldMapMissionExitAfterSpeech=(i,p)=>Atom("exit"),GetDuelLiteralHit=()=>false
            };

            Runtime=new(ports,dispatcher,()=>Epoch,()=>Session,()=>OwnerCurrent);return agent;
        }
        static void Settle()
        {var watch=Stopwatch.StartNew();while(Pending.Count!=0){Require(watch.ElapsedMilliseconds<3000,"pending registry did not release");Thread.Sleep(1);}}
        internal static void Run()
        {
            int pass=0;void Case(string name,Action action){Probe.FailStep=null;action();pass++;Console.WriteLine("PASS "+name);}

            foreach(string change in new[]{"mission","generation","epoch","session","owner","target_missing","agent_reused","target_inactive"})

            Case("system-shout-registry-retire-strong-yield-before-late-queue",()=>{var a=Setup();Task<bool> task=Runtime.Enqueue(a,"reply");Require(Pending.Count==1&&!task.IsCompleted,"actual dispatch task registration");Pending.Reset();Settle();Require(SpinWait.SpinUntil(()=>task.IsCompleted,3000)&&task.Result==false,"actual owner task must settle without any queue tick");Require(Steps.Count==0,"retirement must not need queue drain");Queue[0]();Queue[0]();Require(Steps.Count==0,"retired queue must remain harmless");});
            Case("system-shout-observed-transfer-fact-notification-order",()=>{var a=Setup();Probe.TransferPartial=true;Runtime.Enqueue(a,"reply");Queue[0]();Settle();Require(Probe.Facts==1&&Steps.SequenceEqual(new[]{"issue","fact","private_fact","notification","speech","stare","history","private"}),"actual partial AFEF+notification preserved");});
