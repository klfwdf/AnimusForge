    public static class MyBehavior {
        public static MemoryProbe Owner;
        private static string GetCurrentGameDateTextSafe() => TaleWorlds.CampaignSystem.CampaignTime.Now.ToString();
        public static string DateForTests(int originDay, int currentDay) => MemoryHistoryCommitBannerlordAdapter.ResolveInteractionMemoryOriginGameDate(originDay, currentDay);
        public static string ResolveCurrentMemorySceneLabelForExternal() => "delivery-scene";
        public static bool TryPrepareExternalDialogueHistoryRecoveryIdentity(InteractionMemoryCommit commit, bool isNonHero,
            string npcName, out string recoveryId, out string payloadHash, out string error) {
            recoveryId=""; payloadHash=""; error="";
            if (Owner == null) { error="memory_owner_missing"; return false; }
            if (!Owner.Ready) { error="memory_recovery_not_activated"; return false; }
            if (Owner.RejectPermanent) { error="memory_target_ineligible"; return false; }
            recoveryId=commit.SessionId.PadRight(64, 'A').Substring(0,64);
            payloadHash=commit.SubjectId.PadRight(64, 'E').Substring(0,64);
            return true;
        }
        public static MemoryCommitResult CommitExternalDialogueHistoryRecoverable(InteractionMemoryCommit commit,
            bool isNonHero, string npcName) {
            Owner.Attempts++;
            Owner.LastDay=commit.GameDay;
            Owner.LastLocation=commit.LocationId;
            Owner.LastHour=commit.GameHour;
            Owner.LastDate=DateForTests(commit.GameDay,(int)TaleWorlds.CampaignSystem.CampaignTime.Now.ToDays);
            Owner.LastHistory=commit.AssistantText;
            foreach (FactRecord fact in commit.ConfirmedFacts) Owner.LastFact=fact.Text;
            Owner.Sessions.Add(commit.SessionId);
            if (Owner.Accept) { Owner.Completed=true; return new MemoryCommitResult(MemoryCommitStatus.Applied); }
            return new MemoryCommitResult(MemoryCommitStatus.Failed);
        }
        internal static InteractionMemoryRecoveryLookupStatus GetExternalDialogueHistoryRecoveryStatus(
            string recoveryId, string subjectId, string payloadHash) {
            if (Owner == null || !Owner.Ready) return InteractionMemoryRecoveryLookupStatus.Unavailable;
            if (Owner.Missing) return InteractionMemoryRecoveryLookupStatus.Missing;
            return Owner.Completed ? InteractionMemoryRecoveryLookupStatus.Completed : InteractionMemoryRecoveryLookupStatus.Pending;
        }
    }

    public static class Program {
        private static void Require(bool yes, string why) { if (!yes) throw new Exception("FAIL " + why); }
        private static CourierDeliveryBehavior.CourierSession Session(string id) => new() { Id=id };
        public static void Main() {
            Require(MyBehavior.DateForTests(100,100)=="synthetic-campaign-date:100", "same-day calendar");
            Require(MyBehavior.DateForTests(84,100)=="synthetic-campaign-date:84", "cross-day uses origin calendar");
            TaleWorlds.CampaignSystem.CampaignTime.EmptyDate=true;
            Require(MyBehavior.DateForTests(84,100)=="第 84 日", "empty calendar conservative origin day");
            TaleWorlds.CampaignSystem.CampaignTime.EmptyDate=false;
            TaleWorlds.CampaignSystem.CampaignTime.FailDate=true;
            Require(MyBehavior.DateForTests(84,100)=="第 84 日", "failed calendar conservative origin day");
            TaleWorlds.CampaignSystem.CampaignTime.FailDate=false;
            MyBehavior.Owner=new MemoryProbe { Accept=true };
            var accepted=new CourierDeliveryBehavior(); var a=Session("accepted"); accepted.Deliver(a);
            Require(a.DeliveryApplied && accepted.LettersAdded==1 && accepted.Notices==1 && accepted.Completed==1, "accepted visible delivery");
            Require(MyBehavior.Owner.Attempts==1 && accepted.PendingCount==0, "accepted immediate memory once");
            accepted.Tick(); Require(MyBehavior.Owner.Attempts==1 && accepted.LettersAdded==1, "accepted duplicate tick");

            MyBehavior.Owner=new MemoryProbe { Accept=false };
            var pending=new CourierDeliveryBehavior(); var p=Session("pending"); pending.Deliver(p);
            Require(p.DeliveryApplied && pending.LettersAdded==1 && pending.Notices==1 && pending.Completed==1, "pending visible delivery once");
            Require(MyBehavior.Owner.Attempts==1 && pending.PendingCount==1, "memory failure retains intent");
            pending.Tick(); Require(MyBehavior.Owner.Attempts==1 && pending.PendingCount==1, "pending does not replay commit");
            var pendingLoaded=new CourierDeliveryBehavior(); pendingLoaded.LoadJson(pending.SnapshotJson());
            pendingLoaded.Tick(); Require(MyBehavior.Owner.Attempts==1 && pendingLoaded.PendingCount==1, "loaded pending does not replay commit");
            MyBehavior.Owner.Completed=true; pendingLoaded.Tick();
            Require(pendingLoaded.PendingCount==0 && pendingLoaded.LettersAdded==0, "loaded pending observes completion only");
            Require(pending.LettersAdded==1 && pending.Completed==1, "memory completion cannot replay physical delivery");

            MyBehavior.Owner=new MemoryProbe { Accept=false };
            var ambiguous=new CourierDeliveryBehavior(); ambiguous.Deliver(Session("ambiguous"));
            Require(ambiguous.PendingCount==1 && MyBehavior.Owner.Attempts==1, "ambiguous setup");
            MyBehavior.Owner.Missing=true; ambiguous.Tick();
            Require(ambiguous.QuarantinedCount==1 && MyBehavior.Owner.Attempts==1, "missing after attempt quarantined without replay");

            MyBehavior.Owner=new MemoryProbe { RejectPermanent=true };
            var ineligible=new CourierDeliveryBehavior(); ineligible.Deliver(Session("ineligible"));
            Require(ineligible.QuarantinedCount==1 && ineligible.LettersAdded==1, "permanent rejection preserves visible delivery and quarantines");

            MyBehavior.Owner=null;
            var missing=new CourierDeliveryBehavior(); var m=Session("missing-owner"); missing.Deliver(m);
            Require(m.DeliveryApplied && missing.PendingCount==1 && missing.LettersAdded==1, "missing owner preserves intent and delivery");
            string saved=missing.SnapshotJson();
            var loaded=new CourierDeliveryBehavior(); loaded.LoadJson(saved);
            TaleWorlds.CampaignSystem.CampaignTime.Now.ToDays=101;
            TaleWorlds.CampaignSystem.CampaignTime.Now.ToHours=2424;
            MyBehavior.Owner=new MemoryProbe { Accept=true };
            missing.Tick(); Require(MyBehavior.Owner.Attempts==0, "retired instance cannot write");
            loaded.Tick(); Require(MyBehavior.Owner.Attempts==1 && loaded.PendingCount==0 && MyBehavior.Owner.LastDay==100
                && MyBehavior.Owner.LastLocation=="delivery-scene"
                && MyBehavior.Owner.LastHour==0
                && MyBehavior.Owner.LastDate=="synthetic-campaign-date:100"
                && MyBehavior.Owner.LastHistory.Contains("visible letter")
                && MyBehavior.Owner.LastFact=="[AFEF NPC行为补充] delivered",
                "loaded intent writes memory once at original delivery day and scene");
            Require(loaded.LettersAdded==0 && loaded.Completed==0 && loaded.Notices==0, "loaded intent is memory-only");
            loaded.Tick(); Require(MyBehavior.Owner.Attempts==1, "loaded completed tick duplicate");

            var legacy=new CourierDeliveryBehavior(); legacy.LoadJson("{\"SenderCooldownUntilDays\":{}}"); legacy.Tick();
            Require(legacy.PendingCount==0, "old save missing field");
            var corrupt=new CourierDeliveryBehavior(); corrupt.LoadCorrupt(); corrupt.Tick();
            Require(corrupt.PendingCount==1 && corrupt.QuarantinedCount==1, "corrupt identity quarantined");

            var overflow=new CourierDeliveryBehavior(); overflow.LoadOverflow();
            int beforeOverflowAttempts=MyBehavior.Owner.Attempts;
            overflow.Tick(); overflow.Deliver(Session("overflow-new"));
            Require(overflow.PendingCount==129 && overflow.SnapshotCount()==129,
                "oversize save data preserved");
            Require(MyBehavior.Owner.Attempts==beforeOverflowAttempts && overflow.Held==1 && overflow.LettersAdded==0,
                "oversize queue fail closed without scanning or delivery");

            MyBehavior.Owner=null;
            var capped=new CourierDeliveryBehavior();
            for (int i=0; i<129; i++) capped.Deliver(Session("cap-"+i));
            Require(capped.PendingCount==128 && capped.LettersAdded==128 && capped.Held==1, "bounded pending backpressure before physical delivery");
            MyBehavior.Owner=null;
            var roundRobin=new CourierDeliveryBehavior();
            roundRobin.Deliver(Session("c")); roundRobin.Deliver(Session("a")); roundRobin.Deliver(Session("b"));
            var roundRobinLoaded=new CourierDeliveryBehavior(); roundRobinLoaded.LoadJson(roundRobin.SnapshotJson());
            MyBehavior.Owner=new MemoryProbe { Accept=true };
            roundRobinLoaded.Tick();
            Require(MyBehavior.Owner.Attempts==1 && roundRobinLoaded.PendingCount==2, "one pending intent per tick");
            roundRobinLoaded.Tick(); roundRobinLoaded.Tick(); roundRobinLoaded.Tick();
            Require(string.Join(",",MyBehavior.Owner.Sessions)=="a,b,c" && roundRobinLoaded.PendingCount==0,
                "round robin stable order and completed intents removed");
            Require(roundRobinLoaded.LettersAdded==0 && roundRobinLoaded.Notices==0 && roundRobinLoaded.Completed==0,
                "batched recovery never replays physical effects");
            Console.WriteLine("PASS Courier delivered memory accepted/pending/load/missing/retired/old-save/corrupt/bounded/no-redelivery/calendar/frozen-facts-history/one-intent-round-robin");
        }
