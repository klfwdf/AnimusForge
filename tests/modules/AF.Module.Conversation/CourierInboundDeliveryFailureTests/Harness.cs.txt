    public static class Program
    {
        private static void Require(bool condition, string message)
        {
            if (!condition) throw new Exception("FAIL " + message);
        }

        private static void Check(string label, MyBehavior owner, int attempts, int accepted)
        {
            Campaign.Current = new Campaign { Owner = owner };
            var courier = new CourierDeliveryBehavior();
            var session = new CourierDeliveryBehavior.CourierSession();
            courier.RunForTest(session);
            Require(session.DeliveryApplied, label + " delivery flag");
            Require(courier.LettersAdded == 1 && courier.Notices == 1 && courier.Completed == 1,
                label + " visible delivery must continue");
            Require((owner?.Attempts ?? 0) == attempts, label + " history call must be attempted");
            Require((owner?.Accepted ?? 0) == accepted, label + " accepted history count");
        }

        public static void Main()
        {
            Check("accepted", new MyBehavior { Accept = true }, 1, 1);
            Check("rejected", new MyBehavior { Accept = false }, 1, 0);
            Check("exception", new MyBehavior { Throw = true }, 1, 0);
            Check("no-owner", null, 0, 0);
            Console.WriteLine("PASS CourierInboundDelivery source-derived accepted=1 rejectedStillDelivered=1 exceptionStillDelivered=1 noOwnerStillDelivered=1 live=NOT_RUN");
        }
    }
