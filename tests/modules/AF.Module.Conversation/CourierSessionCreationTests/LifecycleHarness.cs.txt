static partial class Program
{
    static void LifecycleCases()
    {
        foreach (CourierPayloadMode mode in Enum.GetValues<CourierPayloadMode>())
        {
            var owner=CourierDeliveryBehavior.NewOwner();owner.Open("lifecycle");owner.SetMode(mode);owner.ResetLifecycle();
            using var client=Client();var op=client.SubmitCourier(client.CaptureCourierContextTicket(),"life","letter");
            owner.PrepareReply();
            Check(op.Snapshot.Courier.Has(CoreCourierAcceptedSteps.ReplyPrepared)&&op.Snapshot.Reply==""
                &&!op.Completion.IsCompleted&&Life.Actions==0&&Life.History.Count==0,"preflight-no-effects-or-body-"+mode);
            Life.PayloadReenter=()=>owner.Arrive();Life.Reenter=()=>owner.Arrive();owner.Arrive();owner.Arrive();
            Check(Life.Payloads==1&&Life.Actions==1&&Life.History.Count==2,"arrival-and-action-reserved-before-reentry-"+mode);
            Check(op.Snapshot.Courier.Has(CoreCourierAcceptedSteps.Payload|CoreCourierAcceptedSteps.Actions|CoreCourierAcceptedSteps.ReplyHistory)
                &&op.Snapshot.Reply==""&&!op.Completion.IsCompleted,"arrival-history-not-transport-completion-"+mode);
            Check(Life.History[0].player?.Contains("【来信】")==true&&Life.History[0].reply==null&&Life.History[0].fact.StartsWith("[AFEF")
                &&Life.History[1].player==null&&Life.History[1].reply=="【回信】reply"&&Life.History[1].fact.StartsWith("[AFEF NPC"),"history-roles-facts-clean-tags-"+mode);
            owner.ReturnNow();
            Check(op.Completion.IsCompleted&&op.Snapshot.State==CoreDialogueState.Completed&&op.Snapshot.Reply=="reply"
                &&op.Snapshot.Courier.Complete&&owner.LetterCount==1&&Life.Letters==1,"only-letter-and-cleanup-complete-"+mode);
            owner.Retire();Released(owner,"lifecycle-"+mode);
        }
        foreach(string fault in new[]{"history","action","payload","inventory","record","cleanup","contents","destroyed","missing","recipient","generation","retire","replacement"})
        {
            var owner=CourierDeliveryBehavior.NewOwner();owner.Open(fault);owner.SetMode(CourierPayloadMode.Give);owner.ResetLifecycle();
            using var client=Client();var op=client.SubmitCourier(client.CaptureCourierContextTicket(),fault,"letter");
            owner.PrepareReply();
            if(fault=="history")Life.FailHistory=true;
            if(fault=="action")Life.FailAction=true;
            if(fault=="payload")Life.FailPayload=true;
            if(fault=="recipient")owner.KillRecipient();
            owner.Arrive();
            if(fault=="inventory")Life.FailInventory=true;
            if(fault=="record")Life.FailRecord=true;
            if(fault=="contents")Life.FailContents=true;
            if(fault=="cleanup")CourierFixture.FailCleanup=true;
            if(fault=="destroyed")owner.DestroyNow();
            else if(fault=="missing")owner.EndTransport(true);
            else if(fault=="generation"){SaveRuntimeGuard.Generation++;owner.CheckIdentity();}
            else if(fault=="retire")owner.Retire();
            else if(fault=="replacement")owner.ReplaceTransport();
            else owner.ReturnNow();
            Check(op.Completion.IsCompleted&&op.Snapshot.State==CoreDialogueState.Failed,"terminal-failure-settled-"+fault);
            if(fault=="destroyed"||fault=="missing")
                Check(op.Snapshot.Courier.Has(CoreCourierAcceptedSteps.Actions|CoreCourierAcceptedSteps.ReplyHistory)
                    &&op.Snapshot.Reply==""&&op.Snapshot.Courier.Transport==(fault=="destroyed"?CoreCourierTransportOutcome.Destroyed:CoreCourierTransportOutcome.Missing),"partial-confirmed-effects-survive-return-loss-"+fault);
            if(fault=="cleanup")Check(op.Snapshot.Reply=="reply"&&op.Snapshot.Courier.Has(CoreCourierAcceptedSteps.ReplyDelivered)
                &&op.Snapshot.Courier.Transport==CoreCourierTransportOutcome.Unconfirmed,"delivered-reply-survives-cleanup-failure");
            if(fault=="record"||fault=="inventory")Check(op.Snapshot.Reply==""&&!op.Snapshot.Courier.Has(CoreCourierAcceptedSteps.ReplyDelivered),"no-body-without-inventory-and-record-"+fault);
            if(fault=="action")Check(Life.Actions==1,"partial-action-exception-not-replayed-on-return");
            owner.Retire();Released(owner,"fault-"+fault);
        }
        var stale=CourierDeliveryBehavior.NewOwner();stale.Open("old-response");stale.ResetLifecycle();
        using(var client=Client())
        {
            var op=client.SubmitCourier(client.CaptureCourierContextTicket(),"old-response","letter");
            stale.CheckOldFinalizer();
            Check(!op.Snapshot.Courier.Has(CoreCourierAcceptedSteps.ReplyPrepared)&&!stale.HasReply,"old-run-response-cannot-write-new-run");
            stale.Retire();Released(stale,"old-response");
        }
    }
}
