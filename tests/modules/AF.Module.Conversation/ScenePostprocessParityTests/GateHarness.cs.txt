public sealed class ManualContext:SynchronizationContext
{
    private BlockingCollection<(SendOrPostCallback,object)> queue=new();
    public override void Post(SendOrPostCallback callback,object state)=>queue.Add((callback,state));
    public Task Capture(Func<Task> body)
    {
        var prior=Current;SetSynchronizationContext(this);try{return body();}finally{SetSynchronizationContext(prior);}
    }
    public void RunNext()
    {
        if(!queue.TryTake(out var item,TimeSpan.FromSeconds(3)))throw new Exception("wait continuation was not posted");
        var prior=Current;SetSynchronizationContext(this);try{item.Item1(item.Item2);}finally{SetSynchronizationContext(prior);}
    }
    public void WaitForPending()
    {
        if(!SpinWait.SpinUntil(()=>queue.Count>0,TimeSpan.FromSeconds(3)))throw new Exception("wait continuation not ready");
    }

    static void DrainFaultContinuation(TaskCompletionSource<bool> source)
    {

    static void LateTask()
    {

    static void LateWaiter(bool newSession)
