            p.BuildSceneGuideArrivalCallback(next, false)(); Require(next.ArrivalTriggered && p.Arrivals == 1, "new-command-still-admitted");
        });
        Case("mission-end-rejects-late-arrival-same-agent-index", () => {
            var p = new LifecycleProbe(); var guide = new ActiveSceneGuideRequest { GuideAgentIndex = 7 }; var callback = p.BuildSceneGuideArrivalCallback(guide, true);

        Case("mission-replacement-does-not-clean-same-index-new-agent", () => {
            var p = new LifecycleProbe(); p._activeSceneSummonRequests.Add(new()); p._activeSceneGuideRequests.Add(new()); p._activeSceneReturnJobs.Add(new());

        });
        Case("guide-arrival-is-once-and-preserves-door-receipt", () => {

        });
        return passed;
    }
