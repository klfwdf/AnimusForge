internal static class Program
{
    private static int _checks;
    private static void Check(bool good,string name){if(!good)throw new Exception("FAIL "+name);_checks++;Console.WriteLine("PASS "+name);}
    private sealed class Outcome {internal string Text,Error,Popup;internal string[] Trace;internal int Calls,Rollbacks,WrongThread;internal NativeConversationMainReplyStatus Status;}
    private static Outcome Run(string scenario,string reply,bool current)
    {
        Fixture.Reset();var owner=new ShoutBehavior();var outcome=new Outcome();
        Task<string> task=current ? Task.Run(()=>owner.StartCurrent()) : Task.Run(()=>owner.StartOriginal());
        Check(Fixture.ProviderEntered.Task.Wait(5000),"provider-entered:"+scenario);
        Check(!task.IsCompleted,"forced-yield:"+scenario);
        owner.Change(scenario);
        if(scenario=="provider-throw") Fixture.Reply.SetException(Fixture.Failure);else Fixture.Reply.SetResult(reply);
        try{outcome.Text=Fixture.Finish(task);}catch(Exception error){Check(ReferenceEquals(error,Fixture.Failure),"same-exception:"+scenario);outcome.Error=error.Message;}
        outcome.Status=Fixture.LastStatus;outcome.Trace=Fixture.Trace.ToArray();outcome.Calls=Fixture.Calls;outcome.Rollbacks=Fixture.Rollbacks;outcome.WrongThread=Fixture.WrongThread;outcome.Popup=Fixture.Popup;
        return outcome;
    }
    private static void Main()
    {
        Check(!default(NativeConversationMainReplyResult).CanContinue,"default-result-stops");
        var cases=new (string Name,string Reply,NativeConversationMainReplyStatus Status)[]{
