internal static class Program {
 private static int Checks;
 private static void Check(bool value,string name){Checks++;if(!value)throw new InvalidOperationException(name);}
 private static async Task WaitFor(Func<bool> predicate){for(int i=0;i<1000&&!predicate();i++)await Task.Delay(1);Check(predicate(),"worker-started");}
 static async Task Main(){try{await Run();}catch(Exception error){Console.WriteLine("FAIL "+error.GetType().FullName+": "+error.Message);Console.WriteLine(error.StackTrace);Environment.ExitCode=1;}}
 static async Task Run(){
  var provenance=new Fixture{Mode="ready"};
  Check(!NpcInitiatedOpeningRouter.HasPendingNativeOpeningForCurrentConversation(),"ordinary intercepted/native encounter has no proactive opening obligation");
  Check(!ProactiveNpcRequestBehavior.TryBuildMenuText(provenance.Hero,out _,out _),"no active request cannot display wants-talk promise");
  int unchangedLogCount=Logger.Entries.Count;for(int i=0;i<20;i++)ProactiveNpcRequestBehavior.TryBuildMenuText(provenance.Hero,out _,out _);
  Check(Logger.Entries.Count==unchangedLogCount,"repeated unchanged menu initialization emits no log flood");
  ProactiveNpcRequestBehavior.Instance.Start(provenance.Hero);
  Check(ProactiveNpcRequestBehavior.TryBuildMenuText(provenance.Hero,out var title,out var body)&&body.Contains("有事想找你谈谈"),"actual menu producer promises talk only for matching active hero");
  Check(!NpcInitiatedOpeningRouter.HasPendingNativeOpeningForCurrentConversation(),"menu promise precedes explicitly marked native opening");
  ProactiveNpcRequestBehavior.Instance.Open(provenance.Hero);
  Check(NpcInitiatedOpeningRouter.HasPendingNativeOpeningForCurrentConversation(),"actual mark -> opening owner -> pending gate remains visible");
  ShoutBehavior.TargetHero=new Hero{StringId="different"};Check(!NpcInitiatedOpeningRouter.HasPendingNativeOpeningForCurrentConversation(),"pending opening refuses mismatched target rather than creating wrong NPC dialogue");
  ShoutBehavior.TargetHero=provenance.Hero;Check(await provenance.Submit()=="body"&&provenance.Runs==1&&provenance.Visible==1,"matching pending producer -> current admission -> worker -> visible reply");
  Check(!NpcInitiatedOpeningRouter.HasPendingNativeOpeningForCurrentConversation(),"consumed pending no duplicate automatic start");
  var autoFixture=new Fixture{Mode="ready"};var auto=new OpeningGateLeaf{Submit=()=>autoFixture.Submit()};
  auto.Tick();Check(auto.Starts==0,"actual UI automatic-start gate does not create unmarked encounter request");
  autoFixture.Open("lord");ShoutBehavior.ForceBusy=true;auto.Tick();Check(auto.Starts==0&&NpcInitiatedOpeningRouter.HasPendingNativeOpeningForCurrentConversation(),"actual auto gate busy retains marked obligation");
  ShoutBehavior.ForceBusy=false;auto.Tick();await auto.Pending;Check(auto.Starts==1&&auto.Shown==1&&autoFixture.Visible==1,"actual pending UI gate invokes real current admission and displays one ready reply");
  auto.Tick();Check(auto.Starts==1,"actual automatic-start flag prevents repeated submissions");
  var realHarmony=new Harmony("AF.Native.CurrentProducer.Illustrator");ConversationIllustrationPatch.Install(realHarmony);
  var entry=new Fixture{Mode="ready"};int streams=0,posts=0,readyCalls=0;Hero seenHero=null;CharacterObject seenCharacter=null;
  Action<string> stream=t=>{Check(t=="chunk","facade keeps stream callback text");streams++;};Action<string> post=t=>posts++;Action<string,Hero,CharacterObject> ready=(text,hero,character)=>{readyCalls++;seenHero=hero;seenCharacter=character;};
  Check(await ShoutBehavior.SubmitNativeConversationTextForExternalAsync("first player input",stream,"original dialog",post,ready)=="body","public player producer reaches actual admission/worker");
  Check(entry.LastInput=="first player input"&&entry.LastDialog=="original dialog"&&!entry.LastOpening&&streams==1&&posts==1&&readyCalls==1&&ReferenceEquals(seenHero,entry.Hero)&&ReferenceEquals(seenCharacter,entry.Character),"public player facade preserves all callback and dialog identities");
  var presentation=ShoutBehavior.CaptureNativeConversationPresentationScopeForOverlay();Check(presentation!=null,"real overlay producer captures lease after prior completion");
  Check(await ShoutBehavior.SubmitNativeConversationForOverlayAsync(presentation,"second player input",stream,"overlay dialog",post,ready,false)=="body"&&entry.Runs==2&&streams==2&&posts==2&&readyCalls==2,"real overlay facade crosses admission and preserves second-turn callbacks");
  bool scopeReused=false;try{await ShoutBehavior.SubmitNativeConversationForOverlayAsync(presentation,"duplicate",stream,"",post,ready,false);}catch(ShoutBehavior.NativeConversationAdmissionException error){scopeReused=error.ReasonCode=="native.presentation_scope_reused";}
  Check(scopeReused&&entry.Runs==2,"same overlay producer lease cannot replay a turn");
  presentation=ShoutBehavior.CaptureNativeConversationPresentationScopeForOverlay();Mission.Current=new();Check(await ShoutBehavior.SubmitNativeConversationForOverlayAsync(presentation,"old scene",stream,"",post,ready,false)==""&&entry.Runs==2,"stale overlay facade rejected before callbacks and worker");
  foreach(string producerSource in new[]{"lord","companion"}){entry=new Fixture();entry.Open(producerSource);Check(await ShoutBehavior.SubmitNativeConversationNpcInitiatedOpeningForExternalAsync(stream,"opening dialog",post,ready)=="preprocess-unavailable"&&entry.Consumed==1,"public NPC producer crosses "+producerSource+" obligation owner");entry.Mode="ready";int beforeReady=readyCalls;Check(await ShoutBehavior.SubmitNativeConversationNpcInitiatedOpeningForExternalAsync(stream,"retry dialog",post,ready)=="body"&&entry.Consumed==1&&entry.LastOpening&&entry.LastDialog=="retry dialog"&&readyCalls==beforeReady+1&&!entry.HasRetry(),"public NPC retry preserves callback and consumes source exactly once "+producerSource);}

   }

   f=new Fixture();f.Open(source);await f.Submit();var late=f.LateReady;var oldAdmission=f.Last;f.Adapter.EndConversation();

  }
  Check(Logger.Entries.Exists(x=>x.Contains("opening_state stage=menu")&&x.Contains("no_matching_request")),"single menu diagnostic distinguishes no matching request");

 }
