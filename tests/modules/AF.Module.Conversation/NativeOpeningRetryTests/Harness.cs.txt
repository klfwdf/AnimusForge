internal static class Program {
 private static int Checks;
 private static void Check(bool value,string name){Checks++;if(!value)throw new InvalidOperationException(name);}
 private static async Task WaitFor(Func<bool> predicate){for(int i=0;i<1000&&!predicate();i++)await Task.Delay(1);Check(predicate(),"worker-started");}
 static async Task Main(){
  var entry=new Fixture{Mode="ready"};int streams=0,posts=0,readyCalls=0;Hero seenHero=null;CharacterObject seenCharacter=null;
  Action<string> stream=t=>{Check(t=="chunk","facade keeps stream callback text");streams++;};Action<string> post=t=>posts++;Action<string,Hero,CharacterObject> ready=(text,hero,character)=>{readyCalls++;seenHero=hero;seenCharacter=character;};
  Check(await ShoutBehavior.SubmitNativeConversationTextForExternalAsync("first player input",stream,"original dialog",post,ready)=="body","public player producer reaches actual admission/worker");
  Check(entry.LastInput=="first player input"&&entry.LastDialog=="original dialog"&&!entry.LastOpening&&streams==1&&posts==1&&readyCalls==1&&ReferenceEquals(seenHero,entry.Hero)&&ReferenceEquals(seenCharacter,entry.Character),"public player facade preserves all callback and dialog identities");
  var presentation=ShoutBehavior.CaptureNativeConversationPresentationScopeForOverlay();Check(presentation!=null,"real overlay producer captures lease after prior completion");
  Check(await ShoutBehavior.SubmitNativeConversationForOverlayAsync(presentation,"second player input",stream,"overlay dialog",post,ready,false)=="body"&&entry.Runs==2&&streams==2&&posts==2&&readyCalls==2,"real overlay facade crosses admission and preserves second-turn callbacks");
  bool scopeReused=false;try{await ShoutBehavior.SubmitNativeConversationForOverlayAsync(presentation,"duplicate",stream,"",post,ready,false);}catch(ShoutBehavior.NativeConversationAdmissionException error){scopeReused=error.ReasonCode=="native.presentation_scope_reused";}
  Check(scopeReused&&entry.Runs==2,"same overlay producer lease cannot replay a turn");
  presentation=ShoutBehavior.CaptureNativeConversationPresentationScopeForOverlay();Mission.Current=new();Check(await ShoutBehavior.SubmitNativeConversationForOverlayAsync(presentation,"old scene",stream,"",post,ready,false)==""&&entry.Runs==2,"stale overlay facade rejected before callbacks and worker");
  foreach(string producerSource in new[]{"lord","companion"}){entry=new Fixture();entry.Open(producerSource);Check(await ShoutBehavior.SubmitNativeConversationNpcInitiatedOpeningForExternalAsync(stream,"opening dialog",post,ready)=="preprocess-unavailable"&&entry.Consumed==1,"public NPC producer crosses "+producerSource+" obligation owner");entry.Mode="ready";int beforeReady=readyCalls;Check(await ShoutBehavior.SubmitNativeConversationNpcInitiatedOpeningForExternalAsync(stream,"retry dialog",post,ready)=="body"&&entry.Consumed==1&&entry.LastOpening&&entry.LastDialog=="retry dialog"&&readyCalls==beforeReady+1&&!entry.HasRetry(),"public NPC retry preserves callback and consumes source exactly once "+producerSource);}
  foreach(string source in new[]{"lord","companion"}) {
   var f=new Fixture();f.Open(source);
   Check(await f.Submit()=="preprocess-unavailable",source+" first attempt failed without body");
   Check(f.Consumed==1&&f.Runs==1&&f.HasRetry(),source+" first consumes once and keeps obligation");
   Check(f.Last.OpeningExtraFact==source+"-fact"&&f.Last.OpeningPrompt==source+"-prompt",source+" source content captured");
   f.LateReady("",f.Hero,f.Character);Check(f.HasRetry(),source+" empty presentation callback preserves obligation");
   f.Mode="ready";Check(await f.Submit()=="body",source+" manual retry body");
   Check(f.Consumed==1&&f.Runs==2&&f.Visible==1&&!f.HasRetry(),source+" retry no duplicate source consume, ready clears");
   Check(await f.Submit()==""&&f.Runs==2,source+" ready opening not replayed");

   f=new Fixture();f.Open(source);f.Mode="hold";Task<string> normal=f.Submit(false);await WaitFor(()=>f.Runs==1);
   bool normalBusy=false;try{await f.Submit();}catch(ShoutBehavior.NativeConversationAdmissionException error){normalBusy=error.ReasonCode=="native.busy";}
   Check(normalBusy&&f.Consumed==0,source+" normal request busy keeps unconsumed NPC opening");
   f.Hold.SetResult(true);await normal;f.Mode="ready";Check(await f.Submit()=="body"&&f.Consumed==1,source+" opening remains after normal busy rejection");

   f=new Fixture();f.Open(source);f.Mode="hold";Task<string> first=f.Submit();await WaitFor(()=>f.Runs==1);
   bool busy=false;try{await f.Submit();}catch(ShoutBehavior.NativeConversationAdmissionException error){busy=error.ReasonCode=="native.busy";}
   Check(busy&&f.Consumed==1,source+" busy refused before source consumption");
   f.Hold.SetResult(true);await first;Check(!f.HasRetry(),source+" held ready clears");

   foreach(string drift in new[]{"token","hero","character","mission","manager","generation","epoch","owner"}) {
    f=new Fixture();f.Open(source);await f.Submit();Check(f.HasRetry(),source+" pre-drift retry exists");
    switch(drift){
     case "token":Campaign.Current.ConversationManager.ActiveToken++;break;
     case "hero":f.Hero=new(){StringId="replacement"};break;
     case "character":f.Character=new();break;
     case "mission":Mission.Current=new();break;
     case "manager":Campaign.Current.ConversationManager=new(){IsConversationInProgress=true,ActiveToken=1};break;
     case "generation":SaveRuntimeGuard.Generation++;break;
     case "epoch":f.Adapter.EndConversation();break;
     case "owner":f.Current=false;break;
    }
    f.Mode="ready";Check(await f.Submit()==""&&f.Runs==1&&!f.HasRetry(),source+" rejects retry after "+drift);
   }

   f=new Fixture();f.Open(source);await f.Submit();var late=f.LateReady;var oldAdmission=f.Last;f.Adapter.EndConversation();
   f.Open(source);await f.Submit();Check(f.HasRetry(),source+" new conversation holds own retry");
   late("late",f.Hero,f.Character);Check(f.HasRetry(),source+" old ready cannot clear new obligation");
   f.Adapter.RetireOpeningForAcceptedReply(oldAdmission);Check(f.HasRetry(),source+" old effect/exception retirement cannot clear new obligation");
   Check(f.Adapter.CaptureNativeConversationAdmissionOnMainThread(oldAdmission.Generation,oldAdmission.ConversationEpoch,true)==null&&f.HasRetry(),source+" queued old admission cannot clear new obligation");
   f.Mode="ready";Check(await f.Submit()=="body"&&f.Consumed==2,source+" replacement independently completes");

   f=new Fixture();f.Open(source);await f.Submit();f.Adapter.RetireOpeningForAcceptedReply(f.Last);
   Check(!f.HasRetry(),source+" accepted raw reply retires before effects even if visible text is empty");
   Check(await f.Submit()==""&&f.Runs==1,source+" raw-action acceptance cannot replay opening");

   f=new Fixture();f.Open(source);f.Available=false;Check(await f.Submit()==""&&f.Consumed==0,source+" initial target rejected before consume");
   f.Available=true;f.Mode="ready";Check(await f.Submit()=="body"&&f.Consumed==1,source+" target recovery first body");
  }
  Console.WriteLine("PASS NativeOpeningRetry checks="+Checks+"; current public/overlay producers + admission adapter/owner/router + production lord/companion consume statements; game/UI/full turn/network leaves are fixtures, no live opening acceptance");
 }
