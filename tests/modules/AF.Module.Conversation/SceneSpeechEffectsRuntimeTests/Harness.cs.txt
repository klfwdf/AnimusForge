        static void Dispatch(SceneSpeechEffectController owner,SceneSpeechQueueItem item,bool duplicate=false)
        {
            var posts=new ConcurrentQueue<Action>();
            var runtime=new SceneSpeechExecutionRuntime<SceneSpeechQueueItem>(()=>false,action=>{posts.Enqueue(action);if(duplicate)posts.Enqueue(action);},owner.Publish,p=>p.CompletionSource.TrySetResult(false),ms=>Task.CompletedTask);
            runtime.Enqueue(item);var watch=Stopwatch.StartNew();
            while(!item.CompletionSource.Task.IsCompleted || !posts.IsEmpty)
            { while(posts.TryDequeue(out var action))action();if(watch.ElapsedMilliseconds>3000)throw new Exception("fixture_dispatch_timeout");Thread.Sleep(1); }
