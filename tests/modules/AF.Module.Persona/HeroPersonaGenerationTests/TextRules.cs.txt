static void TextRuleCases()
 {
  Check(AnimusForge.NpcPersonaTextRules.NormalizePersonaPromptSourceText("  A\r\n B   C ",4)=="A B", "text source whitespace and bound");
  Check(AnimusForge.NpcPersonaTextRules.NormalizePersonaPromptSourceText(" abc ",0)=="abc", "text source zero unbounded");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("```json\n{\"PERSONALITY\":\"稳重\",\"Background\":\"经历\"}\n```",out var p,out var b)&&p=="稳重\n\n经历"&&b=="", "real persona fenced case-insensitive JSON");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("prefix {\"profile\":\"备用\"} suffix",out p,out b)&&p=="备用"&&b=="", "real persona aliases and partial field");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"personality\":\"unterminated",out p,out b)&&p=="unterminated", "real persona truncated loose fallback");
  Check(!AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("unusable",out p,out b)&&p==""&&b=="", "real persona malformed rejects");
  Check(!AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"traits\":[{\"id\":\"bold\",\"strength\":2}],\"speech_style\":\"brief\"}",out p,out b), "structured data alone is not success");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"profile\":\"Unified 中英\",\"personality\":\"old\",\"background\":\"ignored\"}",out p,out b)&&p=="Unified 中英"&&b=="", "new protocol wins");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"profile\":{\"personality\":\"  节俭  \",\"background\":\"守城经历\"},\"traits\":[{\"id\":\"frugal\",\"strength\":2}],\"motivations\":[\"守住家业\"],\"values\":[\"务实\"],\"fears\":[\"失地\"],\"speech_style\":\"直接\"}",out AnimusForge.NpcPersonaGeneratedData nested)
   &&nested.Description=="节俭\n\n守城经历"&&nested.Traits.Count==1&&nested.Traits[0].Id=="frugal"&&nested.Traits[0].Strength==2&&nested.Motivations[0]=="守住家业"&&nested.Values[0]=="务实"&&nested.Fears[0]=="失地"&&nested.SpeechStyle=="直接", "nested profile regression retains body and expanded fields");
  foreach(var pair in new[]{
   ("{\"PROFILE\":{\"PERSONALITY\":\"稳重\",\"Background\":\"经历\"}}", "稳重\n\n经历"),
   ("{\"profile\":{\"background\":\"经历\"}}", "经历"),
   ("{\"profile\":{\"personality\":\"稳重\"},\"background\":\"unrelated\"}", "稳重"),
   ("{\"description\":\"canonical\",\"profile\":{\"personality\":\"ignored\"}}", "canonical"),
   ("{\"profile\":{},\"personality\":\"legacy\",\"background\":\"body\"}", "legacy\n\nbody")})
   Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson(pair.Item1,out p,out b)&&p==pair.Item2&&b=="", "nested profile precedence and compatibility: "+pair.Item1);
  foreach(var invalid in new[]{"{\"profile\":{\"personality\":[],\"background\":42}}", "{\"profile\":{\"personality\":\" \",\"background\":null}}", "{\"profile\":[{\"personality\":\"invalid\"}]}", "{\"profile\":{\"personality\":{\"text\":\"invalid\"}}}"})
   Check(!AnimusForge.NpcPersonaTextRules.TryParsePersonaJson(invalid,out p,out b), "nested profile requires known string body: "+invalid);
  string replayDir=Environment.GetEnvironmentVariable("AF_PERSONA_RESPONSE_FIXTURES");
  if(!string.IsNullOrEmpty(replayDir))
   foreach(string file in System.IO.Directory.GetFiles(replayDir,"*.json"))
   {
    string raw=System.IO.File.ReadAllText(file);
    var wire=Newtonsoft.Json.Linq.JObject.Parse(raw);
    string expected=AnimusForge.Refactor.Runtime.NpcPersonaProfilePolicy.MergeLegacy((string)wire["profile"]["personality"],(string)wire["profile"]["background"]);
    Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson(raw,out AnimusForge.NpcPersonaGeneratedData replay)&&replay.Description==expected&&replay.Traits.Count>0&&replay.Motivations.Count>0&&replay.SpeechStyle==(string)wire["speech_style"],"actual log response: "+System.IO.Path.GetFileName(file));
   }
  // Claude replies: "persona" alias of profile, extra "name", speech_style as an object, leading "json" language line.
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("json\n{\"name\":\"加里俄斯\",\"persona\":\"他是军营里走出来的统治者。\",\"traits\":[{\"id\":\"ambitious\",\"strength\":3}],\"motivations\":[\"统一帝国\"],\"values\":[\"赏赐必须到手\"],\"fears\":[\"失去军心\"],\"speech_style\":{\"tone\":\"豪爽、亲近\",\"habits\":[\"常提士兵\"],\"sample_lines\":[\"别拦着。\"]}}",out AnimusForge.NpcPersonaGeneratedData claude)
   &&claude.Description=="他是军营里走出来的统治者。"&&claude.Traits.Count==1&&claude.Traits[0].Id=="ambitious"&&claude.Motivations[0]=="统一帝国"&&claude.SpeechStyle=="豪爽、亲近", "persona alias, extra name and object speech_style");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"profile\":\"canonical\",\"persona\":\"ignored alias\",\"speech_style\":\"plain\"}",out claude)&&claude.Description=="canonical"&&claude.SpeechStyle=="plain", "profile wins over persona alias and string speech_style is unchanged");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"profile\":\"x\",\"speech_style\":{\"tone\":\"calm\",\"style\":\"short sentences\"}}",out claude)&&claude.SpeechStyle=="calm short sentences", "object speech_style joins tone and style");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"profile\":\"x\",\"speech_style\":{\"habits\":[\"a\"]}}",out claude)&&claude.SpeechStyle=="", "object speech_style without named strings is dropped, not serialized");
  Check(AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"profile\":\"x\",\"speech_style\":[\"a\",\"b\"]}",out claude)&&claude.SpeechStyle=="", "array speech_style is dropped");
  Check(!AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"persona\":{\"text\":\"object body\"},\"name\":\"n\"}",out p,out b), "persona alias still requires a string body");
  Check(!AnimusForge.NpcPersonaTextRules.TryParsePersonaJson("{\"persona\":\"  \",\"traits\":[{\"id\":\"bold\",\"strength\":1}]}",out p,out b), "blank persona alias is not success");
  foreach(var invalid in new[]{"{\"profile\":42}","{\"profile\":{\"x\":\"wrong\"}}","{\"traits\":[{\"profile\":\"nested-only\"}]}"})
   Check(!AnimusForge.NpcPersonaTextRules.TryParsePersonaJson(invalid,out p,out b),"unknown body shape rejects: "+invalid);
  var edited=new AnimusForge.MyBehavior.NpcPersonaProfile {Motivations=new(){"a\u001fb"}};
  var work=new AnimusForge.MyBehavior.NpcPersonaGenerationWork {OriginalMotivations=new(){"a","b"},OriginalSpeechStyle=""};
  Check(!AnimusForge.Refactor.Adapters.NpcPersonaGenerationApplicationAdapter.ProfileExtrasStillCurrent(edited,work),"structured field separator collision cannot bypass edit protection");
  edited.Motivations=new(){"a "};work.OriginalMotivations=new(){"a"};
  Check(!AnimusForge.Refactor.Adapters.NpcPersonaGenerationApplicationAdapter.ProfileExtrasStillCurrent(edited,work),"structured text edits use exact comparison");
  var native=AnimusForge.NpcPersonaTextRules.BuildNative("detached-facts", "old\r\nP\n\noldB",true," 长篇\r\n要求 ");
  Check(native.System.EndsWith("【玩家自定义生成要求】\n长篇\n要求")&&native.User.Contains("detached-facts")&&native.User.Contains("old P")&&native.User.Contains("oldB")&&!native.User.Contains("旧背景"),"real native prompt detached and reroll requirements");
  var normal=AnimusForge.NpcPersonaTextRules.BuildNative("facts","old\n\noldB",false,"");
  Check(!normal.User.Contains("旧个性")&&!normal.System.Contains("【玩家自定义生成要求】"),"real native ordinary prompt unchanged");
  var promoted=AnimusForge.NpcPersonaTextRules.BuildPromoted("name","full","troop","id","culture","scene","join","history","equipment","facts","custom");
  Check(promoted.System.EndsWith("custom")&&promoted.User.Contains("history")&&promoted.User.Contains("facts")&&promoted.User.Contains("不要把玩家家族身份写成原生出身"),"real promoted facts and requirements");
  var barePromoted=AnimusForge.NpcPersonaTextRules.BuildPromoted("n","f","t","id","c","s","join","history","eq","facts","");
  foreach(var bare in new[]{normal,barePromoted,AnimusForge.NpcPersonaTextRules.BuildNative("facts","oldP\n\noldB",true,"")})
  {
   foreach(var hidden in new[]{"300","完整句子","必须综合","不要复制","文风","写作风格","不要照搬","不同但","谨慎而合理的动机","背景必须解释"})
    Check(!(bare.System+bare.User).Contains(hidden),"no hidden writing requirement: "+hidden);
   Check(bare.System.Contains("严格 JSON")&&bare.System.Contains("事实"),"protocol and facts remain built-in");
   Check(bare.System.Contains("profile 必须是非空字符串")&&bare.System.Contains("不得是对象或数组"),"actual persona request specifies a string profile");
  }
  Check(AnimusForge.NpcPersonaTextRules.BuildFallbackPersonality(new string('n',600),"t","j","f","s","h").Length==420,"promoted personality fallback 420");
  Check(AnimusForge.NpcPersonaTextRules.BuildFallbackBackground("n","t","j","f","s",new string('h',400)).EndsWith(new string('h',240)),"promoted fallback history 240");
  Check(AnimusForge.NpcPersonaTextRules.BuildFallbackBackground("n","t",new string('j',600),"f","s","h").Length==520,"promoted background fallback 520");
  var known=new[]{"onehanded","bow"};
  Check(AnimusForge.NpcPersonaTextRules.TryParseSkills("```json\n{\"Skills\":{\"One-Handed\":400,\"Bow\":-2,\"Unknown\":100,\"bad\":\"oops\"}}\n```",known,out var skills)&&skills.Count==2&&skills[0].Key=="onehanded"&&skills[0].Value==330&&skills[1].Value==0,"real skill normalization clamp and whitelist");
  Check(AnimusForge.NpcPersonaTextRules.TryParseSkills("{\"OneHanded\":10,\"One-Handed\":20}",known,out skills)&&skills.Count==2&&skills[0].Value==10&&skills[1].Value==20,"real skill alias execution order preserved");
  Check(!AnimusForge.NpcPersonaTextRules.TryParseSkills("{\"skills\":{\"Bow\":1.5,\"Unknown\":1}}",known,out skills)&&skills.Count==0,"real skills reject fraction and unknown");
  Check(!AnimusForge.NpcPersonaTextRules.TryParseSkills("not JSON",known,out skills)&&skills.Count==0,"real skills parse failure keeps empty effects");

  foreach(var pair in new[]{("P", "B", "P\n\nB"), ("P", "", "P"), ("", "B", "B"), (" A\r\n中 ", "A\n中", "A\n中"), ("", "", ""), (new string('中',40000), new string('x',40000),new string('中',40000)+"\n\n"+new string('x',40000))})
  {
   string json=Newtonsoft.Json.JsonConvert.SerializeObject(new {HeroId="hero",HeroName="原名",Personality=pair.Item1,Background=pair.Item2,VoiceId="voice",Traits=new[]{new {Id="bold",Strength=2}},Motivations=new[]{"原文"},Values=new[]{"value"},Fears=new[]{"fear"},SpeechStyle="style"});
   var value=Newtonsoft.Json.JsonConvert.DeserializeObject<AnimusForge.MyBehavior.NpcPersonaProfile>(json);
   Check(value.Description==pair.Item3&&value.SchemaVersion==3,"actual Newtonsoft old profile migration "+pair.Item3.Length);
   for(int cycle=0;cycle<3;cycle++)value=Newtonsoft.Json.JsonConvert.DeserializeObject<AnimusForge.MyBehavior.NpcPersonaProfile>(Newtonsoft.Json.JsonConvert.SerializeObject(value));
   Check(value.Description==pair.Item3&&value.Personality==pair.Item1&&value.Background==pair.Item2,"lossless idempotent reload preserves compatibility fields");
   Check(value.VoiceId=="voice"&&value.HeroId=="hero"&&value.HeroName=="原名"&&value.Traits[0].Id=="bold"&&value.Motivations[0]=="原文"&&value.Values[0]=="value"&&value.Fears[0]=="fear"&&value.SpeechStyle=="style","roundtrip identity voice structured fields");
   value.Description="";value.NormalizeDescription();
   value=Newtonsoft.Json.JsonConvert.DeserializeObject<AnimusForge.MyBehavior.NpcPersonaProfile>(Newtonsoft.Json.JsonConvert.SerializeObject(value));
   Check(value.Description=="","version 3 clear cannot revive old fields");
  }
  var profiles = new AnimusForge.PersonaProfileStateOwner();
  profiles.Save("hero", new(){SchemaVersion=3,Description="",Personality="legacy",HeroName="identity"},(_,_)=>{});
  Check(profiles.Get("hero",false)?.Description==""&&profiles.Get("hero",false)?.Personality=="legacy", "normal save retains explicit v3 empty profile and compatibility data");
  profiles.Save("hero",new(),(_,_)=>{});
  Check(profiles.Get("hero",false)==null,"explicit clear-all still removes the whole entry");
  var clearProfile = new AnimusForge.MyBehavior.NpcPersonaProfile {Description="",SchemaVersion=3,Personality="old",VoiceId="voice"};
  AnimusForge.PersonaImportOwner.ApplyDeveloperText(clearProfile,"edited",p=>profiles.Save("hero",p,(_,_)=>{}));
  Check(profiles.GetDescription("hero")=="edited"&&clearProfile.Personality=="old", "editor writes canonical body without rewriting legacy data");
 }
