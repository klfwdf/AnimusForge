 internal sealed class ExperienceHistorySource {
  internal readonly Dictionary<string,List<AnimusForgeDialogueHistoryEntry>> History=new();

  internal int Reads;
  internal bool TryGetLatestCompressedMemoryInternal(Hero hero,out string key,out string date,out string text,out int day){
   Probe.Read();Reads++;key="";date="date-90";day=90;return Summaries.TryGetValue(hero.StringId,out text);
  }
  internal List<AnimusForgeDialogueHistoryEntry> GetDialogueHistoryEntriesById(string id,int limit){
   Probe.Read();Reads++;
   if(!History.TryGetValue(id,out var entries))return new();
   return entries.GetRange(Math.Max(0,entries.Count-limit),Math.Min(limit,entries.Count));
  }
 }
