internal static class ColdPromptCases {
 internal static void Run(Action<bool,string> check){
  KnowledgeLibraryBehavior.Instance=new(){_file=new(){Rules=new(){new(){Id="old",Keywords=new(){"NPC"}}}}};
  var mentions=new MentionedWorldEntities();mentions.Entities.Add("NPC");
  var ports=KnowledgeLibraryBehavior.Ports;ports.MainThread=Environment.CurrentManagedThreadId;
  long v=KnowledgeLibraryBehavior.PreparePromptLoreRetrieval(mentions);
  check(v>0 && ports.MainCalls==0 && ports.Calls==0,"actual game-thread preparation performs zero embeddings");
  KnowledgeLibraryBehavior.Instance._file.Rules[0].Keywords[0]="edited live text";
  ports.Block=true;
  var pending=Task.Run(()=>KnowledgeLibraryBehavior.CollectPromptLoreCandidates(mentions,v,new()));
  try {
   check(ports.Entered.Wait(3000),"cold prompt builds on worker");
   var invalidate=Task.Run(()=>KnowledgeLibraryBehavior.Index.Touch());
   check(invalidate.Wait(1000),"rule invalidation never waits for gated embedding work");
   KnowledgeLibraryBehavior.Instance._file.Rules=new(){new(){Id="new",Keywords=new(){"NPC"}}};
   long next=KnowledgeLibraryBehavior.PreparePromptLoreRetrieval(mentions);
   check(next!=v && ports.MainCalls==0,"new detached publication needs no inference while old worker blocked");
   ports.Block=false;ports.Continue.Set();
   check(pending.Wait(3000)&&pending.Result==null,"stale worker returns no old lore");
   check(!KnowledgeLibraryBehavior.Index.OnnxReady && !KnowledgeLibraryBehavior.Index.SparseReady,"stale build cannot publish either index for new version");
   check(!ports.Inputs.Any(x=>x.Contains("edited live text")),"worker index reads detached published rule text");
   var current=Task.Run(()=>KnowledgeLibraryBehavior.CollectPromptLoreCandidates(mentions,next,new()));
   check(current.Wait(3000)&&current.Result.OrderedRules.All(x=>x.Id=="new")&&current.Result.OrderedRules.Count==1,"new version retrieves complete current rule");
   int calls=ports.Calls;
   KnowledgeLibraryBehavior.PreparePromptLoreRetrieval(mentions);
   check(ports.Calls==calls && ports.MainCalls==0,"warm game-thread prepare performs no inference");
   var disabled=Task.Run(()=>KnowledgeLibraryBehavior.CollectPromptLoreCandidates(mentions,next,new(){Enabled=false}));
   check(disabled.Wait(3000)&&ports.Calls==calls,"disabled retrieval does not build or query embeddings");
  } finally {ports.Block=false;ports.Continue.Set();}
 }
}
