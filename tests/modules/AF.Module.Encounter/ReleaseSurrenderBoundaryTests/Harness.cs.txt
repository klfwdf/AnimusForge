    static PlayerEncounter Reset(bool mission=false)
    {
        ClearPendingNativeConversationNpcSurrender("test"); _releaseOwner.ClearAuthorization();
        EffectParty=null; FinishedEncounter=null; Effects=0; SurrenderAttempts=0; SurrenderSucceeds=true;
        PlayerEncounter.Current=new(){Party=new()}; PlayerEncounter.PlayerSurrender=false; PlayerEncounter.LeaveEncounter=false;
        Campaign.Current=new(); Mission.Current=mission ? new() : null;
        Game.Current=new(); if(mission) Game.Current.GameStateManager.ActiveState=new MissionState();
        SaveRuntimeGuard.Generation=1; Time.ApplicationTime=100;
        return PlayerEncounter.Current;
    }
    internal static void Verify()
    {
        foreach(string change in new[]{"encounter","party","generation","manager","mission","missing","mission_removed"})
        {
            var original=Reset(change=="mission_removed"); var originalMission=Mission.Current;
            Campaign.Current.ConversationManager.OnEnd=()=>Change(change);
            bool released=TryExecuteMeetingPlayerRelease(new Hero(),original.Party,"test");
            Check(!released && FinishedEncounter==null && _releaseOwner.Authorization==null,"callback rejects "+change);
            Check((Mission.Current?.EndCount??0)==0 && (originalMission?.EndCount??0)==0,"callback cannot close changed mission "+change);
        }
        foreach(bool mission in new[]{false,true})
        {
            var original=Reset(mission);
            Check(TryExecuteMeetingPlayerRelease(new Hero(),original.Party,"normal"),"unchanged release succeeds");
            Check(mission ? Mission.Current.EndCount==1 : ReferenceEquals(FinishedEncounter,original),"only captured scope released");
        }
        {
            Reset(); var manager=Campaign.Current.ConversationManager;
            Check(!TryExecuteMeetingPlayerRelease(new Hero(),new PartyBase(),"mismatch") && manager.EndCount==0,"wrong party rejected before ending conversation");
        }
        foreach(string change in new[]{"encounter","party","generation","manager","mission","missing","expiry"})
        {
            Reset(); MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"test");
            Change(change); if(change=="expiry") Time.ApplicationTime=701;
            TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==0 && !_pendingNativeConversationNpcSurrender,"stale surrender cancelled: "+change);
            if(change=="missing") Check(PlayerEncounter.Current==null,"missing encounter is not resurrected");
        }
        {
            var original=Reset(); Campaign.Current.ConversationManager.IsConversationInProgress=true;
            MarkPendingNativeConversationNpcSurrender(null,null,-1,"nonhero");
            Time.ApplicationTime=101; TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==0,"surrender waits for manual dialogue exit");
            Campaign.Current.ConversationManager.IsConversationInProgress=false;
            TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==1 && ReferenceEquals(EffectParty,original.Party),"normal nonhero surrender uses original party");
            TryForcePendingNativeConversationNpcSurrenderIfReady(); Check(Effects==1,"surrender consumed once");
        }
        {
            var original=Reset(true); MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"mission");
            Time.ApplicationTime=101; TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==0,"mission surrender waits for map");
            Mission.Current=null; TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==0 && _pendingNativeConversationNpcSurrender,"mission teardown retains pending task");
            Game.Current.GameStateManager.ActiveState=new(); TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==1 && ReferenceEquals(EffectParty,original.Party),"same encounter survives mission teardown");
        }
        {
            Reset(); MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"old");
            ClearPendingNativeConversationNpcSurrender("session_launched");
            Time.ApplicationTime=101; TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==0,"session reset revokes pending surrender");
        }
        // Reproduce the paused encounter menu: only application/engine ticks run.
        // Native farewell sets LeaveEncounter before EndConversation. The native
        // map teardown may Update/Finish before the next application tick.
        {
            var original=Reset(); var manager=Campaign.Current.ConversationManager;
            manager.IsConversationInProgress=true;
            MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"native_farewell");
            manager.OnEnd=()=> { if(PlayerEncounter.LeaveEncounter) PlayerEncounter.Current=null; };
            PlayerEncounter.LeaveEncounter=true;
            manager.EndConversation();
            Check(ReferenceEquals(PlayerEncounter.Current,original) && !PlayerEncounter.LeaveEncounter,
                "native farewell cannot finish accepted surrender before engine tick");
            Check(SurrenderAttempts==0,"conversation-end hook waits for native teardown before settlement");
            Time.ApplicationTime=101; OnEngineTick(); OnEngineTick();
            Check(Effects==1 && ReferenceEquals(EffectParty,original.Party),"native farewell settles original encounter exactly once");
        }
        foreach(string change in new[]{"encounter","party","generation","manager","mission","missing"})
        {
            Reset(); var manager=Campaign.Current.ConversationManager;
            MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"stale_end");
            Change(change); PlayerEncounter.LeaveEncounter=true; manager.EndConversation();
            Check(PlayerEncounter.LeaveEncounter && Effects==0,"stale conversation end preserves native leave: "+change);
        }
        {
            Reset(); var manager=Campaign.Current.ConversationManager;
            MarkPendingNativeConversationNpcSurrender(null,null,-1,"cancelled");
            ClearPendingNativeConversationNpcSurrender("session_launched");
            PlayerEncounter.LeaveEncounter=true; manager.EndConversation();
            Check(PlayerEncounter.LeaveEncounter && manager.EndSubscribers==0,"cancelled surrender removes end hook and preserves native leave");
        }
        {
            Reset(); var oldManager=Campaign.Current.ConversationManager;
            MarkPendingNativeConversationNpcSurrender(null,null,-1,"old_manager");
            Campaign.Current.ConversationManager=new();
            MarkPendingNativeConversationNpcSurrender(null,null,-1,"replacement");
            PlayerEncounter.LeaveEncounter=true; oldManager.EndConversation();
            Check(PlayerEncounter.LeaveEncounter && oldManager.EndSubscribers==0,"replacement pending surrender detaches old manager");
            Campaign.Current.ConversationManager.EndConversation();
            Check(!PlayerEncounter.LeaveEncounter,"replacement manager owns current surrender end hook");
        }
        {
            Reset(); var manager=Campaign.Current.ConversationManager;
            MarkPendingNativeConversationNpcSurrender(null,null,-1,"repeated");
            MarkPendingNativeConversationNpcSurrender(null,null,-1,"repeated");
            Check(manager.EndSubscribers==1,"repeated surrender tags register one end hook");
        }
        {
            Reset(); var manager=Campaign.Current.ConversationManager;
            MarkPendingNativeConversationNpcSurrender(null,null,-1,"player_surrender");
            PlayerEncounter.PlayerSurrender=true; PlayerEncounter.LeaveEncounter=true;
            manager.EndConversation();
            Check(PlayerEncounter.LeaveEncounter && Effects==0,"NPC surrender end hook cannot override player surrender");
        }
        {
            Reset(); var manager=Campaign.Current.ConversationManager;
            PlayerEncounter.LeaveEncounter=true; manager.EndConversation();
            Check(PlayerEncounter.LeaveEncounter && manager.EndSubscribers==0,"ordinary farewell remains native without pending surrender");
        }
        // No Campaign Tick or direct calls to the pending consumer in these cases.
        {
            var original=Reset(); var manager=Campaign.Current.ConversationManager;
            manager.IsConversationInProgress=true;
            Check(MarkPendingNativeConversationNpcSurrender(null,new CharacterObject(),-1,"paused_nonhero"),"paused nonhero surrender queues");
            Time.ApplicationTime=101; OnEngineTick();
            Check(SurrenderAttempts==0 && _pendingNativeConversationNpcSurrender,"engine tick waits for manual conversation exit");
            manager.IsConversationInProgress=false; manager.IsConversationFlowActive=true; OnEngineTick();
            Check(SurrenderAttempts==0,"engine tick waits for remaining conversation flow");
            manager.IsConversationFlowActive=false; OnEngineTick();
            Check(Effects==1 && ReferenceEquals(EffectParty,original.Party) && !_pendingNativeConversationNpcSurrender,"engine tick alone consumes paused encounter surrender");
            OnEngineTick(); TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(Effects==1 && SurrenderAttempts==1,"engine then campaign consumer cannot duplicate surrender");
            Check(manager.EndCount==0,"pending consumer preserves manual conversation close");
        }
        foreach(string change in new[]{"encounter","party","generation","manager","mission","missing","expiry"})
        {
            Reset(); MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"engine_stale");
            Change(change); Time.ApplicationTime=change=="expiry" ? 701 : 101;
            OnEngineTick();
            Check(SurrenderAttempts==0 && !_pendingNativeConversationNpcSurrender,"engine tick cancels stale surrender: "+change);
        }
        {
            var original=Reset(true); MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"engine_mission");
            Time.ApplicationTime=101; OnEngineTick();
            Check(SurrenderAttempts==0 && _pendingNativeConversationNpcSurrender,"engine tick waits for mission state");
            Mission.Current=null; OnEngineTick();
            Check(SurrenderAttempts==0 && _pendingNativeConversationNpcSurrender,"engine tick waits through mission teardown");
            Game.Current.GameStateManager.ActiveState=new(); OnEngineTick();
            Check(Effects==1 && ReferenceEquals(EffectParty,original.Party),"engine tick executes on map after same encounter mission");
        }
        {
            Reset(); MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"engine_throttle");
            Time.ApplicationTime=100.125f; OnEngineTick();
            Check(SurrenderAttempts==0,"engine tick preserves initial 250ms delay");
            SurrenderSucceeds=false; Time.ApplicationTime=100.25f; OnEngineTick();
            Check(SurrenderAttempts==1 && Effects==0 && _pendingNativeConversationNpcSurrender,"failed surrender stays pending");
            Time.ApplicationTime=100.375f; OnEngineTick(); TryForcePendingNativeConversationNpcSurrenderIfReady();
            Check(SurrenderAttempts==1,"engine and campaign share 250ms retry throttle");
            SurrenderSucceeds=true; Time.ApplicationTime=100.5f; OnEngineTick();
            Check(Effects==1 && SurrenderAttempts==2 && !_pendingNativeConversationNpcSurrender,"engine retry completes once after throttle");
        }
        {
            Reset(); OnEngineTick(); Check(SurrenderAttempts==0,"idle engine tick does no surrender work");
            MarkPendingNativeConversationNpcSurrender(new Hero(),null,-1,"engine_reset");
            ClearPendingNativeConversationNpcSurrender("session_launched"); Time.ApplicationTime=101; OnEngineTick();
            Check(SurrenderAttempts==0,"engine tick cannot revive session-reset surrender");
        }
        Console.WriteLine($"PASS {Checks} release, surrender scope and engine-only tick checks (actual extracted host methods; native effects are fixtures).");
    }
    static void Change(string change)
    {
        switch(change)
        {
            case "encounter": PlayerEncounter.Current=new(){Party=PlayerEncounter.Current.Party}; break;
            case "party": PlayerEncounter.Current.Party=new(); break;
            case "generation": SaveRuntimeGuard.Generation++; break;
            case "manager": Campaign.Current.ConversationManager=new(); break;
            case "mission": Mission.Current=new(); break;
            case "missing": PlayerEncounter.Current=null; break;
            case "mission_removed": Mission.Current=null; break;
        }
    }
    static bool TryExecuteNpcSurrenderFromFreeConversation(Hero hero,CharacterObject c,int index,string reason,bool closeConversation)
    {
        SurrenderAttempts++;
        if(!SurrenderSucceeds) return false;
        if(!TryGetNpcSurrenderEncounterParty(hero,c,out var party,out _)) return false;
        EffectParty=party; Effects++; return true;
    }
    static PartyBase TryGetMeetingReleaseEncounterParty() => PlayerEncounter.Current?.Party;
    static bool TryGetMeetingReleaseContext(Hero hero,out Hero target,out int a,out int b,out int c,out int d,out string e,out bool negotiable)
    { target=hero; a=b=c=d=0; e=""; negotiable=true; return true; }
    static bool IsMissionStateActiveForMeetingRelease() => Mission.Current!=null;
    static void AuthorizeMeetingPlayerRelease(string reason) => _releaseOwner.Authorize(CaptureMeetingPlayerReleaseRequest(_targetHero,reason));
    static void ClearPendingReturnToEncounterMenuAfterUnauthorizedMeetingExit(string reason) {}
    static bool TryEndCurrentMeetingMissionByReflection(string reason) { Mission.Current.EndMission(); return true; }
