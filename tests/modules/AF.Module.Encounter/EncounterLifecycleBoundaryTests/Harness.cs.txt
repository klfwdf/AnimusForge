internal static class Program
{
    private static int _assertions;
    public static void Check(bool condition, string name)
    {
        if (!condition) throw new InvalidOperationException(name);
        _assertions++;
    }
    public static void Main(string[] args)
    {
        if (args.Length == 1) {
            if (args[0] == "hard") HardSuppressionAudit.Run();
            else if (args[0] == "raid") RaidContextAudit.Run();
            else throw new ArgumentException("Unknown audit");
            Console.WriteLine("PASS: " + _assertions + " extracted production boundary assertions.");
            return;
        }
        ReleaseAudit.Run();
        TargetAudit.Run();
        ConversationTargetAudit.Run();
        ReturnAudit.Run();
        NativeHandoffAudit.Run();
        PlayerLeaveAudit.Run();
        HardSuppressionAudit.Run();
        RaidContextAudit.Run();
        DuelAudit.Run();
        InteractionComponentSafePatch.Run();
        Console.WriteLine("PASS: " + _assertions + " extracted production boundary assertions.");
    }
}

internal static class RaidContextAudit
{
    private enum AiBehavior { EngageParty, RaidSettlement }
    private sealed class Settlement { public bool IsVillage = true, IsUnderRaid; public MobileParty LastAttackerParty; }
    private sealed class MapEvent {
        public Settlement MapEventSettlement; public bool IsRaid, IsForcingSupplies, IsForcingVolunteers;
        public bool WasEverInLootingPhase { get; set; }
    }
    private sealed class PartyBase { public bool IsMobile, IsSettlement; public MobileParty MobileParty; public Settlement Settlement; public MapEvent MapEvent; }
    private sealed class MobileParty {
        public MapEvent MapEvent; public PartyBase Party = new PartyBase();
        public AiBehavior DefaultBehavior, ShortTermBehavior;
        public Settlement TargetSettlement, ShortTermTargetSettlement, CurrentSettlement;
    }
    // Needed only when replaying the historical source for the negative control.
    private static bool IsVillageSettlement(Settlement settlement) => settlement != null && settlement.IsVillage;
    @@RAID_CONTEXT_METHODS@@
    internal static void Run() {
        var village = new Settlement();
        var party = new MobileParty { DefaultBehavior = AiBehavior.RaidSettlement, ShortTermBehavior = AiBehavior.EngageParty, TargetSettlement = village };
        Program.Check(!IsVillageRaidMobileParty(party), "Talos raid order plus actual chase is not a village activity encounter");
        var wrapper = new PartyBase { IsMobile = true, MobileParty = party };
        Program.Check(!IsVillageRaidParty(wrapper), "encounter PartyBase does not reintroduce AI-intent exclusion");
        village.IsUnderRaid = true; village.LastAttackerParty = party;
        Program.Check(!IsVillageRaidMobileParty(party), "old target LastAttacker does not turn remote interception into a raid");
        party.ShortTermBehavior = AiBehavior.RaidSettlement; party.ShortTermTargetSettlement = village;
        Program.Check(!IsVillageRaidMobileParty(party), "short-term raid intention alone is not an active event");
        party.CurrentSettlement = village;
        Program.Check(IsVillageRaidMobileParty(party), "actual current raided village still preserves native activity");
        village.LastAttackerParty = new MobileParty();
        Program.Check(!IsVillageRaidMobileParty(party), "another party's raid does not make this party the raider");
        party.CurrentSettlement = null;
        foreach (var kind in new[] { "raid", "supplies", "volunteers", "looting" }) {
            var map = new MapEvent { MapEventSettlement = village, IsRaid = kind == "raid", IsForcingSupplies = kind == "supplies", IsForcingVolunteers = kind == "volunteers", WasEverInLootingPhase = kind == "looting" };
            party.MapEvent = map;
            Program.Check(IsVillageRaidMobileParty(party), "actual village event is retained: " + kind);
        }
        party.MapEvent = null; party.Party.MapEvent = new MapEvent { MapEventSettlement = village, IsRaid = true };
        Program.Check(IsVillageRaidMobileParty(party), "PartyBase raid event remains authoritative");
        party.Party.MapEvent = new MapEvent { MapEventSettlement = village };
        Program.Check(!IsVillageRaidMobileParty(party), "non-raid event is not promoted by AI raid intent");
        Program.Check(IsVillageRaidParty(new PartyBase { IsSettlement = true, Settlement = village }), "raided settlement party remains protected");
        Program.Check(!IsVillageRaidMobileParty(null), "missing mobile party does not qualify as raid");
    }
}

internal sealed class PartyBase { public Hero LeaderHero = new Hero(); public bool IsNonHero; }
internal static class TargetAudit
{
    public static void Run()
    {
        var owner = new AnimusForge.Refactor.Modules.EncounterTargetOwner<Hero, PartyBase>();
        var army = new PartyBase();
        var selectedMember = new Hero();
        var anotherMember = new Hero();
        var members = new Dictionary<PartyBase, HashSet<Hero>>
        {
            [army] = new HashSet<Hero> { army.LeaderHero, selectedMember, anotherMember }
        };
        bool Eligible(Hero hero, PartyBase party) => party != null
            && members.TryGetValue(party, out var roster) && roster.Contains(hero);
        Hero Fallback(PartyBase party) => party?.LeaderHero;
        owner.Set(selectedMember);
        Program.Check(ReferenceEquals(owner.Ensure(army, Eligible, Fallback, out bool refreshed, out bool cleared), selectedMember)
            && !refreshed && !cleared, "selected army member outranks encountered leader");
        Program.Check(ReferenceEquals(owner.Ensure(army, Eligible, Fallback, out refreshed, out cleared), selectedMember)
            && !refreshed, "menu recheck retains selected member");
        owner.Set(anotherMember);
        Program.Check(ReferenceEquals(owner.Ensure(army, Eligible, Fallback, out refreshed, out cleared), anotherMember),
            "second selection replaces previous member");
        var nextEncounter = new PartyBase();
        members[nextEncounter] = new HashSet<Hero> { nextEncounter.LeaderHero };
        Program.Check(ReferenceEquals(owner.Ensure(nextEncounter, Eligible, Fallback, out refreshed, out cleared), nextEncounter.LeaderHero)
            && refreshed && !cleared, "new encounter invalidates old member and falls back to its own leader");
        members[nextEncounter].Remove(nextEncounter.LeaderHero);
        nextEncounter.LeaderHero = null;
        Program.Check(owner.Ensure(nextEncounter, Eligible, Fallback, out refreshed, out cleared) == null
            && !refreshed && cleared, "missing leader clears stale selected target");
        Program.Check(owner.Ensure(nextEncounter, Eligible, Fallback, out refreshed, out cleared) == null
            && !cleared, "empty recheck does not claim another clear");
    }
}
internal static class ConversationTargetAudit
{
    public static void Run()
    {
        var selectedMember = new Hero();
        var leader = new Hero();
        var instanceLord = new Hero();
        var invalid = new Hero();
        object[] args = { invalid, selectedMember, leader };
        int fallbackCalls = 0;
        Hero Resolve(object instance, object[] values, Hero fallback) =>
            EncounterConversationTargetOwner.Resolve(instance, values,
                value => value as Hero, value => value != null && !ReferenceEquals(value, invalid),
                () => { fallbackCalls++; return fallback; });
        Program.Check(ReferenceEquals(Resolve(instanceLord, args, leader), selectedMember),
            "selected army member argument outranks instance and encountered leader");
        Program.Check(fallbackCalls == 0, "valid argument does not evaluate leader fallback");
        Program.Check(ReferenceEquals(Resolve(instanceLord, new object[] { invalid }, leader), instanceLord),
            "usable instance outranks encountered leader after invalid arguments");
        Program.Check(ReferenceEquals(Resolve(invalid, null, leader), leader),
            "leader is last fallback for missing arguments and unusable instance");
        Program.Check(Resolve(null, new object[] { invalid }, invalid) == null,
            "unusable leader cannot become conversation target");
        Program.Check(Resolve(null, null, null) == null,
            "missing source leaves conversation target unset");
    }
}
internal static class ReturnAudit
{
    public static void Run()
    {
        var owner = new EncounterPendingReturnOwner<object, object>();
        object encounter = new object(), party = new object();
        Program.Check(!owner.Mark(null, party, 1) && !owner.IsPending,
            "missing encounter cannot queue return");
        Program.Check(!owner.Mark(encounter, null, 1) && !owner.IsPending,
            "missing party cannot queue return");
        Program.Check(owner.Mark(encounter, party, 1) && owner.IsCurrent(encounter, party, 1),
            "unauthorized exit captures exact encounter/party/generation");
        Program.Check(!owner.IsCurrent(new object(), party, 1) && !owner.IsCurrent(encounter, new object(), 1),
            "new encounter or party cannot consume stale return");
        Program.Check(!owner.IsCurrent(encounter, party, 2), "save generation invalidates pending return");
        owner.Clear();
        Program.Check(!owner.IsPending && !owner.IsCurrent(encounter, party, 1),
            "release/cancel clears return and rejects old callback");
    }

    private static Mission Reset(bool mission=true)
    {
        PlayerEncounter.Current=new PlayerEncounter(); _party=new PartyBase(); _targetHero=new Hero();
        Mission.Current=mission ? new Mission() : null;
        Game.Current.GameStateManager.ActiveState=mission ? new MissionState() : null;
        Time.ApplicationTime=0; SaveRuntimeGuard.Generation=1; _executions=0;
        Campaign.Current.ConversationManager=new ConversationManager();
        _releaseOwner.Schedule(CaptureMeetingPlayerReleaseRequest(_targetHero, "test"));
        return Mission.Current;
    }
    public static void Run()
    {
        var original=Reset(); Time.ApplicationTime=1; TryForcePendingNativeConversationMeetingReleaseIfReady();
        Program.Check(original.EndCount==0 && _executions==0, "release waits from application time zero");
        Time.ApplicationTime=10; TryForcePendingNativeConversationMeetingReleaseIfReady();
        Program.Check(original.EndCount==1, "release exits captured mission at deadline");
        Program.Check(ConsumeMeetingPlayerReleaseAuthorization("exit"), "manual or timed exit remains authorized");
        Program.Check(!ConsumeMeetingPlayerReleaseAuthorization("duplicate"), "authorization consumed once");
        foreach (string change in new[] { "encounter", "party", "mission", "generation", "missing", "expiry" })
        {
            original=Reset(); Time.ApplicationTime=11;
            if(change=="encounter") PlayerEncounter.Current=new PlayerEncounter();
            if(change=="party") _party=new PartyBase();
            if(change=="mission") Mission.Current=new Mission();
            if(change=="generation") SaveRuntimeGuard.Generation++;
            if(change=="missing") PlayerEncounter.Current=null;
            if(change=="expiry") Time.ApplicationTime=121;
            var active=Mission.Current;
            TryForcePendingNativeConversationMeetingReleaseIfReady();
            Program.Check(_pendingNativeConversationMeetingRelease==null && _meetingPlayerReleaseAuthorization==null, change+" cancels owned authorization");
            Program.Check(original.EndCount==0 && active.EndCount==0 && _executions==0, change+" performs no native exit or release");
        }
        Reset(); Time.ApplicationTime=121; Program.Check(!ConsumeMeetingPlayerReleaseAuthorization("expired_without_tick"), "expired authorization cannot be consumed before the next tick");
        Reset(); _party=new PartyBase(); Program.Check(!ConsumeMeetingPlayerReleaseAuthorization("changed"), "authorization cannot cross encounter party");
        original=Reset(); Mission.Current=null; Game.Current.GameStateManager.ActiveState=null; Time.ApplicationTime=1;
        TryForcePendingNativeConversationMeetingReleaseIfReady(); TryForcePendingNativeConversationMeetingReleaseIfReady();
        Program.Check(_executions==1 && original.EndCount==0, "manual map exit finalizes once without closing any mission");
        original=Reset(); Mission.Current=null; Time.ApplicationTime=11;
        TryForcePendingNativeConversationMeetingReleaseIfReady();
        Program.Check(_pendingNativeConversationMeetingRelease!=null && _executions==0, "mission teardown retains pending release until map state");
        Game.Current.GameStateManager.ActiveState=null; TryForcePendingNativeConversationMeetingReleaseIfReady();
        Program.Check(_executions==1 && original.EndCount==0, "mission teardown completes on following map tick");
        Reset(false); _party.IsNonHero=true; TryForcePendingNativeConversationMeetingReleaseIfReady();
        Program.Check(_releasedTarget==null, "nonhero party cannot inherit a stale lord target");
        Reset(); var pending=_pendingNativeConversationMeetingRelease; AuthorizeMeetingPlayerRelease("newer");
        ClearPendingNativeConversationMeetingRelease("cancel_older");
        Program.Check(_meetingPlayerReleaseAuthorization!=null && !ReferenceEquals(_meetingPlayerReleaseAuthorization,pending), "old cancellation preserves a distinct newer authorization");
        original=Reset(); Time.ApplicationTime=11; Campaign.Current.ConversationManager.IsConversationInProgress=true;
        Campaign.Current.ConversationManager.OnEnd=()=> { Mission.Current=new Mission(); };
        TryForcePendingNativeConversationMeetingReleaseIfReady();
        Program.Check(original.EndCount==0 && Mission.Current.EndCount==0 && _pendingNativeConversationMeetingRelease==null, "conversation callback cannot redirect mission exit");
    }
