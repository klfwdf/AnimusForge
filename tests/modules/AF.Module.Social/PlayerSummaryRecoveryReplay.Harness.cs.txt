        public static void RunChecks()
        {
            Fixture.MainThread = Thread.CurrentThread.ManagedThreadId;
            var owner = Owner(); owner.TryStartSummaryProcessing();
            Check(!CampaignEvents.TickEvent.Registered.Contains(owner), "summary pump is not tied to pausable campaign Tick");
            Check(ConfiguredChatApplicationAdapter.Requests.Count == 1 && owner._summaryRun != null, "one real in-flight task started");
            Check(!LlmNonStreamingTransport.CurrentOwnerCancellation.CanBeCanceled, "request cancellation scope does not leak into caller");
            var first = ConfiguredChatApplicationAdapter.Requests[0];
            Fixture.Day += 4;
            for (int i = 0; i < 4; i++) owner.OnDailyTick();
            owner.TryStartSummaryProcessing(true);
            Check(ConfiguredChatApplicationAdapter.Requests.Count == 1, "cross-day daily checks and additions cannot overlap request");
            Complete(Failed());
            Check(owner._state.SummarizedMaterialKeys.Count == 0, "worker completion never writes state");
            Tick(owner);
            Check(owner._state.SummaryRetryCount == 1 && owner._summarySchedulePhase == "retry_wait", "failed attempt schedules retry");
            Tick(owner); Check(ConfiguredChatApplicationAdapter.Requests.Count == 1, "retry respects monotonic delay");
            FakeSummaryClock.Advance(2); Tick(owner);
            Check(ConfiguredChatApplicationAdapter.Requests.Count == 2, "retry starts without another game day or new material");
            Check(ConfiguredChatApplicationAdapter.Requests[1].Prompt == first.Prompt, "same batch/input retained across retries");
            Complete(Ok()); Tick(owner);
            Check(owner._summaryRun == null && owner._state.MajorMaterials.Count == 0 && owner._state.SummarizedMaterialKeys.Count == 1, "successful main-thread commit consumes exact batch");
            Check(owner._state.CultureNotoriety["empire"] == 2 && owner._state.LastSummaryDay == Fixture.Day, "success writes fame once and advances interval");
            Tick(owner); Tick(owner);
            Check(owner._state.CultureNotoriety["empire"] == 2 && ConfiguredChatApplicationAdapter.Requests.Count == 2, "repeated ticks cannot duplicate success");

            owner = Owner(); owner.TryStartSummaryProcessing(); owner.SyncData(new Store { IsSaving = true });
            Check(owner._summaryRun != null && !ConfiguredChatApplicationAdapter.Requests[0].Token.IsCancellationRequested, "saving does not cancel valid in-flight work");
            Complete(Ok()); Tick(owner);
            Check(owner._state.MajorMaterials.Count == 0 && owner._state.SummarizedMaterialKeys.Count == 1, "save normalization preserves source identities for eventual commit");

            owner = Owner(); owner._state.MajorSummary = "OLD-" + new string('o', 290);
            owner._state.MajorMaterials[0].Text = "NEWEST confirmed deed."; owner.TryStartSummaryProcessing();
            Complete(Failed()); Tick(owner); FakeSummaryClock.Advance(2); Tick(owner);
            Complete(new() { Success = true, Content = "not-json" }); Tick(owner); FakeSummaryClock.Advance(4); Tick(owner);
            Complete(new() { Success = true, Content = "" }); Tick(owner);
            Check(ConfiguredChatApplicationAdapter.Requests.Count == 3 && owner._summaryRun == null, "transport/format/empty failures share three-attempt budget");
            Check(owner._state.MajorMaterials.Count == 0 && owner._state.SummarizedMaterialKeys.Count == 1, "fallback consumes only confirmed materials");
            Check(owner._state.MajorSummary.Length <= Fixture.Limit && owner._state.MajorSummary.EndsWith("NEWEST confirmed deed."), "bounded fallback keeps newest facts");
            Check(owner._state.CultureNotoriety.Count == 0 && owner._state.WorldNotoriety == 0, "fallback grants no model fame");
            Check(owner._state.LastSummaryDay == Fixture.Day && owner._state.SummaryRetryCount == 0 && owner._state.LastSummaryError.Length > 0, "fallback records completion clock and retains real error");
            Check(Logger.Lines.Any(x => x.Contains("summary_fallback")), "fallback clearly logged, not model success");
            owner = Owner(); owner.TryStartSummaryProcessing(); Complete(Failed()); Tick(owner);
            FakeSummaryClock.Advance(2); Tick(owner); Complete(Failed()); Tick(owner); FakeSummaryClock.Advance(4); Tick(owner);
            owner._state.MajorMaterials.Add(Material("fallback-later", "A later deed.")); Complete(Failed()); Tick(owner);
