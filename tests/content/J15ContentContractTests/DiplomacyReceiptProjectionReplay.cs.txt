using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Xml.Linq;
using AnimusForge;

internal static class Program
{
    internal static string PackageRoot;
    internal static Dictionary<string, string> Ui;
    private static int assertions;
    private static void Check(bool value, string name)
    { assertions++; if (!value) throw new InvalidOperationException(name); }
    private sealed class State
    {
        internal readonly List<string> Calls = new();
        internal readonly List<string> Logs = new();
        internal string Result;
        internal bool Formal;
    }
    private readonly struct Source : IDiplomacyOralTagSource, IDiplomacyCommitmentTagSource, IDiplomacySubjectReleaseTagSource
    {
        private readonly State state;
        internal Source(State state) => this.state = state;
        public bool HasSpeaker => true;
        public bool IsAvailable => true;
        public string SpeakerHeroId => "speaker";
        public string SpeakerKingdomId => "realm";
        public bool UseFormalCommitments => state.Formal;
        public string SubmitCommitment(string payload) => Apply("commit", payload);
        public string ControlCommitment(string payload) => Apply("control", payload);
        public string SubmitLegacyCommitment(string action, string payload) => Apply(action, payload);
        public string ReleaseSubject(string payload) => Apply("release", payload);
        private string Apply(string action, string payload)
        { state.Calls.Add(action + ":" + payload); return state.Result; }
        public string DeclareWar(string payload) => Apply("war", payload);
        public string MakePeace(string payload) => Apply("peace", payload);
        public string IndependentClanPeace(string payload) => Apply("independent", payload);
        public string FormAlliance(string payload) => Apply("alliance", payload);
        public string BreakAlliance(string payload) => Apply("break", payload);
        public string MakeTrade(string payload) => Apply("trade", payload);
        public string CancelTrade(string payload) => Apply("cancel", payload);
        public void Log(string message) { state.Logs.Add(message); }
    }
    internal static void Main(string[] args)
    {
        PackageRoot = args[0];
        Console.OutputEncoding = System.Text.Encoding.UTF8;
        foreach (string locale in new[] { "en", "cn" })
        {
            Ui = XDocument.Load(args[locale == "en" ? 1 : 2]).Descendants("string")
                .ToDictionary(node => (string)node.Attribute("id"), node => (string)node.Attribute("text"));
            foreach (string channel in new[] { "courier", "native", "scene", "system_scene" })
            {
                var state = new State(); var source = new Source(state);
                string reason = "literal {NAME} {0} $1";
                state.Result = string.Format(DiplomacyMiscPromptTexts.Get("OralDiplomacy_SubmissionRejected", ""), reason);
                string body = "player/model same-text: " + state.Result + " [ACTION:DIPLOMACY:MAKE_PEACE:first] end\n[ACTION:DIPLOMACY:MAKE_TRADE:second]";
                var receipts = new List<string>();
                DiplomacyOralTagApplication.Process(source, ref body, receipts);
                string expectedBody = "player/model same-text: " + state.Result + "  end";
                Check(body == expectedBody, channel + ": same-text model body preserved");
                Check(state.Calls.SequenceEqual(new[] { "peace:first", "trade:second" }), channel + ": once-only ordered execution");
                Check(receipts.Count == 0, channel + ": no system receipts enter prose pipeline");
                Check(state.Logs.Count(line => line.StartsWith("[Outcome]")) == 2, channel + ": outcomes retained in diagnostics");
                string raw = DiplomacyOralTagApplication.ComposeReceiptText(body, receipts);
                string visible = DiplomacyOralTagApplication.ComposeReceiptText(body, receipts, DiplomacyDialogueStatusUi.Project);
                string translated = DiplomacyUiLocalization.Format("AF_DIP_STATUS_OralDiplomacy_SubmissionRejected", "", ("REASON", reason));
                Check(raw == expectedBody, channel + ": history contains NPC speech only");
                Check(visible == expectedBody, channel + ": visible prose contains no appended receipts");
                Check(DiplomacyDialogueStatusUi.Project(state.Result) == translated, channel + ": independent notifications still localize");
                var saved = new List<string> { state.Result, state.Result };
                Check(DiplomacyOralTagApplication.ComposeReceiptText(body, saved, _ => throw new Exception("must not project prose receipts")) == expectedBody,
                    channel + ": saved receipts cannot reappear");
                Check(body == expectedBody && saved.All(value => value == state.Result), channel + ": display never mutates source");
                Check(state.Calls.Count == 2, channel + ": rendering never repeats actions");
                foreach (string key in new[] { "OralDiplomacy_DeferredPrefix", "OralDiplomacy_CancelledPrefix", "OralDiplomacy_Published", "OralDiplomacy_Executed" })
                {
                    string status = DiplomacyMiscPromptTexts.Get(key, "") + (key.EndsWith("Prefix") ? reason : "");
                    string projected = DiplomacyDialogueStatusUi.Project(status);
                    Check(projected == Ui["AF_DIP_STATUS_" + key] + (key.EndsWith("Prefix") ? reason : ""), channel + ": status/causal detail " + key);
                }
                string unknown = "user unchanged {NAME} {0} $1";
                Check(DiplomacyDialogueStatusUi.Project(unknown) == unknown, channel + ": unknown/user text remains original");
            }
        }
        var keys = (string[])typeof(DiplomacyReplyTextRules).GetField("Keys", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static).GetValue(null);
        foreach (string key in keys)
        {
            string template = DiplomacyMiscPromptTexts.GetRequired(key);
            string status = System.Text.RegularExpressions.Regex.Replace(template, @"\{[0-9]+}", _ => "literal {NAME} $1");
            if (key.EndsWith("Prefix")) status += "opaque reason";
            Check(DiplomacyReplyTextRules.CleanNpcText("prose\r\n" + status) == "prose", "legacy canonical status removed: " + key);
            Check(DiplomacyReplyTextRules.CleanNpcText(status + "\nprose") == "prose", "status-only first line removed: " + key);
            Check(DiplomacyReplyTextRules.CleanNpcText("Ada: " + status, "Ada") == "", "known speaker-wrapped status removed: " + key);
            string quote = "NPC quoting: \"" + status + "\"";
            Check(DiplomacyReplyTextRules.CleanNpcText(quote) == quote, "inline quotation retained: " + key);
            Check(CourierVisibleLetterSanitizer.Clean(status) == status, "player letter literal retained: " + key);
            Check(CourierVisibleLetterSanitizer.CleanNpcLetter("prose\n" + status) == "prose", "NPC letter status removed: " + key);
        }
        foreach (string wrapper in new[] { "【回信】", "[Reply]" })
        {
            string wrappedStatus = wrapper + DiplomacyMiscPromptTexts.GetRequired("OralDiplomacy_InvalidFormat");
            Check(DiplomacyReplyTextRules.CleanNpcText(wrappedStatus) == "", "both embedded reply wrappers recognize exact status across packages");
            string ordinary = wrapper + "Keep this actual reply and its conditions.";
            Check(DiplomacyReplyTextRules.CleanNpcText(ordinary) == ordinary, "reply wrapper never removes ordinary prose");
        }
        foreach (var pair in Ui.Where(pair => pair.Key.StartsWith("AF_DIP_STATUS_") && keys.Contains(pair.Key.Substring("AF_DIP_STATUS_".Length))))
        {
            string status = pair.Value.Replace("{REASON}", "opaque reason");
            if (pair.Key.EndsWith("Prefix")) status += "opaque reason";
            Check(DiplomacyReplyTextRules.CleanNpcText("prose\n" + status) == "prose", "UI status removed: " + pair.Key);
        }
        foreach (bool formal in new[] { false, true })
        foreach (string tag in new[] { "COMMIT:malformed", "COMMITMENT:malformed", "RELEASE_SUBJECT:malformed" })
        {
            var state = new State { Formal = formal, Result = DiplomacyMiscPromptTexts.GetRequired("OralDiplomacy_InvalidFormat") };
            string body = "prose\n[ACTION:DIPLOMACY:" + tag + "]";
            var receipts = new List<string>();
            DiplomacyOralTagApplication.Process(new Source(state), ref body, receipts);
            Check(body == "prose" && receipts.Count == 0, "formal/disabled late tag never leaks: " + formal + tag);
            Check(state.Logs.Any(line => line.StartsWith("[Outcome]")), "rejected outcomes remain diagnosable: " + formal + tag);
        }
        foreach (string status in new[] { "外交约定暂未提交：条款格式无效。", "Diplomatic commitment not submitted: clause format is invalid." })
        {
            string original = "ordinary speech\n" + status;
            Check(CourierVisibleLetterSanitizer.CleanNpcLetter(original) == "ordinary speech", "both languages available independent of pack");
            Check(ConversationSpeechTextRules.SanitizeSceneSpeechText(original, default) == "ordinary speech", "Native/Scene visible guard");
            Check(ConversationSpeechTextRules.PrepareSceneHistorySpeechText(original) == "ordinary speech", "new NPC history guard");
            Check(ConversationSpeechTextRules.SanitizeSceneSpeechTextForTts(original) == "ordinary speech", "early/final TTS guard");
            Check(!string.IsNullOrWhiteSpace(ConversationSpeechTextRules.NormalizeNativeConversationVisibleTextKey(status, default)),
                "generic player/history deduplication must not clear literal receipt text");
            var document = new WorldDiplomacyDocument { Body = original };
            Check(WorldDiplomacyPresentationQueries.ProjectDocumentBody(document) == "ordinary speech", "NPC formal document body guard");
            document.IsPlayerAuthored = true;
            Check(WorldDiplomacyPresentationQueries.ProjectDocumentBody(document) == original && document.Body == original,
                "player-authored formal documents and stored bodies retained");
            foreach (string role in new[] { "assistant", "user", "system" })
            {
                var message = new ConversationMessage { Role = role, SpeakerName = "Ada", Content = original };
                Check(MainPromptMessageAssemblyOwner.TryConvertCourierMemoryMessageToChatMessage(message, "Ada", "Player", out var courier), "Courier history projects");
                string courierText = (string)courier.GetType().GetProperty("content").GetValue(courier);
                Check(courierText.Contains(status) == (role != "assistant"), "Courier preserves player/system and cleans NPC history: " + role);
                Check(HistorySectionProjectionOwner.TryRenderSceneHistoryLine(message, null, out string scene), "Scene/Native history projects");
                Check(scene.Contains(status) == (role != "assistant"), "Scene preserves player/system and cleans NPC history: " + role);
                Check(message.Content == original, "stored identity/raw history remains unchanged");
            }
        }
        string opaqueProse = "自然外交台词。\r\nUnknown dynamic text {NAME} $1";
        Check(DiplomacyReplyTextRules.CleanNpcText(opaqueProse) == opaqueProse, "unknown prose retains exact bytes");
        Check(DiplomacyReplyTextRules.CleanNpcText("[AFEF NPC行为补充] 正式宣言已发布，外交行动已生效。")
            == "[AFEF NPC行为补充] 正式宣言已发布，外交行动已生效。", "AFEF fact remains intact");
        Console.WriteLine("PASS diplomacy prose boundary: " + assertions + " assertions; production parser/composer/cleaner/speech and source-derived Courier/history spans; game effects/fact leaves synthetic.");
    }
}

namespace AnimusForge
{
    // Package paths and the game-language TextObject leaf are test-owned boundaries.
    // The production JSON reader, tag parser, composer and status projector are compiled unchanged.
    internal static class AnimusForgeModulePaths
    { internal static string GetModuleDataFilePath(string name) => Path.Combine(Program.PackageRoot, name); }
    internal static class DiplomacyUiLocalization
    {
        internal static string Text(string key, string englishFallback) => Program.Ui.TryGetValue(key, out string text) ? text : englishFallback;
        internal static string Format(string key, string fallback, params (string Name, string Value)[] variables)
        {
            string text = Text(key, fallback);
            return System.Text.RegularExpressions.Regex.Replace(text, @"\{([A-Z_]+)}", match =>
            {
                foreach (var variable in variables) if (variable.Name == match.Groups[1].Value) return variable.Value;
                return match.Value;
            });
        }
    }
}
