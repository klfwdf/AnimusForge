    }
    static void CompletionGuards()
    {
        foreach(bool failFirst in new[]{false,true})
        {
            var fixture=new Fixture(failFirst?"normalize-throws-no-reentry":"complete-once","duel"){ThrowOnNormalize=failFirst};
            F.Reset(fixture);
            var prepare=typeof(NewHost).GetMethod("PrepareSceneUnifiedActionPostprocess",Flags);
            var complete=typeof(NewHost).GetMethod("CompleteSceneUnifiedActionPostprocess",Flags);
            var work=prepare.Invoke(null,fixture.Args(prepare));
            object[] args={work,true,fixture.Raw,null};
            bool failed=false;
            try { complete.Invoke(null,args); }
            catch(TargetInvocationException e) when(e.InnerException is InvalidOperationException) {failed=true;}
            Check(failed==failFirst,fixture.Name+" unexpected first completion");
            int before=F.Calls.Count;
            bool rejected=false;
            try { complete.Invoke(null,args); }
            catch(TargetInvocationException e) when(e.InnerException is InvalidOperationException) {rejected=true;}
            Check(rejected,fixture.Name+" second completion was not rejected");
            Check(F.Calls.Count==before,fixture.Name+" second completion repeated domain effects");
            Check(F.Calls.Count(c=>c.StartsWith("MarkWeekly"))==(failFirst?0:1),fixture.Name+" weekly side effect count");
            Console.WriteLine("PASS "+fixture.Name);
        }
    }
    public static int Main()
    {
        try
        {
            bool phases=typeof(NewHost).GetMethod("PrepareSceneUnifiedActionPostprocess",Flags)!=null;
            foreach(var fixture in Fixtures())
            {
                var old=Run(typeof(OldHost),fixture); var current=Run(typeof(NewHost),fixture);
                Equal(old,current,fixture.Name+" wrapper");
                if(phases) Equal(old,Run(typeof(NewHost),fixture,true),fixture.Name+" phases");
                Check(current.Item2.Count(c=>c.StartsWith("NETWORK:"))<=1,fixture.Name+" duplicate network");
                var latest=current.Item2.FirstOrDefault(c=>c.StartsWith("LATEST:"));
                if(latest!=null)
                {
                    using var json=JsonDocument.Parse(latest.Substring("LATEST:".Length));
                    Check(json.RootElement[0].GetString()==(fixture.Direct?fixture.Player:""),fixture.Name+" direct/relay player attribution");
                    if(!fixture.Direct) Check(json.RootElement[3].ValueKind==JsonValueKind.Null,fixture.Name+" relay inherited direct-player history");
                }
                if(fixture.NoRules) Check(current.Item2.Any(c=>c.Contains("mergedRules=(none; mood postprocess still runs)")),fixture.Name+" fixture did not actually remove all action rules");

                if(!fixture.Aux||!fixture.Templates||!fixture.NetworkSuccess) Check(!current.Item2.Any(c=>c.StartsWith("MarkWeekly")),fixture.Name+" fallback wrote memory");
                Console.WriteLine("PASS "+fixture.Name+" requests="+current.Item2.Count(c=>c.StartsWith("NETWORK:"))+" calls="+current.Item2.Count);Count++;
            }
            if(phases) CompletionGuards();
            Console.WriteLine("PASS ScenePostprocessParity fixtures="+Count+" originalMethod=executed candidateMethod=executed splitPhases="+phases+" domainHelpers=STUBBED live=NOT_RUN save=NOT_RUN apiNetwork=NOT_RUN");
            return 0;
        }
        catch(Exception e) {Console.Error.WriteLine(e);return 1;}
    }
