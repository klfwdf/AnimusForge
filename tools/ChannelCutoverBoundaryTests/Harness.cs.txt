        string moduleId, string providerId, CancellationToken cancellationToken)
    {
        Scenario scenario = Scenario.Current;
        scenario.HostCalls++;
        scenario.CapturePortDelegates(Ports, envelope);
        scenario.AfterGeneration?.Invoke();
        if (scenario.HostThrows) throw new InvalidOperationException("generation-failed-before-commit");
        if (scenario.NullResult) return Task.FromResult<InteractionResult>(null);
        ActionPlan resultActions = scenario.UnexpectedGeneratedActions
            ? new ActionPlan(new[] { new ActionRequest("ACTION:GIVE_GOLD", "npc-alpha", new Dictionary<string, string>()) }, "[ACTION:GIVE_GOLD:100]")
            : scenario.CapturedActionPlan;

            resultActions, Array.Empty<FactRecord>(), "fixture-error", scenario.VisibleReply));
    }

    {
        Scenario.Current.EarlyCommitCalls++;
        throw new InvalidOperationException("Main generation must never commit");
    }

    {
        if (Scenario.Current.DisposeThrows) throw new InvalidOperationException("dispose-failed");
    }

    {

public static class Program
{
    private static int failures, passes;
    private static void Require(bool condition, string message)
    {
        if (!condition) throw new Exception(message);
    }
    private static async Task Case(string name, bool courier, Action<Scenario> arrange, Action<Scenario> assert,
        Func<ProductionBlockHost, Task> execute = null)
    {
        var scenario = new Scenario { Name = name, Status = courier ? InteractionStatus.Executed : InteractionStatus.Succeeded };
        Scenario.Current = scenario;
        ProductionBlockHost.SetSceneSessionForFixture(1);
        arrange(scenario);
        try
        {
            var host = new ProductionBlockHost();
            if (execute != null) await execute(host);
            else if (courier) await host.CourierAsync(); else await host.SceneAsync();
            scenario.Drain();
            assert(scenario);
            passes++;
            Console.WriteLine("PASS " + name);
        }
        catch (Exception ex)
        {
            failures++;
            Console.WriteLine("FAIL " + name + ": " + ex.Message + " host=" + scenario.HostCalls + " legacy=" + scenario.LegacyCalls);
        }
    }
    private static void LegacyOnce(Scenario s)
    {
        Require(s.HostCalls == 0 && s.LegacyCalls == 1, "Pre-submit path must invoke legacy exactly once");
    }
    private static void NoSecondRequest(Scenario s)
    {
        Require(s.HostCalls == 1 && s.LegacyCalls == 0, "Submitted host owns outcome; must not issue second legacy request");
    }
    private static void TerminalScene(Scenario s)
    {
        NoSecondRequest(s);
        Require(s.SceneCleanup, "Terminal scene must break to tail cleanup rather than return");
        Require(!s.ScenePostBlock, "Terminal scene must not continue reply/action processing");
    }
    private static void SceneGenerationOnly(Scenario s)
    {
        Require(!s.SceneCommitted && s.EarlyCommitCalls == 0,
            "Scene main result is not a game/history commit; early effects must remain zero");
        Require(s.CapturedPostprocess == null && s.PostprocessCompositions == 0,
            "Main generation must not compose the reduced detached postprocess");
        Require(s.CapturedActionPlan != null && s.CapturedActionPlan.Actions.Count == 0,
            "Main generation must not create executable actions from reply text");
    }
    private static void SceneFallbackOnce(Scenario s)
    {
        Require(s.HostCalls == 1 && s.LegacyCalls == 1, "Only explicit pre-effect failures allow one internal fallback");
        Require(s.SceneCleanup && s.ScenePostBlock && s.SceneReply == s.LegacyReply,
            "Internal fallback must preserve its reply and Scene downstream ownership");
        SceneGenerationOnly(s);
    }
    private static void TerminalCourier(Scenario s)
    {
        NoSecondRequest(s);
        Require(s.FinalizeCalls == 0, "Terminal failure cannot finalize as a successful reply");
        Require(s.Session.PostprocessConsumed, "Unknown/failed commit must reserve action consumption");

    public static async Task<int> Main()
    {
        foreach (bool courier in new[] { false, true })
        {
            string channel = courier ? "courier" : "scene";
            await Case(channel + "/bridge-disabled", courier, s => s.BridgeEnabled = false, LegacyOnce);
            await Case(channel + "/capture-null", courier, s => s.CaptureNull = true, LegacyOnce);
            await Case(channel + "/prepare-exception", courier, s => s.PrepareThrows = true, LegacyOnce);
            foreach (InteractionStatus status in new[] {
                InteractionStatus.NonRetryableFailure, InteractionStatus.RejectedByValidation,
                InteractionStatus.RetryableFailure, InteractionStatus.SkippedByEligibility, InteractionStatus.DegradedWithoutProvider })
            {
                bool allowsMainFallback = !courier && (status == InteractionStatus.RetryableFailure
                    || status == InteractionStatus.SkippedByEligibility || status == InteractionStatus.DegradedWithoutProvider);
                await Case(channel + (allowsMainFallback ? "/pre-effect-fallback-" : "/terminal-") + status,
                    courier, s => s.Status = status,
                    allowsMainFallback ? SceneFallbackOnce : courier ? TerminalCourier : TerminalScene);
            }
            await Case(channel + "/host-throws", courier, s => s.HostThrows = true, courier ? TerminalCourier : TerminalScene);
            await Case(channel + "/null-result", courier, s => s.NullResult = true, courier ? TerminalCourier : TerminalScene);
            await Case(channel + "/stale", courier, s => s.Status = InteractionStatus.CancelledAsStale, s =>
            {
                NoSecondRequest(s);
                if (!courier) Require(!s.SceneCleanup && !s.ScenePostBlock, "Stale scene must return without touching a new epoch");
                else Require(s.FailureCalls == 0 && s.FinalizeCalls == 0, "Stale courier result must not mutate current session");
            });
            foreach (InteractionStatus status in courier
                ? new[] { InteractionStatus.Executed, InteractionStatus.Succeeded }
                : new[] { InteractionStatus.Succeeded })
            foreach (string reply in new[] { "detached-reply", "" })
            {
                await Case(channel + "/success-" + status + (reply == "" ? "-empty" : "-text"), courier,
                    s => { s.Status = status; s.VisibleReply = reply; }, s =>
                    {
                        NoSecondRequest(s);
                        Require(s.FailureCalls == 0, "Success cannot become failure");
                        if (courier) Require(s.FinalizeCalls == 1, "Success must finalize once");
                        else
                        {
                            Require(s.SceneCleanup && s.SceneReply == reply, "Success must preserve visible reply for later authoritative Scene processing");
                            SceneGenerationOnly(s);
                        }
                    });
            }
            foreach (string reply in new[] { "legacy-reply", "" })
            {
                await Case(channel + "/internal-fallback" + (reply == "" ? "-empty" : "-text"), courier,
                    s => { s.UsedLegacyFallback = true; s.VisibleReply = reply; s.LegacyReply = reply;
                        if (!courier) s.Status = InteractionStatus.RetryableFailure; }, s =>
                    {
                        Require(s.HostCalls == 1 && s.LegacyCalls == 1, "Internal fallback is already executed and must not run again");
                        Require(s.FailureCalls == 0, "Internal fallback must not be reclassified as terminal failure");
                        if (!courier) SceneFallbackOnce(s);
                        else Require(s.FinalizeCalls == 0 && s.LegacyCompletionCalls == 1, "Legacy fallback owns its queued finalization");
                    });
            }
        }
        await Case("scene/capture-exception", false, s => s.CaptureThrows = true, LegacyOnce);
        foreach (string reply in new[] { "unexpected-executed", "" })
            await Case("scene/unexpected-executed-status-" + (reply.Length == 0 ? "empty" : "text"), false,
                s => { s.Status = InteractionStatus.Executed; s.VisibleReply = reply; }, TerminalScene);
        await Case("scene/internal-fallback-failure", false,
            s => { s.Status = InteractionStatus.RetryableFailure; s.LegacyThrows = true; }, s =>
            {
                Require(s.HostCalls == 1 && s.LegacyCalls == 1, "Failed internal fallback cannot trigger another legacy request");
                Require(s.SceneCleanup && !s.ScenePostBlock, "Failed internal fallback must stop reply processing and reach cleanup");
            });
        await Case("courier/internal-fallback-failure-owns-completion", true,
            s => { s.UsedLegacyFallback = true; s.Status = InteractionStatus.NonRetryableFailure; }, s =>
            {
                Require(s.HostCalls == 1 && s.LegacyCalls == 1 && s.LegacyCompletionCalls == 1, "Already-started legacy callback retains completion ownership");
                Require(s.FailureCalls == 0, "Outer failure must not clear pending legacy completion");
            });
        await Case("courier/pre-delivery-legacy-generation", true, s => s.Session.DeliveryApplied = false, LegacyOnce);
        await Case("scene/success-before-dispose-throws", false, s => s.DisposeThrows = true, TerminalScene);
        await Case("scene/stale-before-generation", false, s => s.RuntimeStale = true, s =>
        {
            Require(s.HostCalls == 0 && s.LegacyCalls == 0 && !s.SceneCleanup && !s.ScenePostBlock,
                "Stale generation must stop before request/fallback and avoid new-epoch mutation");
        });
        await Case("scene/stale-after-generation", false, s => s.AfterGeneration = () => s.RuntimeStale = true, s =>
        {
            NoSecondRequest(s);
            Require(!s.SceneCleanup && !s.ScenePostBlock, "Generation changed while awaiting main result must stop all later processing");
            SceneGenerationOnly(s);
        });
        await Case("scene/stale-before-fallback", false, s =>
        {
            s.Status = InteractionStatus.RetryableFailure;
            s.AfterGeneration = () => s.RuntimeStale = true;
        }, s =>
        {
            NoSecondRequest(s);
            Require(!s.SceneCleanup && !s.ScenePostBlock, "Stale failure must not start legacy fallback");
        });
        await Case("scene/stale-during-fallback", false, s =>
        {

            s.DuringFallback = () => s.RuntimeStale = true;
        }, s =>
        {
            Require(s.HostCalls == 1 && s.LegacyCalls == 1 && !s.SceneCleanup && !s.ScenePostBlock,
                "Old fallback completion cannot write history or actions into a new generation");
        });
        await Case("courier/finalize-queued-before-dispose-throws", true, s => s.DisposeThrows = true, s =>
        {
            NoSecondRequest(s);
            Require(s.FinalizeCalls == 1 && s.FailureCalls == 0, "Late exception must not overwrite successful queued finalization");
            Require(s.Session.ReplyText == s.VisibleReply && s.Session.ReplyGenerated, "Success must remain intact");
        });
        await Case("courier/terminal-queued-before-dispose-throws", true,
            s => { s.DisposeThrows = true; s.Status = InteractionStatus.NonRetryableFailure; }, s =>
            {
                TerminalCourier(s);
                Require(s.Notifications == 1, "Two queued failure actions must be idempotent");
            });
        await Case("courier/legacy-completion-queued-before-dispose-throws", true,
            s => { s.DisposeThrows = true; s.UsedLegacyFallback = true; }, s =>
            {
                Require(s.LegacyCalls == 1 && s.LegacyCompletionCalls == 1, "Dispose failure must retain a single legacy completion");
                Require(s.FailureCalls == 0 && s.Session.ReplyText == s.LegacyReply, "Failure must not clear queued legacy reply");
            });
        await Case("courier/callback-queued-before-host-throws", true, s => s.CallbackThenThrows = true, s =>
        {
            Require(s.HostCalls == 1 && s.LegacyCalls == 1 && s.LegacyCompletionCalls == 1, "Callback ownership survives host exception");
            Require(s.FailureCalls == 0 && s.Session.ReplyText == s.LegacyReply, "Queued callback completion cannot be cleared");
        });
        await Case("courier/new-generation-before-failure-drain", true,
            s => { s.Status = InteractionStatus.NonRetryableFailure; s.StaleBeforeDrain = true; }, s =>
            {
                NoSecondRequest(s);
                Require(!s.Session.PostprocessConsumed && s.FailureCalls == 0, "Stale queued failure must not mutate new generation");
            });
        await PromptCases();
        await MainResultCases();
        await HistoryAndTailCases();
        await MainSpeechCases();
        await SceneLifecycleCases();
        Console.WriteLine("ChannelCutoverBoundary passes=" + passes + " failures=" + failures + " extractedProductionBlocks=true live=NOT_RUN apiNetwork=NOT_RUN");
        return failures == 0 ? 0 : 1;
    }

    private static (string Role, string Content)[] Rows(IEnumerable<object> messages)
        => messages.Select(item => JObject.FromObject(item)).Select(item => ((string)item["role"], (string)item["content"])).ToArray();

    private static void ExactMessages(PromptPackage actual, (string Role, string Content)[] expected)
    {
        Require(actual != null, "Actual default Scene caller did not compose a prompt");
        Require(actual.Messages.Count == expected.Length, "Prepared message count/order must not be recomposed or appended");
        for (int index = 0; index < expected.Length; index++)
            Require(actual.Messages[index].Role == expected[index].Role && actual.Messages[index].Content == expected[index].Content,
                "Prepared role/content changed at index " + index);
    }

                s => { s.SpeechDirectResponse = direct; s.SpeechInput = "安全回复[ACTION:GIVE_GOLD:100]"; }, s =>
            {
                Require(s.SpeechQueueResult && s.SpeechEnqueues == 1 && s.TargetChecks == 1,
                    "One validated game-thread boundary must enqueue the main reply exactly once");
                Require(s.EnqueuedSpeech == "安全回复" && s.EnqueuedAudienceCount == 2 && !s.EnqueuedCommitHistory,
                    "Speech queue must receive safe text and full audience without a second history commit");
                Require(s.EnqueuedAllowActions && s.EnqueuedEpoch == 1 && s.EnqueuedCanPublish != null,
                    "Validated Scene control support and publication lifetime guard must be retained");
                Require(s.EnqueuedTimeout == 17f && s.EnqueuedParticipants == 2 && !string.IsNullOrEmpty(s.EnqueuedInfo),
                    "Existing interaction timeout, participants and pending-postprocess hint must be forwarded");
                Require(s.EnqueuedPlayerText == (direct ? "original-player-input" : "")
                    && s.EnqueuedReplyText == (direct ? "安全回复" : ""), "Relay rounds must not gain player-directed action context");
            }, InvokeProductionSpeechQueueHelper);
        await Case("scene-speech/no-postprocess-has-no-pending-hint", false,
            s => s.SpeechPostprocess = false, s =>
        {
            Require(s.SpeechQueueResult && s.SpeechEnqueues == 1 && s.EnqueuedInfo == null,
                "Main text without postprocess must not display a stuck pending-action hint");
        }, InvokeProductionSpeechQueueHelper);
        foreach (string reason in new[] { "generation", "epoch", "session", "target", "tag-only" })
            await Case("scene-speech/reject-before-enqueue/" + reason, false, s =>
            {
                s.RuntimeStale = reason == "generation";
                s.EpochStale = reason == "epoch";
                s.SpeechSessionId = reason == "session" ? 99 : 1;
                s.TargetUnavailable = reason == "target";
                if (reason == "tag-only") s.SpeechInput = "[ACTION:DUEL]";
            }, s => Require(!s.SpeechQueueResult && s.SpeechEnqueues == 0,
                "Stale, unavailable or tag-only main text must not enter the speech worker"), InvokeProductionSpeechQueueHelper);
        foreach (string reason in new[] { "generation", "epoch", "session", "target" })
            await Case("scene-speech/revalidate-publication/" + reason, false, s => { }, s =>
            {
                Require(s.SpeechQueueResult && s.EnqueuedCanPublish != null, "Expected an actual captured production publication guard");
                int capturedChecks = s.TargetChecks;
                s.RuntimeStale = reason == "generation";
                s.EpochStale = reason == "epoch";
                s.TargetUnavailable = reason == "target";
                if (reason == "session") ProductionBlockHost.SetSceneSessionForFixture(2);
                bool backgroundCurrent = s.EnqueuedCanPublish();
                Require(s.TargetChecks == capturedChecks, "Background publication check must not resolve game objects");
                Require(backgroundCurrent == (reason == "target"), "Background must reject stale state but defer entity reads to game thread");
                s.OnMainThread = true;
                try { Require(!s.EnqueuedCanPublish(), "Delayed main-thread publication must reject stale/replaced/unavailable target"); }
                finally { s.OnMainThread = false; }
                Require(s.SpeechEnqueues == 1, "Rejected delayed publication must not re-enqueue or retry the reply");
            }, InvokeProductionSpeechQueueHelper);
    }

    private static async Task SceneLifecycleCases()
    {
        foreach (string operation in new[] { "primary-release", "hold-participants", "battle-suppression", "idle-timeout", "failure-release" })
        foreach (string state in new[] { "current", "generation", "session", "epoch" })
            await Case("scene-lifecycle/" + operation + "/" + state, false, s =>
            {
                s.RuntimeStale = state == "generation";
                s.EpochStale = state == "epoch";
                if (state == "session") ProductionBlockHost.SetSceneSessionForFixture(2);
            }, s =>
            {
                Require(s.LifecycleResult == (state == "current"), "Actual lifecycle delegate must reject stale request provenance");
                string[] expected = state != "current" ? Array.Empty<string>()
                    : operation == "hold-participants" ? new[] { "movement-suppression", "hold" }
                    : operation == "battle-suppression" ? new[] { "battle-suppression" }
                    : operation == "idle-timeout" ? new[] { "idle-timeout" } : new[] { "release" };
                Require(s.LifecycleOwnerCalls.SequenceEqual(expected),
                    "Old generation/session/epoch must not release, hold, suppress or re-arm a replacement scene participant");
            }, host =>
            {
                Scenario.Current.LifecycleResult = host.ExecuteSceneLifecycleDelegate(operation);
                return Task.CompletedTask;
            });
    }
