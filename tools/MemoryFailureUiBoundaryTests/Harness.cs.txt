        static void Case(string name,Action<MyBehavior,long> body)
        {
            TWParallel.MainId=Thread.CurrentThread.ManagedThreadId;Logger.Fail=false;InformationManager.Shown.Clear();InformationManager.OffThread=0;InformationManager.DuringShow=null;
            var owner=new MyBehavior();MyBehavior.Instance=owner;TaleWorlds.CampaignSystem.Campaign.Available=true;TaleWorlds.CampaignSystem.Campaign.Owner=owner;TaleWorlds.CampaignSystem.Campaign.OffThread=0;long gen=SaveRuntimeGuard.AdvanceGeneration("fixture");

        }
        static void Send(MyBehavior owner,long gen,string title="failure") => Task.Run(()=>owner.Notify(title,"original body",gen)).GetAwaiter().GetResult();
        static void Tick(MyBehavior o) {o.OnEngineTick();Check(o.OtherTickPhases>0,"remaining engine phases still execute");}
        static int Count=>InformationManager.Shown.Count;
        public static int Main()
        {
            Case("background",(o,g)=>{Send(o,g);Check(Count==0&&!o.Active,"background only publishes");Tick(o);Check(Count==1&&o.Active,"main tick displays once");Check(InformationManager.OffThread==0,"UI never off main");Check(TaleWorlds.CampaignSystem.Campaign.OffThread==0,"campaign accessed only on main");Check(InformationManager.Paused,"original pause behavior");Check(InformationManager.Shown[0].Button=="知道了"&&InformationManager.Shown[0].Positive&&!InformationManager.Shown[0].Negative,"original acknowledgement controls");Check(InformationManager.Shown[0].Title=="failure"&&InformationManager.Shown[0].Text=="original body","original content preserved");Tick(o);Check(Count==1,"single consumption");InformationManager.Shown[0].AffirmativeAction();Check(!o.Active,"ack releases active");});
