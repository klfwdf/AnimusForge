  internal static int RunCommitWriterTests(){
   TWParallel.MainId=Thread.CurrentThread.ManagedThreadId;var cases=new List<(string Name,Action Run)>{("actual-campaign-owner",ActualCampaignOwner),("editor-source-index-shift",()=>EditorSourceChanged(true)),("editor-source-body-update",()=>EditorSourceChanged(false)),("editor-worker-callback",EditorWorkerCallback),("editor-current-cancel",()=>EditorCancel(false)),("editor-stale-cancel",()=>EditorCancel(true)),("editor-active",()=>EditorActive(false,false)),("editor-invalidates-success",()=>EditorActive(true,false)),("editor-invalidates-failure",()=>EditorActive(true,true)),("editor-late-generation",()=>EditorLateSave(false)),("editor-late-owner",()=>EditorLateSave(true)),("partial-daily-reject-recent-published",DailyRefusedButRecentPublished),("ordinary-no-own-dedup",OrdinaryHasNoOwnDedup),("commit-invalidates-old-success",()=>CommitInvalidatesPendingSummary(false)),("commit-invalidates-old-failure",()=>CommitInvalidatesPendingSummary(true))};
   foreach(bool single in new[]{false,true}){var one=single;cases.Add(($"import-direct-{one}",()=>ImportNoDuplicate(one)));foreach(bool overwrite in new[]{false,true}){var ow=overwrite;cases.Add(($"import-normal-{one}-{ow}",()=>ImportNormal(one,ow)));foreach(bool replaceOwner in new[]{false,true}){var replace=replaceOwner;cases.Add(($"import-late-{one}-{replace}-{ow}",()=>ImportLate(one,replace,ow)));}}}
   foreach(bool nonHero in new[]{false,true})foreach(int scene in new[]{-1,7}){var n=nonHero;var s=scene;cases.Add(($"complete-{n}-{s}",()=>OrdinaryComplete(n,s)));}
   foreach(string mode in new[]{"worker","owner","empty","invalid-id","mismatched-kind","ineligible"}){var m=mode;cases.Add(("guard-"+m,()=>Guard(m)));}
   foreach(int nth in new[]{1,2,3}){var n=nth;cases.Add(("Daily-partial-"+n,()=>DailyPartial(n)));}
   foreach(bool existing in new[]{false,true})foreach(bool silent in new[]{false,true}){var e=existing;var s=silent;cases.Add(($"Recent-partial-{e}-{s}",()=>RecentPartial(e,s)));}
   foreach(string kind in new[]{"player","assistant","fact"}){var k=kind;cases.Add(("single-"+k,()=>SinglePayload(k)));}
   int failures=0;foreach(var c in cases){try{c.Run();Console.WriteLine("PASS "+c.Name);}catch(Exception ex){failures++;Console.WriteLine("FAIL "+c.Name+": "+ex.Message);}}
   Console.WriteLine($"COMMIT_WRITERS_RESULT scenarios={cases.Count} failures={failures} real_commit=true real_daily_recent=true exact_readback=true actual_editor_closure=true actual_import_apply=true game_save=NOT_TESTED");return failures==0?0:1;
