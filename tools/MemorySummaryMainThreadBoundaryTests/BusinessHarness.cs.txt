        // This is a scripted provider boundary, NOT the real Execute*/network/retry loop.
        // The incomplete gate forces the actual caller to cross an await on a worker.
        private async Task RunDailySummaryQueueItemsAsync(List<object> items, int burst,
            List<MemorySummaryExecutionResult> daily, List<MajorActionSummaryExecutionResult> major,
            List<MemoryOverviewExecutionResult> overview)
        {
            var batch = new ProviderBatch { Items = items };
            batches.Enqueue(batch);
            if (providerCompletesSynchronously) batch.Ready.TrySetResult(true);
            await batch.Ready.Task.ConfigureAwait(false);
            if (providerThrows) throw new InvalidOperationException("scripted provider failure");
            foreach (object item in items)
            {
                string key = JobKey(item);
                bool fail = failures.Contains(key), stale = obsolete.Contains(key);
                if (item is MemorySummaryJob d)
                    daily.Add(new MemorySummaryExecutionResult { Job = d,
#if HAS_INPUT_SOURCE
                        Source = FixtureSource(key),
#endif
                        Error = " provider-error ", IsObsolete = stale,
                        Block = fail ? null : Block(d.HeroId, d.GameDayIndex) });
                else if (item is MajorActionSummaryJob m)
                    major.Add(new MajorActionSummaryExecutionResult { Job = m,
#if HAS_INPUT_SOURCE
                        Source = FixtureSource(key),
#endif
                        Error = " provider-error ", IsObsolete = stale,
                        State = fail ? null : new MajorActionSummaryState { Summary = " major-result ", LastSummarizedDay = 10, LastSummarizedSequence = 1 } });
                else if (item is MemoryOverviewJob o)
                    overview.Add(new MemoryOverviewExecutionResult { Job = o,
#if HAS_INPUT_SOURCE
                        Source = FixtureSource(key),
#endif
                        Error = " provider-error ", IsObsolete = stale,
                        State = fail ? null : new MemoryOverviewState { Summary = " overview-result ", IncludedBlockIds = new List<string> { o.HeroId + "/10" } } });
            }
            if (includeMalformedResults)
            {
                daily.Add(null); daily.Add(new MemorySummaryExecutionResult());
                major.Add(null); major.Add(new MajorActionSummaryExecutionResult());
                overview.Add(null); overview.Add(new MemoryOverviewExecutionResult());
            }
        }

        private void ReleaseProviders()
        {
            foreach (var batch in batches) Task.Run(() => batch.Ready.TrySetResult(true)).GetAwaiter().GetResult();
        }
        private static void Until(Func<bool> done, Action progress = null)
        {
            var timer = Stopwatch.StartNew();
            while (!done())
            {
                if (timer.ElapsedMilliseconds > 5000) throw new Exception("Timed out waiting for controlled continuation");
                progress?.Invoke(); Thread.Sleep(1);
            }
        }
        private void Finish(Task task, bool releaseClock = true)
        {
            Until(() => task.IsCompleted, () =>
            {
                ReleaseProviders();
                if (releaseClock) Task.Run(() => clockGate.TrySetResult(true)).GetAwaiter().GetResult();
                Tick();
            });
            task.GetAwaiter().GetResult();
            Tick(); // A completed/retired request must not leave replayable work.
        }
        private void AwaitFirstCompletion(Task task)
        {
            // A faulted initial dispatcher may reach the provider asynchronously. Keep
            // releasing only the external provider gate, never pump game work here.
            ReleaseProviders(); Until(() => task.IsCompleted || !_memorySummaryMainThreadActions.IsEmpty, ReleaseProviders);
        }

        private static void SuccessAndReentry()
        {
            var b = NewOwner(); b.AddDaily("daily"); b.AddMajor("major"); b.AddOverview("overview");
            b._dirtyMemoryOverviewIds.Add("dirty");
            var task = b.Start(); b.AwaitFirstCompletion(task);
            bool untouched = applied == 0 && b._memorySummaryProcessing;
            b.Finish(task);
            Check(untouched, "business writes/release happened before a main-thread tick");
            Check(b._compressedMemoryBlocks["daily"].Single().Summary == "daily-result" && b._dailyMemoryDrafts["daily"].Count == 0, "daily block/draft mutation");
            Check(b._npcMajorActionSummaries["major"].Summary == "major-result" && b._npcMajorActionSummaries["major"].HeroId == "major", "major state normalization");
            Check(b._memoryOverviewStates["overview"].Summary == "overview-result" && b._memoryOverviewStates["overview"].HeroId == "overview", "overview state normalization");
            Check(applied == 3 && Witness.Count("weekly:") == 1 && Witness.Count("weekly-public:") == 1 && Witness.Count("publicity:") == 1 && Witness.Count("native-clear:") == 1, "duplicate or missing daily side effect");
            Check(Witness.Events.Where(x => x.Name.StartsWith("Apply")).Select(x => x.Name).SequenceEqual(
                new[] { "ApplyMemorySummarySuccess", "ApplyMajorActionSummarySuccess", "ApplyMemoryOverviewSuccess" }), "business application order changed");
            Check(Witness.Count("deferred-overview:dirty") == 1 && b._dirtyMemoryOverviewIds.Count == 0, "dirty overview handoff");
            Check(Witness.Count("ui:AnimusForge 日结压缩完成：对话记忆 1 个，重大履历 1 个，记忆总览 1 个。") == 1, "success UI counts");
            Check(!b._memorySummaryProcessing, "processing not released"); b.CleanQueues(); MainThreadEffects();
            int count = applied; var again = b.Start(); b.Finish(again);
            Check(applied == count && b.batches.Count == 1 && !b._memorySummaryProcessing, "empty reentry repeated provider/effects");
        }

        private static void ExtraWave(bool fail)
        {
            var b = NewOwner(); b.AddDaily("daily"); b.enqueueOverviewAfterDaily.Add("daily");

            var task = b.Start();

            b.Finish(task);

                !b._memoryOverviewStates.ContainsKey("overview-stale") && !b._memorySummaryProcessing,

        private static void BudgetWitness(bool synchronous = false)
        {
            var b = NewOwner(); b.providerCompletesSynchronously = synchronous;
            for (int i = 0; i < 12; i++) b.AddDaily("hero" + i);
            b.Finish(b.Start()); Check(applied == 12, "backlog lost or duplicated records");
            var counts = Witness.Events.Where(x => x.Main && x.Name.StartsWith("Apply")).GroupBy(x => x.Tick).Select(x => x.Count()).ToList();
            int peak = counts.DefaultIfEmpty(0).Max();
            Check(peak <= 2, "completion acceptance exceeds two actual Apply jobs per tick: " + peak);
            Console.WriteLine("BUDGET_OBSERVATION actual_apply_max_per_tick=" + peak + " (jobs only; not record/time budget acceptance)");
            MainThreadEffects();
        }
        private static void PartialCompletionException(string stage)
        {
            var b=NewOwner();var first=b.AddDaily("first");b.AddDaily("later");
            b.throwAfterBlockPublish=stage=="block";b.throwAfterWeeklyPublish=stage=="weekly";b.throwAfterRetryMark=stage=="mark";
            if(stage=="mark")b.failures.Add(JobKey(first));
            b.Finish(b.Start());
            Check(Witness.Count("popup-publish:压缩记忆总结异常")==1 && Witness.Count("ui:AnimusForge 日结压缩完成")==0,
                "partial "+stage+" error was silently rejected or reported complete");
            Check(Witness.Events.Any(x=>x.Name.Contains("任务可能已有部分写入")),"partial exception did not explain partial execution");
            Check(!b._memorySummaryProcessing&&b.batches.Count==1&&b._dailyMemoryDrafts["later"].Count==1&&!b._compressedMemoryBlocks.ContainsKey("later"),
                "partial exception lost release, replayed provider or applied later results");
            if(stage=="mark") {
                Check(first.RetryCount==3&&first.LastError=="provider-error"&&b._dailyMemoryDrafts["first"].Single().SummaryRetryCount==0,
                    "real Mark partial queue state was undone or remaining draft write continued");
                Check(applied==0&&Witness.Count("Mark")==1,"Mark exception was repeated or applied success");
            } else {
                Check(b._compressedMemoryBlocks["first"].Count==1&&applied==1,"published block was lost or replayed");
                Check(b._dailyMemoryDrafts["first"].Count==(stage=="block"?1:0),"source state does not match the actual failure boundary");
                Check(Witness.Count("weekly-public:")==(stage=="weekly"?1:0)&&Witness.Count("weekly:")==0,"weekly side effects continued/replayed after failure");
            }
            int effects=Witness.Events.Count;for(int i=0;i<5;i++)b.Tick();
            Check(Witness.Events.Count==effects,"extra ticks replayed stopped partial work");MainThreadEffects();
        }
        private void FinishAdmission()
        {
            Until(() => !_memorySummaryProcessing, () => { ReleaseProviders(); clockGate.TrySetResult(true); Tick(); });
            Tick();
        }
        private static void AdmissionDefersEligibility(bool maintenance)
        {
            var b = NewOwner(); b.AddDaily("daily");
            b.RunMemorySummaryMainThreadAsync(SaveRuntimeGuard.CaptureGeneration(), () => true).GetAwaiter().GetResult();
            b.RunMemorySummaryMainThreadAsync(SaveRuntimeGuard.CaptureGeneration(), () => true).GetAwaiter().GetResult();
            if (maintenance) b.TryRunCampaignMemoryMaintenance(); else b.TryStartMemorySummaryQueue();
            Check(b._memorySummaryProcessing && b.batches.Count == 0 && Witness.Count("eligibility") == 0,
                "admission rescanned sources before guarded planner could run");
            b.FinishAdmission(); Check(applied == 1 && b.batches.Count == 1, "admitted work did not complete once"); MainThreadEffects();
        }
        private static void TerminalDuplicateKeepsRunnable(string kind)
        {
            var b = NewOwner(); object live;
            if (kind == "daily") { live = b.AddDaily("daily"); b._memorySummaryQueue.Insert(0,new MemorySummaryJob { HeroId="daily",GameDayIndex=10,RetryCount=3 }); }
            else if (kind == "major") { live=b.AddMajor("major");b._npcMajorActionSummaryQueue.Insert(0,new MajorActionSummaryJob {HeroId="major",RetryCount=3}); }
            else { live=b.AddOverview("overview");b._memoryOverviewQueue.Insert(0,new MemoryOverviewJob {HeroId="overview",RetryCount=3}); }
            b.TryStartMemorySummaryQueue(); b.FinishAdmission();
            Check(b.batches.Count == 1 && b.batches.Single().Items.Count == 1 && ReferenceEquals(b.batches.Single().Items[0],live),
                "terminal duplicate masked the later runnable "+kind+" entry");
            Check(applied == 1, "runnable duplicate not applied");b.CleanQueues();MainThreadEffects();
        }

        private static int Main()
        {
            TWParallel.MainId = Thread.CurrentThread.ManagedThreadId;
            int passed = 0, failed = 0;
            var cases = new (string Name, Action Run)[]
            {
                ("success/reentry/side-effects", SuccessAndReentry), ("mixed-three-type-failures", MixedFailures),
                ("obsolete/null-results", ObsoleteAndMalformed), ("extra-overview-success", () => ExtraWave(false)),
                ("extra-overview-failure", () => ExtraWave(true)), ("owner-swap-queued", () => StaleQueued(true, false)),
                ("generation-change-queued", () => StaleQueued(false, false)), ("load-retirement", () => StaleQueued(false, true)),
                ("load-during-extra-delay", LoadDuringExtraDelay), ("provider-exception", ProviderException),
                ("empty-filtered-queues", EmptyFiltering), ("source-owner-mismatch", SourceOwnerMismatch),
                ("target-removed-after-await", TargetRemovedAfterAwait), ("actual-work-budget-observation", () => BudgetWitness()),
                ("synchronous-provider-job-budget", () => BudgetWitness(true)),
                ("same-generation-source-changed", () => RejectedSources(false)), ("missing-source-proof", () => RejectedSources(true)),
                ("changed-source-failure-discard", RejectedFailureSources), ("mixed-accepted-counts", MixedAcceptedCounts),
                ("admission-defers-eligibility",()=>AdmissionDefersEligibility(false)),("maintenance-defers-eligibility",()=>AdmissionDefersEligibility(true)),
                ("daily-terminal-duplicate",()=>TerminalDuplicateKeepsRunnable("daily")),("major-terminal-duplicate",()=>TerminalDuplicateKeepsRunnable("major")),("overview-terminal-duplicate",()=>TerminalDuplicateKeepsRunnable("overview")),
                ("invalid-only-no-maintenance-spin",InvalidOnlyDoesNotSpin),("force-scan-after-invalid-admission",ForceRescanAfterInvalidAdmission),
                ("empty-candidate-scan-throttle",EmptyRescanThrottle),("new-queue-entry-survives-cleanup",NewQueueEntrySurvivesCleanup),
                ("worker-admission-no-source-read",BackgroundAdmissionDoesNotFilter),("busy-admission-retains-work",BusyAdmissionPreservesQueue),
                ("partial-apply-block-exception",()=>PartialCompletionException("block")),
                ("partial-apply-weekly-exception",()=>PartialCompletionException("weekly")),
                ("partial-mark-exception",()=>PartialCompletionException("mark"))
            };
            foreach (var test in cases)
            {
                Witness.Reset(); SaveRuntimeGuard.AdvanceGeneration("case-" + test.Name);
                try { test.Run(); passed++; Console.WriteLine("PASS " + test.Name); }
                catch (Exception ex) { failed++; Console.WriteLine("FAIL " + test.Name + ": " + ex.Message); }
            }
            Console.WriteLine($"BUSINESS_RESULT passed={passed} failed={failed}"); return failed == 0 ? 0 : 1;
        }
