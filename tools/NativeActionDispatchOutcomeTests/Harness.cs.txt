    class ActionQueue
    {
        readonly Queue<Action> actions = new Queue<Action>();
        internal void Enqueue(Action action)
        {
            if (Host.ThrowEnqueue) throw new InvalidOperationException("queue unavailable");
            if (Host.RunBeforeQueueError) { Host.Main=true;action();Host.Main=false;throw new InvalidOperationException("queue failed after claim"); }
            actions.Enqueue(action);
            if (Host.ThrowAfterPublish) throw new InvalidOperationException("queue failed after publish");
        }
        internal int Count => actions.Count;
        internal void Pump() { Host.Main=true;while(actions.Count>0) { var a=actions.Dequeue();a();if(Host.DuplicateCallback)a(); }Host.Main=false; }
    }
