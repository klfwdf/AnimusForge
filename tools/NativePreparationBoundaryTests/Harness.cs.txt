public static class Program
{
    private static int Checks,Failures;
    private static void Check(bool value,string name) { Checks++;if(!value){Failures++;Console.WriteLine("FAIL "+name);} }
    private static string Finish(ShoutBehavior owner,Task<string> task)
    {
        var until=Stopwatch.StartNew();
        while(!task.IsCompleted&&until.ElapsedMilliseconds<3000)
        {
            while(owner._mainThreadActions.TryDequeue(out var work)) work();
            Thread.Sleep(1);
        }
        Check(task.IsCompleted,"completion bounded");return task.GetAwaiter().GetResult();
    }
    public static int Main()
    {
        for(int mode=0;mode<12;mode++)
        {
            foreach(bool hero in new[]{false,true}) foreach(int index in new[]{-1,7})
            {
                Host.Reset(mode);var owner=new ShoutBehavior();Hero h=hero?new():null;var c=new CharacterObject();
                string reference=Finish(owner,owner.BaselinePrepare(h,c,index));string referenceTrace=string.Join(";",Host.Trace);
                Host.Reset(mode);owner=new ShoutBehavior();string direct=Finish(owner,owner.Prepare(h,c,index));string directTrace=string.Join(";",Host.Trace);
                Check(direct==reference,"payload equals original");Check(directTrace==referenceTrace,"helper arguments/order equal original");
                Check(direct!=null,"healthy capture result");Check(Host.OffThread==0,"direct main-thread reads");Check(Host.Guards==1,"one context check");
                Host.Reset(mode);owner=new ShoutBehavior();string queued=Finish(owner,Task.Run(()=>owner.Prepare(h,c,index)));
                Check(queued==direct,"same payload direct/queued");Check(string.Join(";",Host.Trace)==directTrace,"same helper order/arguments");Check(Host.OffThread==0,"queued game reads only on main");Check(Host.Guards==1,"one queued context check");
            }
        }
        foreach(bool queued in new[]{false,true})
        {
            Host.Reset(1);Host.Current=false;var owner=new ShoutBehavior();var t=queued?Task.Run(()=>owner.Prepare(new Hero(),new CharacterObject(),7)):owner.Prepare(new Hero(),new CharacterObject(),7);
            Check(Finish(owner,t)==null,"stale returns no payload");Check(Host.Reads==0,"stale cannot read game or evaluate rules");
        }
        Host.Reset(1);var late=new ShoutBehavior();var pending=Task.Run(()=>late.Prepare(new Hero(),new CharacterObject(),7));
        Check(SpinWait.SpinUntil(()=>late._mainThreadActions.Count>0,2500),"published before retirement");Host.Current=false;
        Check(Finish(late,pending)==null,"revalidate at consumption");Check(Host.Reads==0,"retired queued request cannot read");
        Host.Reset(1);late=new ShoutBehavior();pending=Task.Run(()=>late.Prepare(new Hero(),new CharacterObject(),7));
        Check(SpinWait.SpinUntil(()=>pending.IsCompleted,2500),"unconsumed request expires");Check(pending.GetAwaiter().GetResult()==null,"timeout no snapshot");
        while(late._mainThreadActions.TryDequeue(out var callback))callback();Check(Host.Reads==0,"expired preparation never reads late");
        Console.WriteLine($"{(Failures==0?"PASS":"FAIL")} NativePreparation checks={Checks} failures={Failures} productionCapture=extracted queue=real-helper domainHelpers=recording-stubs LIVE=NOT_RUN");return Failures==0?0:1;
    }
