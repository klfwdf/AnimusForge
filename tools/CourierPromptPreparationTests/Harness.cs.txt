 static IEnumerable<string> Strings(JsonElement value){if(value.ValueKind==JsonValueKind.String){yield return value.GetString();yield break;}if(value.ValueKind==JsonValueKind.Object){foreach(var property in value.EnumerateObject())foreach(string text in Strings(property.Value))yield return text;}else if(value.ValueKind==JsonValueKind.Array){foreach(var item in value.EnumerateArray())foreach(string text in Strings(item))yield return text;}}
 static void PumpUntil(Func<bool> done,bool pump=true){var watch=Stopwatch.StartNew();while(!done()){if(pump)while(AnimusForge.CourierDeliveryBehavior.MainThreadActions.TryDequeue(out var action))action();if(watch.ElapsedMilliseconds>5000)throw new Exception("fixture wait timeout");Thread.Sleep(1);}}
 static void Case(bool inbound,string scenario){
  cases++;AnimusForge.Probe.Reset();while(AnimusForge.CourierDeliveryBehavior.MainThreadActions.TryDequeue(out _)){}

  AnimusForge.Probe.NullContext=scenario=="null_context";AnimusForge.Probe.NullHits=scenario=="null_hits";AnimusForge.Probe.Command=scenario=="commands";AnimusForge.Probe.EmptyExtras=scenario=="empty_extras";AnimusForge.Probe.EmptyHistory=scenario=="empty_history";AnimusForge.Probe.ThrowRouting=scenario=="routing_error";AnimusForge.Probe.ThrowAssembly=scenario=="assembly_error";AnimusForge.Probe.KnowledgeVariant=scenario.StartsWith("knowledge_text_")?scenario:null;
  if(scenario=="letter_seed")session.InboundIntentText=" ";if(scenario=="fallback_seed"){session.InboundIntentText=null;session.LetterText=null;}
  if(scenario=="null_input")session.LetterText=null;
  bool knowledgeAfter=scenario.EndsWith("_knowledge_after");bool after=scenario.EndsWith("_after");bool before=scenario.EndsWith("_before");
  if(knowledgeAfter){AnimusForge.Probe.BlockKnowledge=true;AnimusForge.Probe.KnowledgeRelease.Reset();}
  else if(after)AnimusForge.Probe.Release.Reset();
  string old=null;
  if(!after&&!before&&scenario!="timeout"&&scenario!="routing_error"&&scenario!="assembly_error"){
   AnimusForge.Probe.SyncAllowed=true;old=owner.Sync(inbound,true);var actual=owner.Sync(inbound,false);Check(old==actual,scenario+" synchronous field/message parity");AnimusForge.Probe.SyncAllowed=false;AnimusForge.Probe.Routing=AnimusForge.Probe.Contexts=AnimusForge.Probe.Assemblies=0;
  }
  Task<string> task=scenario=="main_caller"?owner.Start(session,inbound):Task.Run(()=>owner.Start(session,inbound));

  else if(after){PumpUntil(()=>AnimusForge.Probe.Entered.IsSet);Alter(owner,session,scenario);AnimusForge.Probe.Release.Set();}
  PumpUntil(()=>task.IsCompleted,scenario!="timeout");
  if(scenario=="timeout"){
   Check(task.IsCanceled,"timeout propagates");while(AnimusForge.CourierDeliveryBehavior.MainThreadActions.TryDequeue(out var action))action();Check(AnimusForge.Probe.Assemblies==0&&AnimusForge.Probe.Routing==0,"expired callback cannot execute");return;
