static class Program
{
    static int checks, failures;
    static void Check(bool value, string name)
    {
        checks++;
        if (!value) { failures++; Console.WriteLine("FAIL " + name); }
    }
    static int Main()
    {
        foreach (bool fallback in new[] { false, true })
        {
            CourierLetterInputPopup.UseFallback = fallback;
            var owner = CourierDeliveryBehavior.NewOwner();
            owner.Open("old"); var old = CourierLetterInputPopup.Last;
            owner.Open("new"); old.Send("old text");
            Check(owner.Sends == 0 && owner.PendingName == "new" && owner.InputOpen,
                "old-confirm-cannot-send-or-close-new-draft fallback=" + fallback);
            old.Cancel();
            Check(owner.PendingName == "new" && owner.InputOpen,
                "old-cancel-cannot-clear-new-draft fallback=" + fallback);
            CourierLetterInputPopup.Last.Send("current text");
            Check(owner.Sends == 1 && owner.LastRecipient == "new" && owner.PendingName == null,
                "current-draft-sends-once fallback=" + fallback);
            old.Send("late retry");
            Check(owner.Sends == 1, "late-retry-does-not-dispatch fallback=" + fallback);

            owner = CourierDeliveryBehavior.NewOwner(); owner.Open("retired"); old = CourierLetterInputPopup.Last;
            CourierDeliveryBehavior.NewOwner(); old.Send("retired text");
            Check(owner.Sends == 0, "replaced-owner-cannot-dispatch fallback=" + fallback);

            owner = CourierDeliveryBehavior.NewOwner(); owner.Open("old-save"); old = CourierLetterInputPopup.Last;
            SaveRuntimeGuard.Generation++; old.Send("old-save text");
            Check(owner.Sends == 0, "old-generation-cannot-dispatch fallback=" + fallback);

            owner = CourierDeliveryBehavior.NewOwner(); owner.Open("same-draft"); old = CourierLetterInputPopup.Last;
            owner.Reopen(); old.Send("old revision"); old.Cancel();
            Check(owner.Sends == 0 && owner.PendingName == "same-draft" && owner.InputOpen,
                "old-step-cannot-send-or-clear-same-draft fallback=" + fallback);
            CourierLetterInputPopup.Last.Send("new revision");
            Check(owner.Sends == 1, "current-revision-still-sends fallback=" + fallback);

            owner = CourierDeliveryBehavior.NewOwner(); owner.Open("retired-phase"); old = CourierLetterInputPopup.Last;
            owner.Retire(); old.Send("retired");
            Check(owner.Sends == 0, "sealed-owner-cannot-send fallback=" + fallback);

            owner = CourierDeliveryBehavior.NewOwner(); owner.Open("wrong-thread"); old = CourierLetterInputPopup.Last;
            TWParallel.OnMain = false; old.Send("worker"); TWParallel.OnMain = true;
            Check(owner.Sends == 0, "non-main-callback-cannot-send fallback=" + fallback);

            owner = CourierDeliveryBehavior.NewOwner(); owner.Open("partial"); old = CourierLetterInputPopup.Last;
            owner.ThrowAfterCreate = true; old.Send("partial effect"); old.Send("replay");
            Check(owner.Sends == 1 && owner.PendingName == null, "partial-dispatch-not-replayed fallback=" + fallback);

            owner = CourierDeliveryBehavior.NewOwner(); owner.Open("first"); old = CourierLetterInputPopup.Last;
            owner.AfterCreate = () => owner.Open("reentrant-new"); old.Send("first");
            Check(owner.Sends == 1 && owner.PendingName == "reentrant-new" && owner.InputOpen,
                "old-dispatch-tail-cannot-clear-reentrant-new-draft fallback=" + fallback);
        }
        foreach (string scenario in new[] { "dead", "unknown", "in-party", "active", "missing-party", "crew-short", "crew-wounded", "crew-xp", "empty-crew", "troop-ineligible", "prisoner-ineligible", "settlement-ineligible", "cargo-short", "cargo-source", "cargo-duplicate", "normal-with-cargo", "invalid-mode" })
        {
            var owner = CourierDeliveryBehavior.NewOwner(); owner.Open("validation"); owner.Change(scenario);
            CourierLetterInputPopup.Last.Send("validate before dispatch");
            Check(owner.Sends == 0, "dispatch-revalidates-" + scenario);
        }
        foreach (CourierPayloadMode mode in Enum.GetValues<CourierPayloadMode>())
        {
            var owner = CourierDeliveryBehavior.NewOwner(); owner.Open("mode"); owner.SetMode(mode);
            CourierLetterInputPopup.Last.Send("mode regression");
            Check(owner.Sends == 1, "valid-mode-" + mode);
        }
        {
            var owner = CourierDeliveryBehavior.NewOwner(); owner.Open("reply"); owner.Change("unknown-reply");
            CourierLetterInputPopup.Last.Send("reply exception");
            Check(owner.Sends == 1, "unknown-recipient-letter-reply-exception-preserved");
        }
#if COURIER_DRAFT_TICKETS
        {
            var owner=CourierDeliveryBehavior.NewOwner();
            Check(CourierDeliveryBehavior.IssueModuleCourierTicket("a")==null,"no-draft-no-ticket");
            owner.Open("ticket");
            TWParallel.OnMain=false;
            Check(CourierDeliveryBehavior.IssueModuleCourierTicket("a")==null,"capture-main-thread-only");
            TWParallel.OnMain=true;
            string ticket=CourierDeliveryBehavior.IssueModuleCourierTicket("a");
            Check(ticket!=null && owner.Sends==0 && owner.InputOpen,"capture-does-not-claim-or-send");
            Check(!owner.SendTicket("b",ticket),"wrong-client-cannot-consume-ticket");
            var ui=CourierLetterInputPopup.Last;
            Check(owner.SendTicket("a",ticket) && owner.Sends==1,"right-client-dispatches-prepared-draft");
            ui.Send("late ui");ui.Cancel();
            Check(owner.Sends==1 && !owner.SendTicket("a",ticket),"api-wins-ui-race-and-replay");

            owner=CourierDeliveryBehavior.NewOwner();owner.Open("ui-first");
            ticket=CourierDeliveryBehavior.IssueModuleCourierTicket("a");CourierLetterInputPopup.Last.Send("ui first");
            Check(!owner.SendTicket("a",ticket) && owner.Sends==1,"ui-wins-api-race");

            foreach(string invalidation in new[]{"draft","revision","generation","owner","recipient","retire","revoke"})
            {
                owner=CourierDeliveryBehavior.NewOwner();owner.Open("old");ticket=CourierDeliveryBehavior.IssueModuleCourierTicket("a");
                switch(invalidation)
                {
                    case "draft":owner.Open("new");break;
                    case "revision":owner.Reopen();break;
                    case "generation":SaveRuntimeGuard.Generation++;break;
                    case "owner":CourierDeliveryBehavior.NewOwner();break;
                    case "recipient":owner.ReplaceRecipient();break;
                    case "retire":owner.Retire();break;
                    case "revoke":CourierDeliveryBehavior.RevokeModuleCourierTickets("a");break;
                }
                Check(!owner.SendTicket("a",ticket) && owner.Sends==0,"ticket-invalidated-by-"+invalidation);
            }
            owner=CourierDeliveryBehavior.NewOwner();owner.Open("bounded");
            int issued=0;for(int i=0;i<128;i++)if(CourierDeliveryBehavior.IssueModuleCourierTicket("a")!=null)issued++;
            Check(issued==128 && CourierDeliveryBehavior.IssueModuleCourierTicket("b")==null,"owner-ticket-capacity-shared-across-clients");
            CourierDeliveryBehavior.RevokeModuleCourierTickets("b");
            Check(CourierDeliveryBehavior.IssueModuleCourierTicket("b")==null,"revoke-is-client-scoped");
            CourierDeliveryBehavior.RevokeModuleCourierTickets("a");
            Check(CourierDeliveryBehavior.IssueModuleCourierTicket("b")!=null,"revoke-releases-bounded-storage");
            owner.Open("replacement");
            Check(owner.TicketCount==0,"draft-step-releases-live-ticket-references");
            owner.MarkUnready();
            Check(CourierDeliveryBehavior.IssueModuleCourierTicket("a")==null,"unfinished-draft-cannot-be-captured");
        }
#endif
        Console.WriteLine($"CourierDraftAdmission checks={checks} failures={failures}; production UI callbacks; fixture game dispatch; LIVE=NOT_RUN");
        return failures == 0 ? 0 : 1;
    }
}

    internal void Open(string name)
    {
        _pendingFlow=new PendingCourierFlow{Recipient=new Hero{Name=name}};
        typeof(PendingCourierFlow).GetField("RuntimeGeneration")?.SetValue(_pendingFlow,SaveRuntimeGuard.Generation);
        var crew=new CharacterObject();

        MobileParty.MainParty=new(){MemberRoster=new(){Entries=new(){new(){Character=crew,Number=3,WoundedNumber=1,Xp=20}}}};

    internal void SetMode(CourierPayloadMode mode)
    {
        _pendingFlow.Mode=mode;_pendingFlow.SelectedEntries.Clear();_options.Clear();

        string kind=mode switch {CourierPayloadMode.Give=>"item",CourierPayloadMode.Show=>"show_item",CourierPayloadMode.GiveTroops=>"troop",CourierPayloadMode.GivePrisoners=>"prisoner",_=>"settlement"};

            case "crew-short":MobileParty.MainParty.MemberRoster.Entries[0].Number=1;break;

            case "invalid-mode":_pendingFlow.Mode=(CourierPayloadMode)99;break;
