 static void Case(string kind) {
  Probe.OffThread=0;AnimusForge.InformationManager.Refreshed=0;AnimusForge.SaveRuntimeGuard.Generation=1;
  var hero=new Hero("hero");var o=new AnimusForge.MyBehavior();AnimusForge.MyBehavior.Instance=o;AnimusForge.MyBehavior.RegisteredHero=hero;
  bool ui=kind.StartsWith("ui_");bool reroll=kind.StartsWith("reroll")||ui;o.Seed(reroll?"oldP":kind=="partial_p"?"keptP":"",reroll?"oldB":kind=="partial_b"?"keptB":"");
  if(kind=="complete")o.Seed("oldP","oldB");
  if(kind=="ui_reset_before")o.UseAllowance();
  var t=ui?o.RerollUi(hero):kind.StartsWith("external")?Task.Run(()=>o.External(hero)):Task.Run(()=>o.Generate(kind=="null"?null:hero,reroll));if(kind=="ui_reset_before")o.ResetGenerationState();
  Pump(o,()=>o.Calls>0||t.IsCompleted);
  if(o.Calls>0){
   if(kind=="owner")AnimusForge.MyBehavior.Instance=new();
   if(kind=="generation")AnimusForge.SaveRuntimeGuard.Generation++;
   if(kind=="target")AnimusForge.MyBehavior.RegisteredHero=new Hero("hero");
   if(kind=="reset" || kind=="reroll_reset" || kind=="ui_reset")o.ResetGenerationState();
   if(kind=="reroll_edit" || kind=="ui_edit")o.Seed("editedP","editedB");
   if(kind=="normal_edit")o.Seed("editedP","");
   if(kind=="reroll_voice")o.Seed("oldP","oldB","newVoice");
   o.Release((kind=="parse" || kind=="external_failure" || kind=="ui_failure")?"bad":kind=="one_field"?"P|":"P|B",kind!="http",kind=="throw");
  }
  if(kind=="ui_reset_accept"){
   var deadline=DateTime.UtcNow.AddSeconds(8);while(!o.Pending&&!t.IsCompleted){if(DateTime.UtcNow>deadline)throw new Exception("accept not queued");Thread.Sleep(1);}
   o.ResetGenerationState();
  }
  Pump(o,()=>t.IsCompleted);string failure=t.GetAwaiter().GetResult();var v=o.Values();
  Check(Probe.OffThread==0,kind+" all game/config/profile reads and writes on main");
  bool noSave=kind is "owner" or "generation" or "target" or "reset" or "reroll_reset" or "reroll_edit" or "parse" or "http" or "throw" or "complete" or "null" or "external_failure" or "ui_failure" or "ui_edit" or "ui_reset" or "ui_reset_before" or "ui_reset_accept";
  Check(o.Saves==(noSave?0:1),kind+" save count");
  if(!noSave){Check(v[0]==(kind=="partial_p"?"keptP":kind=="normal_edit"?"editedP":"P"),kind+" personality");Check(v[1]==(kind=="partial_b"?"keptB":kind=="one_field"?"P":"B"),kind+" background");Check(v[2]==(kind=="reroll_voice"?"newVoice":"voice"),kind+" voice preserved");}
  if(ui)Check(AnimusForge.InformationManager.Refreshed==(noSave?0:1),kind+" truthful UI success");
  if(kind=="reroll_reset")Check(!string.IsNullOrWhiteSpace(failure),"reset must not report reroll success");
  if(kind=="reroll_edit")Check(v[0]=="editedP"&&v[1]=="editedB"&&!string.IsNullOrWhiteSpace(failure),"reroll protects intervening edit");
  if(kind is "parse" or "http" or "throw")Check(!string.IsNullOrWhiteSpace(failure)&&o.Cooling("hero"),kind+" failure + cooldown");
  if(o.Calls>0)Check(o.SentSystem.EndsWith("custom-requirements")&&o.SentUser.Contains("facts:hero"),kind+" original prompt inputs");
