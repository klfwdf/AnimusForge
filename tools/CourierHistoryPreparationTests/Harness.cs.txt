internal static class Program
{
    private static int checks;
    private static void Check(bool ok,string name){if(!ok)throw new Exception("FAIL "+name);checks++;}
    private static void PumpUntil(Func<bool> ready,bool pump=true)
    {
        var clock=Stopwatch.StartNew();while(!ready()){
            if(pump)while(AnimusForge.CourierDeliveryBehavior.MainThreadActions.TryDequeue(out var action))action();
            if(clock.ElapsedMilliseconds>5000)throw new Exception("FAIL fixture wait timeout");Thread.Sleep(1);
        }
    }
    private static void Case(bool inbound,string scenario)
    {
        AnimusForge.HistoryProbe.Reset();
        while(AnimusForge.CourierDeliveryBehavior.MainThreadActions.TryDequeue(out _)) { }
        var session=new AnimusForge.CourierDeliveryBehavior.CourierSession{Inbound=inbound};
        var owner=new AnimusForge.CourierDeliveryBehavior{Session=session};AnimusForge.CourierDeliveryBehavior.Instance=owner;
        if(scenario=="empty")AnimusForge.HistoryProbe.SourceText="   ";
        if(scenario=="capture_error")AnimusForge.HistoryProbe.ThrowCapture=true;
        if(scenario=="resolve_error")AnimusForge.HistoryProbe.ThrowResolve=true;
        bool hold=scenario is "snapshot" or "replace_after" or "participant_after" or "generation_after" or "owner_after" or "terminal_after";
        if(hold)AnimusForge.HistoryProbe.ResolveRelease.Reset();
        Task<string[]> task=Task.Run(()=>owner.Start(session,session.Participant,inbound));
        if(scenario is "replace_before" or "generation_before" or "owner_before" or "terminal_before"){
            PumpUntil(()=>!AnimusForge.CourierDeliveryBehavior.MainThreadActions.IsEmpty,false);
            if(scenario=="replace_before")owner.Session=new(){Inbound=inbound};
            if(scenario=="generation_before")AnimusForge.SaveRuntimeGuard.Generation++;
            if(scenario=="owner_before")AnimusForge.CourierDeliveryBehavior.Instance=new();
            if(scenario=="terminal_before")session.Terminal=true;
        }
        if(hold){
            PumpUntil(()=>AnimusForge.HistoryProbe.ResolveEntered.IsSet);
            if(scenario=="snapshot"){session.Fact="new-fact";AnimusForge.HistoryProbe.SourceText="new-history";}
            if(scenario=="replace_after")owner.Session=new(){Inbound=inbound};
            if(scenario=="participant_after")session.Participant=new();
            if(scenario=="generation_after")AnimusForge.SaveRuntimeGuard.Generation++;
            if(scenario=="owner_after")AnimusForge.CourierDeliveryBehavior.Instance=new();
            if(scenario=="terminal_after")session.Terminal=true;
            AnimusForge.HistoryProbe.ResolveRelease.Set();
        }
        PumpUntil(()=>task.IsCompleted,scenario!="timeout");
        if(scenario=="timeout"){
            Check(task.IsCanceled,"same-owner timeout not propagated");
            while(AnimusForge.CourierDeliveryBehavior.MainThreadActions.TryDequeue(out var action))action();
            Check(AnimusForge.HistoryProbe.Captures==0,"timed-out queued callback executed later");return;
        }
        var result=task.GetAwaiter().GetResult();
        bool stale=scenario.Contains("before") || scenario.Contains("after");
        Check((result==null)==stale,scenario+" context acceptance "+inbound);
        Check(AnimusForge.HistoryProbe.WrongThread==0,scenario+" thread isolation");
        if(stale){Check(AnimusForge.HistoryProbe.Captures==(scenario.Contains("before")?0:1),scenario+" stale source counts");return;}
        string fact="original-fact:"+(inbound?"inbound:False":"reply:True");
        Check(result[0]==fact && AnimusForge.HistoryProbe.Fact==fact,"frozen delivery fact direction");
        Check(AnimusForge.HistoryProbe.Input==(inbound?null:"letter") && !AnimusForge.HistoryProbe.IncludeScene,"history inputs/scene exclusion unchanged");
        Check(result[1]==(scenario is "empty" or "capture_error" or "resolve_error"?"":"history"),scenario+" frozen/empty history text");
        Check(AnimusForge.HistoryProbe.Captures==1 && AnimusForge.HistoryProbe.Resolves==(scenario=="capture_error"?0:1),"exactly one capture/resolve, empty does not recapture");
    }
    private static int Main()
    {
        TaleWorlds.Library.TWParallel.MainThread=Environment.CurrentManagedThreadId;
        try {
            foreach(bool inbound in new[]{false,true})foreach(string scenario in new[]{"normal","empty","snapshot","capture_error","resolve_error","replace_before","generation_before","owner_before","terminal_before","replace_after","participant_after","generation_after","owner_after","terminal_after","timeout"})Case(inbound,scenario);
            foreach(bool inbound in new[]{false,true}) {
                AnimusForge.HistoryProbe.Reset();var session=new AnimusForge.CourierDeliveryBehavior.CourierSession{Inbound=inbound};
                var result=AnimusForge.CourierDeliveryBehavior.Legacy(session,inbound);
                Check(result[1]=="history" && AnimusForge.HistoryProbe.Input==(inbound?null:"letter"),"legacy synchronous capture compatibility");
                Check(AnimusForge.HistoryProbe.Fact=="original-fact:"+(inbound?"inbound:False":"reply:True"),"legacy delivery fact parity");
            }
            Console.WriteLine($"PASS {checks} Courier history assertions / 30 scenarios; actual owner-phase + new helper, physical thread pump, history/game seams are fixtures.");return 0;
        }catch(Exception ex){Console.Error.WriteLine("FAIL CourierHistory "+ex);return 1;}
        finally{AnimusForge.HistoryProbe.ResolveRelease.Set();}
    }
