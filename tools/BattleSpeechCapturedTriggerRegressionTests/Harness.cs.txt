public static class Program {

  foreach(string state in new[]{"new-ui","epoch","session","save-generation","player","consumed","off-main"})await Case("late natural success rejects "+state,async()=>{var h=new ShoutBehavior();var input=Input(h);var s=new Subject();var c=new Classifier();var task=s.Start(c,input);switch(state){case"new-ui":h.OpenNewUi();break;case"epoch":h.NextEpoch();break;case"session":h.NextSession();break;case"save-generation":SaveRuntimeGuard.Generation++;break;case"player":Agent.Main=new();break;case"consumed":h.Consume(input.OriginalScenePlayerShoutRequest);break;case"off-main":h.MainThread=false;break;}await Complete(s,c,task);Assert(s.Sessions==0&&s.Prepared==0&&s.Replays==0,"stale callback started/replayed speech");});
  await Case("last action boundary rechecks after phase resolution",async()=>{var h=new ShoutBehavior();var s=new Subject{PhaseHook=h.OpenNewUi};var c=new Classifier();var task=s.Start(c,Input(h));await Complete(s,c,task);Assert(s.Sessions==0&&s.Prepared==0,"phase callback revived old input");});
