    class Program
    {
        static int checks;
        static void Check(bool value, string name) { if (!value) throw new Exception("FAIL " + name); checks++; Console.WriteLine("PASS " + name); }
        static void PumpUntil(Func<bool> done)
        {
            Stopwatch watch = Stopwatch.StartNew();
            while (!done()) { ShoutBehavior.Pump(); if (watch.ElapsedMilliseconds > 5000) throw new Exception("FAIL timed out"); Thread.Sleep(1); }
            ShoutBehavior.Pump();
        }
        static string Finish(Task<string> task) { PumpUntil(() => task.IsCompleted); return task.GetAwaiter().GetResult(); }
        static ShoutBehavior.NativeConversationPresentationScope Scope(bool submit = true)
        {
            var scope = ShoutBehavior.CaptureNativeConversationPresentationScopeForOverlay();
            if (submit) Check(Finish(ShoutBehavior.SubmitNativeConversationForOverlayAsync(scope, "text", null, null, null, null, false)) == "A:0", "scope bound to real internal legacy-chain admission");
            return scope;
        }
        static void QueueStream(AnimusForgeNativeConversationOverlay ui, bool opening = false)
            => Task.Run(() => ui.Stream("reply for A", opening)).GetAwaiter().GetResult();
        static void Main()
        {
            Host.Reset(); ConversationHelper.Reset(); var scope = Scope();
            Check(!ShoutBehavior.CurrentInstance.SlotExists && scope.IsCurrent(), "observation survives backend completion without retaining execution slot");
            var ui = new AnimusForgeNativeConversationOverlay(scope); QueueStream(ui); ui.Drain();
            Check(ConversationHelper.Text == "reply for A" && ui.Busy, "valid late stream still renders");
            Check(ui.Finish() && !ui.Busy && ConversationHelper.Ends == 1, "valid final cleanup releases local busy and its stream");
            var reused = ShoutBehavior.SubmitNativeConversationForOverlayAsync(scope, "again", null, null, null, null, false);
            try { Finish(reused); throw new Exception("FAIL reused scope accepted"); } catch (ShoutBehavior.NativeConversationAdmissionException ex) { Check(ex.ReasonCode == "native.presentation_scope_reused", "scope is single-submission observation"); }
            Check(!Task.Run(() => scope.IsCurrent()).GetAwaiter().GetResult(), "off-main scope probe cannot read game state");

            foreach (bool opening in new[] { false, true })
            {
                Host.Reset(); scope = Scope(); ui = new AnimusForgeNativeConversationOverlay(scope); ConversationHelper.Reset("B native text");
                QueueStream(ui, opening); Host.Target = new Hero { StringId = "B" }; ui.Drain();
                Check(ConversationHelper.Writes == 0 && ConversationHelper.Text == "B native text", "queued old reply cannot render on another eligible target opening=" + opening);
                Check(!ui.Busy && ui.Generation == 2, "target mismatch retires only old UI submission opening=" + opening);
                Check(ConversationHelper.Ends == 0 && !ui.Finish(), "stale finally cannot end newer stream opening=" + opening);
            }
            foreach (string change in new[] { "load", "end", "token", "manager", "mission", "instance" })
            {
                Host.Reset(); scope = Scope(); ui = new AnimusForgeNativeConversationOverlay(scope); ConversationHelper.Reset();
                if (change == "load") SaveRuntimeGuard.Generation++;
                if (change == "end") ShoutBehavior.InvalidateNativeConversationAdmissionOnConversationEnd();
                if (change == "token") Campaign.Current.ConversationManager.ActiveToken++;
                if (change == "manager") Campaign.Current.ConversationManager = new ConversationManager();
                if (change == "mission") Mission.Current = new Mission();
                if (change == "instance") ShoutBehavior.CurrentInstance = new ShoutBehavior();
                int reads = Host.GameReads; ui.CheckStamp();
                Check(!ui.Busy && ConversationHelper.Writes == 0 && ConversationHelper.Ends == 0, "idle stale stamp releases old controls without global display effects " + change);
                Check(reads == Host.GameReads, "stamp check adds no target/agent enumeration " + change);
            }
            Host.Reset(); scope = Scope(); ui = new AnimusForgeNativeConversationOverlay(scope); ConversationHelper.Reset(); QueueStream(ui);
            Check(Finish(ShoutBehavior.SubmitNativeConversationTextForExternalAsync("other client", null, null, null, null)) == "A:0", "later admission uses same full legacy entry");
            ui.Drain(); Check(ConversationHelper.Writes == 0 && !ui.Busy, "same-context later admission invalidates previous presentation revision");

            Host.Reset(); scope = Scope(); ui = new AnimusForgeNativeConversationOverlay(scope); ConversationHelper.Reset();
            SaveRuntimeGuard.Generation++;
            Check(!ui.Finish() && !ui.Busy && ConversationHelper.Ends == 0, "stale completion without an earlier callback releases local state but cannot end global stream");

            Host.Reset(); scope = Scope(); ui = new AnimusForgeNativeConversationOverlay(scope); ConversationHelper.Reset();
            ui.EnqueueNotice(); Campaign.Current.ConversationManager.ActiveToken++; InformationManager.Notices = 0; ui.FlushNotice();
            Check(InformationManager.Notices == 0, "queued postprocess notice revalidates captured context when flushed");
            ui.Dots(); Check(ConversationHelper.Writes == 0 && !ui.Busy, "waiting animation cannot replace a new conversation line");

            Host.Reset(); scope = Scope(); ui = new AnimusForgeNativeConversationOverlay(scope); ConversationHelper.Reset();
            ui.EnqueueNotice(); InformationManager.Notices = 0; ui.FlushNotice(); Check(InformationManager.Notices == 1, "valid postprocess notice preserved");
            ui.Dots(); Check(ConversationHelper.Writes == 1, "valid waiting animation preserved");
            ui.CloseForTest(); QueueStream(ui); ui.Drain(); Check(ConversationHelper.Writes == 1 && !ui.Finish(), "closed overlay has no late display authority");

            Host.Reset(); scope = Scope(false); ui = new AnimusForgeNativeConversationOverlay(scope); ConversationHelper.Reset();
            Check(scope.IsCurrent() && !ShoutBehavior.CurrentInstance.SlotExists, "origin capture does not reserve backend or require a submitted request");
            Check(ui.Finish() && ConversationHelper.Ends == 1, "valid pre-admission rejection can restore its own UI");
            Console.WriteLine($"NativePresentation PASS={checks} FAIL=0 realScope=true realQueuedStreamCallbacks=2 realUiHelpers=true downstream=STUBBED LIVE=NOT_RUN");
        }
